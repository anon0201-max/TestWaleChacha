    23→// ==================== ADMIN LOGIN ====================
    24→function AdminLogin() {
    25→  const { setAdminData } = useAppStore();
    26→  const [username, setUsername] = useState('');
    27→  const [password, setPassword] = useState('');
    28→  const [error, setError] = useState('');
    29→  const [loading, setLoading] = useState(false);
    30→  const [showPassword, setShowPassword] = useState(false);
    31→  const [changePasswordOpen, setChangePasswordOpen] = useState(false);
    32→  const [cpCurrentPassword, setCpCurrentPassword] = useState('');
    33→  const [cpNewPassword, setCpNewPassword] = useState('');
    34→  const [cpConfirmPassword, setCpConfirmPassword] = useState('');
    35→  const [cpLoading, setCpLoading] = useState(false);
    36→  const [cpMsg, setCpMsg] = useState<{ type: 'success' | 'error'; text: string } | null>(null);
    37→
    38→  async function handleLogin() {
    39→    setLoading(true);
    40→    setError('');
    41→    try {
    42→      const res = await fetch('/api/admin/login', {
    43→        method: 'POST',
    44→        headers: { 'Content-Type': 'application/json' },
    45→        body: JSON.stringify({ username, password }),
    46→      });
    47→      const data = await res.json();
    48→      if (data.success) {
    49→        setAdminData({ isLoggedIn: true });
    50→      } else {
    51→        setError(data.error || 'Login failed');
    52→      }
    53→    } catch { setError('Server error'); }
    54→    setLoading(false);
    55→  }
    56→
    57→  async function handleChangePassword() {
    58→    setCpMsg(null);
    59→    if (!cpCurrentPassword || !cpNewPassword || !cpConfirmPassword) {
    60→      setCpMsg({ type: 'error', text: 'All fields are required' });
    61→      return;
    62→    }
    63→    if (cpNewPassword !== cpConfirmPassword) {
    64→      setCpMsg({ type: 'error', text: 'New password and confirm password do not match' });
    65→      return;
    66→    }
    67→    if (cpNewPassword.length < 4) {
    68→      setCpMsg({ type: 'error', text: 'New password must be at least 4 characters' });
    69→      return;
    70→    }
    71→    setCpLoading(true);
    72→    try {
    73→      const res = await fetch('/api/admin/reset-password', {
    74→        method: 'POST',
    75→        headers: { 'Content-Type': 'application/json' },
    76→        body: JSON.stringify({ currentPassword: cpCurrentPassword, newPassword: cpNewPassword }),
    77→      });
    78→      const data = await res.json();
    79→      if (data.success) {
    80→        setCpMsg({ type: 'success', text: 'Password updated successfully! Use new password to login next time.' });
    81→        setCpCurrentPassword('');
    82→        setCpNewPassword('');
    83→        setCpConfirmPassword('');
    84→      } else {
    85→        setCpMsg({ type: 'error', text: data.error || 'Failed to update password' });
    86→      }
    87→    } catch {
    88→      setCpMsg({ type: 'error', text: 'Server error. Please try again.' });
    89→    }
    90→    setCpLoading(false);
    91→  }
    92→
    93→  return (
    94→    <div className="min-h-[70vh] flex items-center justify-center px-4 pt-12">
    95→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="w-full">
    96→        <Card className="w-full max-w-md mx-auto shadow-xl border-0">
    97→          <CardContent className="p-5 sm:p-6">
    98→            <div className="text-center mb-6">
    99→              <div className="w-16 h-16 rounded-2xl bg-blue-700 flex items-center justify-center mx-auto mb-4">
   100→                <Shield className="w-8 h-8 text-white" />
   101→              </div>
   102→              <h2 className="text-xl font-bold text-gray-900">Admin Login</h2>
   103→              <p className="text-xs text-muted-foreground mt-1">Enter credentials to manage tests & questions</p>
   104→            </div>
   105→            <div className="space-y-4">
   106→              <div>
   107→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Username</Label>
   108→                <Input value={username} onChange={(e) => setUsername(e.target.value)} placeholder="admin" className="h-11" />
   109→              </div>
   110→              <div>
   111→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Password</Label>
   112→                <div className="relative">
   113→                  <Input
   114→                    type={showPassword ? 'text' : 'password'}
   115→                    value={password}
   116→                    onChange={(e) => setPassword(e.target.value)}
   117→                    placeholder="••••••••"
   118→                    className="h-11 pr-10"
   119→                    onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
   120→                  />
   121→                  <button
   122→                    type="button"
   123→                    onClick={() => setShowPassword(!showPassword)}
   124→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   125→                  >
   126→                    {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   127→                  </button>
   128→                </div>
   129→              </div>
   130→              <p className="text-right">
   131→                <button onClick={() => setChangePasswordOpen(true)} className="text-xs text-blue-600 hover:text-blue-700 font-medium">Change Password?</button>
   132→              </p>
   133→              {error && (
   134→                <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="bg-red-50 border border-red-200 text-red-700 text-sm rounded-xl p-3 flex items-center gap-2">
   135→                  <AlertTriangle className="w-4 h-4 shrink-0" /> {error}
   136→                </motion.div>
   137→              )}
   138→              <Button className="w-full bg-blue-700 hover:bg-blue-800 h-11 font-semibold" onClick={handleLogin} disabled={loading || !username || !password}>
   139→                {loading ? 'Logging in...' : 'Login to Admin Panel'}
   140→              </Button>
   141→            </div>
   142→          </CardContent>
   143→        </Card>
   144→
   145→        {/* Admin Change Password Dialog */}
   146→        <Dialog open={changePasswordOpen} onOpenChange={(open) => { if (!open) { setChangePasswordOpen(false); setCpMsg(null); setCpCurrentPassword(''); setCpNewPassword(''); setCpConfirmPassword(''); } }}>
   147→          <DialogContent className="max-w-[calc(100vw-2rem)] sm:max-w-sm">
   148→            <DialogHeader>
   149→              <DialogTitle className="flex items-center gap-2 text-base"><Shield className="w-5 h-5 text-blue-600" />Change Admin Password</DialogTitle>
   150→              <DialogDescription className="text-xs">Enter current password, then set new password</DialogDescription>
   151→            </DialogHeader>
   152→            <div className="space-y-3">
   153→              {cpMsg && (
   154→                <motion.div initial={{ opacity: 0, y: -5 }} animate={{ opacity: 1, y: 0 }}
   155→                  className={`p-3 rounded-xl text-sm flex items-center gap-2 border ${
   156→                    cpMsg.type === 'success' ? 'bg-green-50 border-green-200 text-green-700' : 'bg-red-50 border-red-200 text-red-700'
   157→                  }`}>
   158→                  {cpMsg.type === 'success' ? <CheckCircle2 className="w-4 h-4 shrink-0" /> : <AlertTriangle className="w-4 h-4 shrink-0" />}
   159→                  {cpMsg.text}
   160→                </motion.div>
   161→              )}
   162→              <div>
   163→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Current Password</Label>
   164→                <Input type="password" value={cpCurrentPassword} onChange={(e) => setCpCurrentPassword(e.target.value)} placeholder="Enter current password" className="h-11" />
   165→              </div>
   166→              <div>
   167→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">New Password</Label>
   168→                <Input type="password" value={cpNewPassword} onChange={(e) => setCpNewPassword(e.target.value)} placeholder="Enter new password (min 4 chars)" className="h-11" />
   169→              </div>
   170→              <div>
   171→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Confirm New Password</Label>
   172→                <Input type="password" value={cpConfirmPassword} onChange={(e) => setCpConfirmPassword(e.target.value)} placeholder="Confirm new password" className="h-11" onKeyDown={(e) => e.key === 'Enter' && handleChangePassword()} />
   173→              </div>
   174→              <div className="flex gap-2">
   175→                <DialogClose asChild>
   176→                  <Button variant="outline" className="flex-1 h-11">Cancel</Button>
   177→                </DialogClose>
   178→                <Button className="flex-1 h-11 bg-blue-700 hover:bg-blue-800 font-semibold" onClick={handleChangePassword} disabled={cpLoading || !cpCurrentPassword || !cpNewPassword || !cpConfirmPassword}>
   179→                  {cpLoading ? <><Loader2 className="w-4 h-4 mr-2 animate-spin" />Updating...</> : <><Save className="w-4 h-4 mr-2" />Update Password</>}
   180→                </Button>
   181→              </div>
   182→            </div>
   183→          </DialogContent>
   184→        </Dialog>
   185→      </motion.div>
   186→    </div>
   187→  );
   188→}

   420→// ==================== CATEGORIES TAB ====================
   421→function AdminCategoriesTab({ onRefresh }: { onRefresh: () => void }) {
   422→  const { categories, setCategories } = useAppStore();
   423→  const [newName, setNewName] = useState('');
   424→  const [newSlug, setNewSlug] = useState('');
   425→  const [newExamType, setNewExamType] = useState('General');
   426→  const [adding, setAdding] = useState(false);
   427→
   428→  async function handleAdd() {
   429→    if (!newName || !newSlug) return;
   430→    setAdding(true);
   431→    await fetch('/api/admin/categories', {
   432→      method: 'POST',
   433→      headers: { 'Content-Type': 'application/json' },

   842→  function fillQuestionsFromExtract(extracted: QuestionForm[]): number {
   843→    if (!extracted || extracted.length === 0) {
   844→      setFillStatus({ msg: '⚠️ No questions received from VLM to fill.', type: 'error' });
   845→      return 0;
   846→    }
   847→    const count = extracted.length;
   848→    setFillStatus({ msg: `🔧 Filling ${count} questions into boxes...`, type: 'info' });
   849→
   850→    // Use ref to track where to start filling (supports multi-image upload)
   851→    let startIdx = fillStartIndexRef.current;
   852→
   853→    setQuestions(prev => {
   854→      const next = [...prev];
   855→      // Clamp startIdx if it's beyond array length
   856→      if (startIdx > next.length) startIdx = next.length;
   857→      // Fill each extracted question into the boxes
   858→      for (let k = 0; k < extracted.length; k++) {
   859→        const targetIdx = startIdx + k;
   860→        if (targetIdx < next.length) {
   861→          // Replace entire box with extracted data (not merge — merge keeps empty defaults)
   862→          next[targetIdx] = { ...extracted[k] };
   863→        } else {
   864→          next.push(extracted[k]);
   865→        }
   866→      }
   867→      return next;
   868→    });
   869→
   870→    // Advance the fill index for next image batch
   871→    fillStartIndexRef.current = startIdx + count;
   872→
   873→    setTimeout(() => {
   874→      setFillStatus({ msg: `✅ ${count} questions filled into boxes!`, type: 'success' });
   875→      setTimeout(() => setFillStatus(null), 8000);
   876→    }, 100);
   877→    return count;

   909→            continue;
   910→          }

   921→            const newQ: QuestionForm = {
   922→              question: data.question.question || '', optionA: data.question.optionA || '', optionB: data.question.optionB || '', optionC: data.question.optionC || '', optionD: data.question.optionD || '',
   923→              correctOption: data.question.correctOption || 'A', explanation: data.question.explanation || '', section: data.question.section || 'General', negativeMark: data.question.negativeMark || '0',
   924→            };

   926→            totalExtracted += filled;
   927→            toast.success(`Image ${i + 1}: 1 question extracted & filled!`);

   932→            toast.error(`Image ${i + 1}: ${errMsg}`);

   940→            setFillStatus({ msg: `❌ ${err?.message || 'Network error. Check your connection.'}`, type: 'error' });
   941→          }
