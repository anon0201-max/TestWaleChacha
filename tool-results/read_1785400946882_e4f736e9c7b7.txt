   941→  function fillQuestionsFromExtract(extracted: QuestionForm[]): number {
   942→    if (!extracted || extracted.length === 0) {
   943→      setFillStatus({ msg: '⚠️ No questions received from VLM to fill.', type: 'error' });
   944→      return 0;
   945→    }
   946→    const count = extracted.length;
   947→    setFillStatus({ msg: `🔧 Filling ${count} questions into boxes...`, type: 'info' });
   948→
   949→    // Use ref to track where to start filling (supports multi-image upload)
   950→    let startIdx = fillStartIndexRef.current;
   951→
   952→    setQuestions(prev => {
   953→      const next = [...prev];
   954→      // Clamp startIdx if it's beyond array length
   955→      if (startIdx > next.length) startIdx = next.length;
   956→      // Fill each extracted question into the boxes
   957→      for (let k = 0; k < extracted.length; k++) {
   958→        const targetIdx = startIdx + k;
   959→        if (targetIdx < next.length) {
   960→          // Replace entire box with extracted data (not merge — merge keeps empty defaults)
   961→          next[targetIdx] = { ...extracted[k] };
   962→        } else {
   963→          next.push(extracted[k]);
   964→        }
   965→      }
   966→      return next;
   967→    });
   968→
   969→    // Advance the fill index for next image batch
   970→    fillStartIndexRef.current = startIdx + count;
   971→
   972→    setTimeout(() => {
   973→      setFillStatus({ msg: `✅ ${count} questions filled into boxes!`, type: 'success' });
   974→      setTimeout(() => setFillStatus(null), 8000);
   975→    }, 100);
   976→    return count;
   977→  }

   979→  async function handleImageExtract(e: React.ChangeEvent<HTMLInputElement>) {
   980→    const files = e.target.files;
   981→    if (!files || files.length === 0) return;
   982→    const fileList = Array.from(files);

   987→      setExtractElapsed(Math.floor((Date.now() - startTime) / 1000));
   988→    }, 1000);
   989→    try {
   990→      let totalExtracted = 0;

   992→      for (let i = 0; i < fileList.length; i++) {
   993→        const file = fileList[i];
   994→        const formData = new FormData();

   996→        const controller = new AbortController();
   997→        const timeoutId = setTimeout(() => controller.abort(), 300000);
   998→        try {

  1000→          // Use Next.js API route with z-ai-web-dev-sdk VLM + Grok fallback
  1001→          const res = await fetch('/api/admin/extract-question', { method: 'POST', body: formData, signal: controller.signal });
  1002→          clearTimeout(timeoutId);
  1003→          if (!res.ok) {
  1004→            const errText = await res.text().catch(() => 'Unknown error');
  1005→            setFillStatus({ msg: `❌ Image ${i + 1}: Server error (HTTP ${res.status}). ${errText.substring(0, 80)}`, type: 'error' });
  1006→            toast.error(`Image ${i + 1}: Server error ${res.status}`);
  1007→            hasError = true;
  1008→            continue;
  1009→          }
  1010→          const data = await res.json();
  1011→          if (data.success && data.questions && data.questions.length > 0) {
  1012→            const newQuestions: QuestionForm[] = data.questions.map((q: any) => ({
  1013→              question: q.question || '', optionA: q.optionA || '', optionB: q.optionB || '', optionC: q.optionC || '', optionD: q.optionD || '',
  1014→              correctOption: q.correctOption || 'A', explanation: q.explanation || '', section: q.section || 'General', negativeMark: q.negativeMark || '0',
  1015→            }));
  1016→            const filled = fillQuestionsFromExtract(newQuestions);
  1017→            totalExtracted += filled;
  1018→            toast.success(`Image ${i + 1}: ${filled} questions extracted & filled!`);
  1019→          } else if (data.success && data.question) {
  1020→            const newQ: QuestionForm = {
  1021→              question: data.question.question || '', optionA: data.question.optionA || '', optionB: data.question.optionB || '', optionC: data.question.optionC || '', optionD: data.question.optionD || '',
  1022→              correctOption: data.question.correctOption || 'A', explanation: data.question.explanation || '', section: data.question.section || 'General', negativeMark: data.question.negativeMark || '0',
  1023→            };
  1024→            const filled = fillQuestionsFromExtract([newQ]);
  1025→            totalExtracted += filled;
  1026→            toast.success(`Image ${i + 1}: 1 question extracted & filled!`);
  1027→          } else {
  1028→            hasError = true;
  1029→            const errMsg = data.error || 'Failed to extract questions';
  1030→            setFillStatus({ msg: `❌ ${errMsg}`, type: 'error' });
  1031→            toast.error(`Image ${i + 1}: ${errMsg}`);
  1032→          }
  1033→        } catch (err: any) {
  1034→          clearTimeout(timeoutId);
  1035→          hasError = true;
  1036→          if (err?.name === 'AbortError') {
  1037→            setFillStatus({ msg: '❌ Timed out (5 min). Try a smaller/clearer image.', type: 'error' });
  1038→          } else {
  1039→            setFillStatus({ msg: `❌ ${err?.message || 'Network error. Check your connection.'}`, type: 'error' });
  1040→          }
  1041→        }
  1042→      }
  1043→      if (totalExtracted > 0) {
  1044→        setFillStatus({ msg: `✅ Done! Total ${totalExtracted} questions extracted & filled.`, type: 'success' });
  1045→        toast.success(`🎉 Total: ${totalExtracted} questions extracted!`);
  1046→      } else if (hasError) {

  1050→        toast.error('No questions could be extracted. Empty boxes remain for manual entry.');
  1051→      }
  1052→    } finally {
  1053→      clearInterval(timerInterval);
  1054→      setExtractingImage(false);
  1055→      setExtractElapsed(0);
  1056→      e.target.value = '';
  1057→    }
  1058→  }
