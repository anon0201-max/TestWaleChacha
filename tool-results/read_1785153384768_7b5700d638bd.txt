    22→// ==================== ADMIN LOGIN ====================
    23→function AdminLogin() {
    24→  const { setAdminData } = useAppStore();
    25→  const [username, setUsername] = useState('');
    26→  const [password, setPassword] = useState('');
    27→  const [error, setError] = useState('');
    28→  const [loading, setLoading] = useState(false);
    29→  const [showPassword, setShowPassword] = useState(false);
    30→
    31→  async function handleLogin() {
    32→    setLoading(true);
    33→    setError('');
    34→    try {
    35→      const res = await fetch('/api/admin/login', {
    36→        method: 'POST',
    37→        headers: { 'Content-Type': 'application/json' },
    38→        body: JSON.stringify({ username, password }),
    39→      });
    40→      const data = await res.json();
    41→      if (data.success) {
    42→        setAdminData({ isLoggedIn: true });
    43→      } else {
    44→        setError(data.error || 'Login failed');
    45→      }
    46→    } catch { setError('Server error'); }
    47→    setLoading(false);
    48→  }
    49→
    50→  return (
    51→    <div className="min-h-[70vh] flex items-center justify-center">
    52→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
    53→        <Card className="w-full max-w-sm shadow-xl border-0">
    54→          <CardContent className="p-6">
    55→            <div className="text-center mb-6">
    56→              <div className="w-16 h-16 rounded-2xl bg-blue-700 flex items-center justify-center mx-auto mb-4">
    57→                <Shield className="w-8 h-8 text-white" />
    58→              </div>
    59→              <h2 className="text-xl font-bold text-gray-900">Admin Login</h2>
    60→              <p className="text-xs text-muted-foreground mt-1">Enter credentials to manage tests & questions</p>
    61→            </div>
    62→            <div className="space-y-4">
    63→              <div>
    64→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Username</Label>
    65→                <Input value={username} onChange={(e) => setUsername(e.target.value)} placeholder="admin" className="h-11" />
    66→              </div>
    67→              <div>
    68→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Password</Label>
    69→                <div className="relative">
    70→                  <Input
    71→                    type={showPassword ? 'text' : 'password'}
    72→                    value={password}
    73→                    onChange={(e) => setPassword(e.target.value)}
    74→                    placeholder="••••••••"
    75→                    className="h-11 pr-10"
    76→                    onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
    77→                  />
    78→                  <button
    79→                    type="button"
    80→                    onClick={() => setShowPassword(!showPassword)}
    81→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
    82→                  >
    83→                    {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
    84→                  </button>
    85→                </div>
    86→              </div>
    87→              {error && (
    88→                <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="bg-red-50 border border-red-200 text-red-700 text-sm rounded-xl p-3 flex items-center gap-2">
    89→                  <AlertTriangle className="w-4 h-4 shrink-0" /> {error}
    90→                </motion.div>
    91→              )}
    92→              <Button className="w-full bg-blue-700 hover:bg-blue-800 h-11 font-semibold" onClick={handleLogin} disabled={loading || !username || !password}>
    93→                {loading ? 'Logging in...' : 'Login to Admin Panel'}
    94→              </Button>
    95→            </div>
    96→          </CardContent>
    97→        </Card>
    98→      </motion.div>
    99→    </div>
   100→  );
   101→}

   333→// ==================== CATEGORIES TAB ====================
   334→function AdminCategoriesTab({ onRefresh }: { onRefresh: () => void }) {

   337→  const [newSlug, setNewSlug] = useState('');
   338→  const [newExamType, setNewExamType] = useState('General');
   339→  const [adding, setAdding] = useState(false);
   340→
   341→  async function handleAdd() {
   342→    if (!newName || !newSlug) return;
   343→    setAdding(true);
   344→    await fetch('/api/admin/categories', {
   345→      method: 'POST',
   346→      headers: { 'Content-Type': 'application/json' },
   347→      body: JSON.stringify({ name: newName, slug: newSlug, examType: newExamType }),
   348→    });
   349→    setNewName(''); setNewSlug('');
   350→    setAdding(false);
   351→    onRefresh();
   352→  }
   353→
   354→  async function handleDelete(id: string) {
   355→    if (!confirm('Delete this category and ALL its tests & questions? This cannot be undone.')) return;

   357→      method: 'DELETE',
   358→      headers: { 'Content-Type': 'application/json' },
   359→      body: JSON.stringify({ id }),
   360→    });
   361→    onRefresh();
   362→  }
   363→
   364→  return (
   365→    <div className="space-y-4">
   366→      {/* Add Category */}
   367→      <Card className="border-0 shadow-sm">
   368→        <CardContent className="p-5">

   429→    </div>
   430→  );

   433→// ==================== TESTS TAB ====================
   434→function AdminTestsTab({ onRefresh }: { onRefresh: () => void }) {
   435→  const { tests } = useAppStore();
   436→  const [expandedTestId, setExpandedTestId] = useState<string | null>(null);
   437→  const [testQuestions, setTestQuestions] = useState<Record<string, unknown[]>>({});
   438→  const [loadingQ, setLoadingQ] = useState<string | null>(null);
   439→
   440→  async function fetchQuestions(testId: string) {
   441→    if (testQuestions[testId]) {
   442→      setExpandedTestId(expandedTestId === testId ? null : testId);
   443→      return;
   444→    }
   445→    setLoadingQ(testId);
   446→    try {
   447→      const res = await fetch(`/api/tests/${testId}`);
   448→      if (res.ok) {
   449→        const data = await res.json();
   450→        setTestQuestions(prev => ({ ...prev, [testId]: data.questions || [] }));
   451→        setExpandedTestId(testId);
   452→      }
   453→    } catch {}
   454→    setLoadingQ(null);
   455→  }
   456→
   457→  async function handleDeleteTest(id: string) {
   458→    if (!confirm('Delete this test and ALL its questions? This cannot be undone.')) return;
   459→    await fetch('/api/admin/tests', {
   460→      method: 'DELETE',
   461→      headers: { 'Content-Type': 'application/json' },

   774→  // Fill existing empty boxes with extracted data — finds first empty box dynamically.

   776→  function fillQuestionsFromExtract(extracted: QuestionForm[]): number {
   777→    if (!extracted || extracted.length === 0) {
   778→      console.warn('⚠️ fillQuestionsFromExtract: no questions to fill');
   779→      setFillStatus({ msg: '⚠️ No questions received from VLM to fill.', type: 'error' });
   780→      return 0;
   781→    }
   782→    console.log(`🔧 fillQuestionsFromExtract: START — filling ${extracted.length} questions`);
   783→    console.log(`🔧 First extracted question:`, extracted[0]?.question?.substring(0, 60));
   784→    setFillStatus({ msg: `🔧 Filling ${extracted.length} questions into boxes...`, type: 'info' });
   785→
   786→    let filledCount = 0;
   787→    setQuestions(prev => {
   788→      console.log(`🔧 setQuestions updater — prev count: ${prev.length}`);
   789→      const next = [...prev];
   790→
   791→      // Find first empty box and fill sequentially from there
   792→      let startIdx = -1;
   793→      for (let i = 0; i < next.length; i++) {
   794→        if (!next[i].question.trim()) {
   795→          startIdx = i;
   796→          break;
   797→        }
   798→      }
   799→      console.log(`🔧 First empty box at index: ${startIdx}`);
   800→
   801→      if (startIdx === -1) {
   802→        // No empty boxes — append all
   803→        console.log(`🔧 No empty boxes — appending ${extracted.length}`);
   804→        for (const ex of extracted) {
   805→          next.push(ex);
   806→          filledCount++;
   807→        }
   808→      } else {
   809→        for (let k = 0; k < extracted.length; k++) {
   810→          const targetIdx = startIdx + k;
   811→          if (targetIdx < next.length) {
   812→            next[targetIdx] = { ...next[targetIdx], ...extracted[k] };
   813→          } else {
   814→            next.push(extracted[k]);
   815→          }
   816→          filledCount++;
   817→        }
   818→        console.log(`🔧 Filled boxes ${startIdx}–${startIdx + extracted.length - 1}`);
   819→      }
   820→
   821→      console.log(`🔧 AFTER fill — total: ${next.length}, Box[0]:`, next[0]?.question?.substring(0, 60));
   822→      return next;
   823→    });
   824→
   825→    // Show success status (slight delay so it appears after state update)

   828→      // Auto-clear after 8 seconds
   829→      setTimeout(() => setFillStatus(null), 8000);
