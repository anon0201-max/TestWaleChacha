    22→// ==================== ADMIN LOGIN ====================
    23→function AdminLogin() {
    24→  const { setAdminData } = useAppStore();
    25→  const [username, setUsername] = useState('');
    26→  const [password, setPassword] = useState('');
    27→  const [error, setError] = useState('');
    28→  const [loading, setLoading] = useState(false);
    29→  const [showPassword, setShowPassword] = useState(false);
    30→
    31→  async function handleLogin() {
    32→    setLoading(true);
    33→    setError('');
    34→    try {
    35→      const res = await fetch('/api/admin/login', {
    36→        method: 'POST',
    37→        headers: { 'Content-Type': 'application/json' },
    38→        body: JSON.stringify({ username, password }),
    39→      });
    40→      const data = await res.json();
    41→      if (data.success) {
    42→        setAdminData({ isLoggedIn: true });
    43→      } else {
    44→        setError(data.error || 'Login failed');
    45→      }
    46→    } catch { setError('Server error'); }
    47→    setLoading(false);
    48→  }
    49→
    50→  return (
    51→    <div className="min-h-[70vh] flex items-center justify-center">
    52→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
    53→        <Card className="w-full max-w-sm shadow-xl border-0">
    54→          <CardContent className="p-6">
    55→            <div className="text-center mb-6">
    56→              <div className="w-16 h-16 rounded-2xl bg-blue-700 flex items-center justify-center mx-auto mb-4">
    57→                <Shield className="w-8 h-8 text-white" />
    58→              </div>
    59→              <h2 className="text-xl font-bold text-gray-900">Admin Login</h2>
    60→              <p className="text-xs text-muted-foreground mt-1">Enter credentials to manage tests & questions</p>
    61→            </div>
    62→            <div className="space-y-4">
    63→              <div>
    64→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Username</Label>
    65→                <Input value={username} onChange={(e) => setUsername(e.target.value)} placeholder="admin" className="h-11" />
    66→              </div>
    67→              <div>
    68→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Password</Label>
    69→                <div className="relative">
    70→                  <Input
    71→                    type={showPassword ? 'text' : 'password'}
    72→                    value={password}
    73→                    onChange={(e) => setPassword(e.target.value)}
    74→                    placeholder="••••••••"
    75→                    className="h-11 pr-10"
    76→                    onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
    77→                  />
    78→                  <button
    79→                    type="button"
    80→                    onClick={() => setShowPassword(!showPassword)}
    81→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
    82→                  >
    83→                    {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
    84→                  </button>
    85→                </div>
    86→              </div>
    87→              {error && (
    88→                <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="bg-red-50 border border-red-200 text-red-700 text-sm rounded-xl p-3 flex items-center gap-2">
    89→                  <AlertTriangle className="w-4 h-4 shrink-0" /> {error}
    90→                </motion.div>
    91→              )}
    92→              <Button className="w-full bg-blue-700 hover:bg-blue-800 h-11 font-semibold" onClick={handleLogin} disabled={loading || !username || !password}>
    93→                {loading ? 'Logging in...' : 'Login to Admin Panel'}
    94→              </Button>
    95→            </div>
    96→          </CardContent>
    97→        </Card>
    98→      </motion.div>
    99→    </div>
   100→  );
   101→}

   313→// ==================== CATEGORIES TAB ====================
   314→function AdminCategoriesTab({ onRefresh }: { onRefresh: () => void }) {
   315→  const { categories, setCategories } = useAppStore();
   316→  const [newName, setNewName] = useState('');
   317→  const [newSlug, setNewSlug] = useState('');
   318→  const [newExamType, setNewExamType] = useState('General');
   319→  const [adding, setAdding] = useState(false);
   320→
   321→  async function handleAdd() {
   322→    if (!newName || !newSlug) return;
   323→    setAdding(true);
   324→    await fetch('/api/admin/categories', {
   325→      method: 'POST',
   326→      headers: { 'Content-Type': 'application/json' },
   327→      body: JSON.stringify({ name: newName, slug: newSlug, examType: newExamType }),
   328→    });
   329→    setNewName(''); setNewSlug('');
   330→    setAdding(false);
   331→    onRefresh();
   332→  }
   333→
   334→  async function handleDelete(id: string) {
   335→    if (!confirm('Delete this category and ALL its tests & questions? This cannot be undone.')) return;
   336→    await fetch('/api/admin/categories', {
   337→      method: 'DELETE',
   338→      headers: { 'Content-Type': 'application/json' },
   339→      body: JSON.stringify({ id }),
   340→    });
   341→    onRefresh();
   342→  }
   343→
   344→  return (
   345→    <div className="space-y-4">
   346→      {/* Add Category */}
   347→      <Card className="border-0 shadow-sm">
   348→        <CardContent className="p-5">
   349→          <h3 className="text-sm font-semibold mb-3 flex items-center gap-2"><Plus className="w-4 h-4 text-blue-600" /> Add New Category</h3>
   350→          <div className="grid grid-cols-1 sm:grid-cols-4 gap-3">
   351→            <div className="sm:col-span-2">
   352→              <Label className="text-xs">Category Name</Label>

   374→        </CardContent>
   375→      </Card>
   376→
   377→      {/* Categories List */}
   378→      <Card className="border-0 shadow-sm">
   379→        <CardContent className="p-5">
   380→          <h3 className="text-sm font-semibold mb-3">All Categories ({categories.length})</h3>

   382→            {categories.map((cat) => (

   392→                    </div>
   393→                  </div>
   394→                </div>
   395→                <Button
   396→                  variant="ghost"
   397→                  size="icon"
   398→                  className="h-8 w-8 text-red-500 hover:text-red-700 hover:bg-red-50 opacity-0 group-hover:opacity-100 transition-opacity"

   400→                >
   401→                  <Trash2 className="w-4 h-4" />
   402→                </Button>
   403→              </div>
   404→            ))}
   405→            {categories.length === 0 && <p className="text-sm text-muted-foreground text-center py-6">No categories yet</p>}
   406→          </div>
   407→        </CardContent>
   408→      </Card>
   409→    </div>
   410→  );

   413→// ==================== TESTS TAB ====================
   414→function AdminTestsTab({ onRefresh }: { onRefresh: () => void }) {
   415→  const { tests } = useAppStore();
   416→  const [expandedTestId, setExpandedTestId] = useState<string | null>(null);
   417→  const [testQuestions, setTestQuestions] = useState<Record<string, unknown[]>>({});
   418→  const [loadingQ, setLoadingQ] = useState<string | null>(null);
   419→
   420→  async function fetchQuestions(testId: string) {
   421→    if (testQuestions[testId]) {
   422→      setExpandedTestId(expandedTestId === testId ? null : testId);
   423→      return;
   424→    }
   425→    setLoadingQ(testId);
   426→    try {
   427→      const res = await fetch(`/api/tests/${testId}`);
   428→      if (res.ok) {
   429→        const data = await res.json();
   430→        setTestQuestions(prev => ({ ...prev, [testId]: data.questions || [] }));
   431→        setExpandedTestId(testId);
   432→      }
   433→    } catch {}
   434→    setLoadingQ(null);
   435→  }
   436→
   437→  async function handleDeleteTest(id: string) {
   438→    if (!confirm('Delete this test and ALL its questions? This cannot be undone.')) return;
   439→    await fetch('/api/admin/tests', {
   440→      method: 'DELETE',
   441→      headers: { 'Content-Type': 'application/json' },
   442→      body: JSON.stringify({ id }),
   443→    });
   444→    onRefresh();
   445→  }
   446→
   447→  async function handleDeleteQuestion(questionId: string, testId: string) {
   448→    if (!confirm('Delete this question?')) return;
   449→    await fetch('/api/admin/tests/questions', {
   450→      method: 'DELETE',
   451→      headers: { 'Content-Type': 'application/json' },
   452→      body: JSON.stringify({ questionId }),
   453→    });
   454→    // Refetch
   455→    const res = await fetch(`/api/tests/${testId}`);
   456→    if (res.ok) {
   457→      const data = await res.json();
   458→      setTestQuestions(prev => ({ ...prev, [testId]: data.questions || [] }));
   459→    }
   460→    onRefresh();
   461→  }
   462→
   463→  return (
   464→    <div className="space-y-3">
   465→      <div className="flex items-center justify-between">
   466→        <h3 className="text-sm font-semibold">All Tests ({tests.length})</h3>
   467→        <Badge variant="secondary" className="text-[10px]">{tests.reduce((s, t) => s + t.totalQuestions, 0)} total questions</Badge>
   468→      </div>
   469→
   470→      {tests.length === 0 ? (
   471→        <Card className="border-0 shadow-sm">
   472→          <CardContent className="p-10 text-center">
   473→            <FileText className="w-10 h-10 text-muted-foreground mx-auto mb-3" />
   474→            <p className="text-sm text-muted-foreground">No tests yet. Go to &quot;Create Test&quot; tab to add your first test.</p>
   475→          </CardContent>
   476→        </Card>

   754→  // Fill existing empty boxes with extracted data — finds first empty box dynamically.
   755→  // Uses a DIRECT state replacement approach for maximum reliability.
   756→  function fillQuestionsFromExtract(extracted: QuestionForm[]): number {
   757→    if (!extracted || extracted.length === 0) {
   758→      console.warn('⚠️ fillQuestionsFromExtract: no questions to fill');
   759→      setFillStatus({ msg: '⚠️ No questions received from VLM to fill.', type: 'error' });
   760→      return 0;
   761→    }
   762→    console.log(`🔧 fillQuestionsFromExtract: START — filling ${extracted.length} questions`);
   763→    console.log(`🔧 First extracted question:`, extracted[0]?.question?.substring(0, 60));
   764→    setFillStatus({ msg: `🔧 Filling ${extracted.length} questions into boxes...`, type: 'info' });
   765→
   766→    let filledCount = 0;
   767→    setQuestions(prev => {
   768→      console.log(`🔧 setQuestions updater — prev count: ${prev.length}`);
   769→      const next = [...prev];
   770→
   771→      // Find first empty box and fill sequentially from there
   772→      let startIdx = -1;
   773→      for (let i = 0; i < next.length; i++) {
   774→        if (!next[i].question.trim()) {
   775→          startIdx = i;
   776→          break;
   777→        }
   778→      }
   779→      console.log(`🔧 First empty box at index: ${startIdx}`);
   780→
   781→      if (startIdx === -1) {
   782→        // No empty boxes — append all
   783→        console.log(`🔧 No empty boxes — appending ${extracted.length}`);
   784→        for (const ex of extracted) {
   785→          next.push(ex);
   786→          filledCount++;
   787→        }
   788→      } else {
   789→        for (let k = 0; k < extracted.length; k++) {
   790→          const targetIdx = startIdx + k;
   791→          if (targetIdx < next.length) {
   792→            next[targetIdx] = { ...next[targetIdx], ...extracted[k] };
   793→          } else {
   794→            next.push(extracted[k]);
   795→          }
   796→          filledCount++;
   797→        }
   798→        console.log(`🔧 Filled boxes ${startIdx}–${startIdx + extracted.length - 1}`);
   799→      }
   800→
   801→      console.log(`🔧 AFTER fill — total: ${next.length}, Box[0]:`, next[0]?.question?.substring(0, 60));
   802→      return next;
   803→    });
   804→
   805→    // Show success status (slight delay so it appears after state update)
   806→    setTimeout(() => {
   807→      setFillStatus({ msg: `✅ ${extracted.length} questions filled into boxes!`, type: 'success' });
   808→      // Auto-clear after 8 seconds
   809→      setTimeout(() => setFillStatus(null), 8000);
   810→    }, 100);
   811→
   812→    return filledCount;
   813→  }

   815→  async function handleImageExtract(e: React.ChangeEvent<HTMLInputElement>) {
   816→    const files = e.target.files;
   817→    if (!files || files.length === 0) return;
   818→
   819→    const fileList = Array.from(files);
   820→    setExtractingImage(true);
   821→    setExtractElapsed(0);
   822→
   823→    // Start a timer to show elapsed seconds
   824→    const startTime = Date.now();
   825→    const timerInterval = setInterval(() => {
   826→      setExtractElapsed(Math.floor((Date.now() - startTime) / 1000));
   827→    }, 1000);
   828→
   829→    try {
   830→      let totalExtracted = 0;
   831→      let hasError = false;
   832→
   833→      // Process each image file sequentially
   834→      for (let i = 0; i < fileList.length; i++) {
   835→        const file = fileList[i];
   836→        const formData = new FormData();
   837→        formData.append('image', file);
   838→
   839→        // Use AbortController with 5-minute timeout per file (VLM can take 2-3 min for large images)
   840→        const controller = new AbortController();
   841→        const timeoutId = setTimeout(() => controller.abort(), 300000);
   842→
   843→        try {
   844→          console.log(`📤 Uploading image ${i + 1}/${fileList.length}: ${file.name}`);
   845→          setFillStatus({ msg: `📤 Uploading image to VLM... (this can take 1-3 minutes)`, type: 'info' });
   846→          const res = await fetch('/api/admin/extract-question', {
   847→            method: 'POST',
   848→            body: formData,
   849→            signal: controller.signal,
   850→          });
   851→          clearTimeout(timeoutId);
   852→
   853→          console.log(`📡 Response status: ${res.status}`);
   854→          if (!res.ok) {
   855→            const errText = await res.text().catch(() => 'Unknown error');
   856→            console.error(`❌ HTTP ${res.status}:`, errText.substring(0, 200));
   857→            setFillStatus({ msg: `❌ VLM error (HTTP ${res.status}): ${errText.substring(0, 100)}`, type: 'error' });
   858→            toast.error(`Image ${i + 1}: VLM returned HTTP ${res.status}`);
   859→            hasError = true;
   860→            continue;
   861→          }
   862→          const data = await res.json();
   863→          console.log(`📋 Extraction result:`, { success: data.success, count: data.count, hasQuestions: !!data.questions, error: data.error });
   864→
   865→          if (data.success && data.questions && data.questions.length > 0) {
   866→            console.log(`🎉 VLM returned ${data.questions.length} questions — mapping to QuestionForm...`);
   867→            const newQuestions: QuestionForm[] = data.questions.map((q: any) => ({
   868→              question: q.question || '',
   869→              optionA: q.optionA || '',
   870→              optionB: q.optionB || '',
   871→              optionC: q.optionC || '',
   872→              optionD: q.optionD || '',
   873→              correctOption: q.correctOption || 'A',
   874→              explanation: q.explanation || '',
   875→              section: q.section || 'General',
   876→              negativeMark: q.negativeMark || '0',
   877→            }));
   878→            console.log(`🎉 Mapped ${newQuestions.length} questions. First:`, newQuestions[0]);
   879→
   880→            // FILL existing empty boxes with extracted data
   881→            const filled = fillQuestionsFromExtract(newQuestions);
   882→            totalExtracted += newQuestions.length;
   883→            console.log(`✅ Fill complete — ${filled} boxes filled (total extracted so far: ${totalExtracted})`);
   884→            toast.success(`Image ${i + 1}: ${newQuestions.length} questions extracted & filled!`);
   885→          } else if (data.success && data.question) {
   886→            // Fallback: single question
   887→            const newQ: QuestionForm = {
   888→              question: data.question.question || '',
   889→              optionA: data.question.optionA || '',
   890→              optionB: data.question.optionB || '',
   891→              optionC: data.question.optionC || '',
   892→              optionD: data.question.optionD || '',
   893→              correctOption: data.question.correctOption || 'A',
   894→              explanation: data.question.explanation || '',
   895→              section: data.question.section || 'General',
   896→              negativeMark: data.question.negativeMark || '0',
   897→            };
   898→            const filled = fillQuestionsFromExtract([newQ]);
   899→            totalExtracted += 1;
   900→            console.log(`✅ Fill complete — ${filled} boxes filled`);
   901→            toast.success(`Image ${i + 1}: 1 question extracted & filled!`);
   902→          } else {
   903→            hasError = true;
   904→            console.error(`❌ Extraction failed for image ${i + 1}:`, data.error);
   905→            setFillStatus({ msg: `❌ ${data.error || 'Failed to extract questions'}`, type: 'error' });
   906→            toast.error(`Image ${i + 1}: ${data.error || 'Failed to extract'}`);
   907→          }
   908→        } catch (err: any) {
   909→          clearTimeout(timeoutId);
   910→          hasError = true;
   911→          console.error(`❌ Fetch error for image ${i + 1}:`, err);
   912→          if (err?.name === 'AbortError') {
   913→            setFillStatus({ msg: '❌ Timed out (5 min). Try a smaller image.', type: 'error' });
   914→            toast.error(`Image ${i + 1}: Timed out (5 min). Try a smaller image.`);
   915→          } else {
   916→            setFillStatus({ msg: `❌ ${err?.message || 'Failed to extract'}`, type: 'error' });
   917→            toast.error(`Image ${i + 1}: ${err?.message || 'Failed to extract'}`);
   918→          }
   919→        }
   920→      }
   921→
   922→      if (totalExtracted > 0) {
   923→        toast.success(`🎉 Total: ${totalExtracted} questions extracted from ${fileList.length} image${fileList.length > 1 ? 's' : ''} & filled into boxes!`);
   924→      } else if (!hasError) {
   925→        toast.error('No questions could be extracted from the image(s). Empty boxes remain for manual entry.');
   926→      }
   927→    } finally {
   928→      clearInterval(timerInterval);
   929→      setExtractingImage(false);
   930→      setExtractElapsed(0);
   931→      e.target.value = '';
   932→    }
   933→  }
