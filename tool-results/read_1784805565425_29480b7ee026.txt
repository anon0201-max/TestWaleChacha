    20→    20→// ==================== ADMIN LOGIN ====================
    21→    21→function AdminLogin() {
    22→    22→  const { setAdminData } = useAppStore();
    23→    23→  const [username, setUsername] = useState('');
    24→    24→  const [password, setPassword] = useState('');
    25→    25→  const [error, setError] = useState('');
    26→    26→  const [loading, setLoading] = useState(false);
    27→    27→  const [showPassword, setShowPassword] = useState(false);
    28→    28→
    29→    29→  async function handleLogin() {
    30→    30→    setLoading(true);
    31→    31→    setError('');
    32→    32→    try {
    33→    33→      const res = await fetch('/api/admin/login', {
    34→    34→        method: 'POST',
    35→    35→        headers: { 'Content-Type': 'application/json' },
    36→    36→        body: JSON.stringify({ username, password }),
    37→    37→      });
    38→    38→      const data = await res.json();
    39→    39→      if (data.success) {
    40→    40→        setAdminData({ isLoggedIn: true });
    41→    41→      } else {
    42→    42→        setError(data.error || 'Login failed');
    43→    43→      }
    44→    44→    } catch { setError('Server error'); }
    45→    45→    setLoading(false);
    46→    46→  }
    47→    47→
    48→    48→  return (
    49→    49→    <div className="min-h-[70vh] flex items-center justify-center">
    50→    50→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
    51→    51→        <Card className="w-full max-w-sm shadow-xl border-0">
    52→    52→          <CardContent className="p-6">
    53→    53→            <div className="text-center mb-6">
    54→    54→              <div className="w-16 h-16 rounded-2xl bg-blue-700 flex items-center justify-center mx-auto mb-4">
    55→    55→                <Shield className="w-8 h-8 text-white" />
    56→    56→              </div>
    57→    57→              <h2 className="text-xl font-bold text-gray-900">Admin Login</h2>
    58→    58→              <p className="text-xs text-muted-foreground mt-1">Enter credentials to manage tests & questions</p>
    59→    59→            </div>
    60→    60→            <div className="space-y-4">
    61→    61→              <div>
    62→    62→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Username</Label>
    63→    63→                <Input value={username} onChange={(e) => setUsername(e.target.value)} placeholder="admin" className="h-11" />
    64→    64→              </div>
    65→    65→              <div>
    66→    66→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Password</Label>
    67→    67→                <div className="relative">
    68→    68→                  <Input
    69→    69→                    type={showPassword ? 'text' : 'password'}
    70→    70→                    value={password}
    71→    71→                    onChange={(e) => setPassword(e.target.value)}
    72→    72→                    placeholder="••••••••"
    73→    73→                    className="h-11 pr-10"
    74→    74→                    onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
    75→    75→                  />
    76→    76→                  <button
    77→    77→                    type="button"
    78→    78→                    onClick={() => setShowPassword(!showPassword)}
    79→    79→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
    80→    80→                  >
    81→    81→                    {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
    82→    82→                  </button>
    83→    83→                </div>
    84→    84→              </div>
    85→    85→              {error && (
    86→    86→                <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="bg-red-50 border border-red-200 text-red-700 text-sm rounded-xl p-3 flex items-center gap-2">
    87→    87→                  <AlertTriangle className="w-4 h-4 shrink-0" /> {error}
    88→    88→                </motion.div>
    89→    89→              )}
    90→    90→              <Button className="w-full bg-blue-700 hover:bg-blue-800 h-11 font-semibold" onClick={handleLogin} disabled={loading || !username || !password}>
    91→    91→                {loading ? 'Logging in...' : 'Login to Admin Panel'}
    92→    92→              </Button>
    93→    93→            </div>
    94→    94→          </CardContent>
    95→    95→        </Card>
    96→    96→      </motion.div>
    97→    97→    </div>
    98→    98→  );
    99→    99→}

   312→   312→function AdminCategoriesTab({ onRefresh }: { onRefresh: () => void }) {
   313→   313→  const { categories, setCategories } = useAppStore();
   314→   314→  const [newName, setNewName] = useState('');
   315→   315→  const [newSlug, setNewSlug] = useState('');
   316→   316→  const [newExamType, setNewExamType] = useState('General');
   317→   317→  const [adding, setAdding] = useState(false);
   318→   318→
   319→   319→  async function handleAdd() {
   320→   320→    if (!newName || !newSlug) return;
   321→   321→    setAdding(true);
   322→   322→    await fetch('/api/admin/categories', {
   323→   323→      method: 'POST',
   324→   324→      headers: { 'Content-Type': 'application/json' },
   325→   325→      body: JSON.stringify({ name: newName, slug: newSlug, examType: newExamType }),
   326→   326→    });
   327→   327→    setNewName(''); setNewSlug('');
   328→   328→    setAdding(false);
   329→   329→    onRefresh();
   330→   330→  }
   331→   331→
   332→   332→  async function handleDelete(id: string) {

   334→   334→    await fetch('/api/admin/categories', {
   335→   335→      method: 'DELETE',
   336→   336→      headers: { 'Content-Type': 'application/json' },
   337→   337→      body: JSON.stringify({ id }),
   338→   338→    });
   339→   339→    onRefresh();
   340→   340→  }
   341→   341→
   342→   342→  return (
   343→   343→    <div className="space-y-4">
   344→   344→      {/* Add Category */}
   345→   345→      <Card className="border-0 shadow-sm">
   346→   346→        <CardContent className="p-5">

   353→   353→                value={newName}

   382→   382→                <div className="flex items-center gap-3">
