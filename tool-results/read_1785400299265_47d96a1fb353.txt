     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { useAppStore } from '@/store/useAppStore';
     5→import { Button } from '@/components/ui/button';
     6→import { Badge } from '@/components/ui/badge';
     7→import { Card, CardContent } from '@/components/ui/card';
     8→import { motion, AnimatePresence } from 'framer-motion';
     9→import {
    10→  ChevronLeft,
    11→  ChevronRight,
    12→  Clock,
    13→  Flag,
    14→  BookmarkPlus,
    15→  X,
    16→  AlertTriangle,
    17→  Send,
    18→  User,
    19→  RotateCcw,
    20→  Languages,
    21→} from 'lucide-react';
    22→
    23→function formatTime(seconds: number): string {
    24→  const hrs = Math.floor(seconds / 3600);
    25→  const mins = Math.floor((seconds % 3600) / 60);
    26→  const secs = seconds % 60;
    27→  if (hrs > 0) return `${hrs}:${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
    28→  return `${mins}:${secs.toString().padStart(2, '0')}`;
    29→}
    30→
    31→type QuestionStatus = 'not-visited' | 'not-answered' | 'answered' | 'marked' | 'marked-answered';
    32→
    33→export function TestTakingPage() {
    34→  const {
    35→    currentTest,
    36→    answers,
    37→    setAnswer,
    38→    timeRemaining,
    39→    setTimeRemaining,
    40→    isTestActive,
    41→    setIsTestActive,
    42→    currentQuestionIndex,
    43→    setCurrentQuestionIndex,
    44→    deviceId,
    45→    user,
    46→    setView,
    47→    setLastResult,
    48→    clearAnswers,
    49→    setShowSubscriptionModal,
    50→    setUser,
    51→    setStudentData,
    52→  } = useAppStore();
    53→
    54→  const [markedForReview, setMarkedForReview] = useState<Set<string>>(new Set());
    55→  const [showConfirmSubmit, setShowConfirmSubmit] = useState(false);
    56→  const [showProfilePanel, setShowProfilePanel] = useState(false);
    57→  const [visited, setVisited] = useState<Set<string>>(new Set());
    58→  const [language, setLanguage] = useState<'en' | 'hi'>('en');
    59→  const [showInstructions, setShowInstructions] = useState(true);
    60→  const [submitting, setSubmitting] = useState(false);
    61→
    62→  const questions = currentTest?.questions || [];
    63→  const currentQuestion = questions[currentQuestionIndex];
    64→  const totalQuestions = questions.length;
    65→  const studentId = user?.id || null;
    66→
    67→  // Visit current question when navigating
    68→  if (currentQuestion && isTestActive && !visited.has(currentQuestion.id)) {
    69→    setVisited((prev) => new Set(prev).add(currentQuestion.id));
    70→  }
    71→
    72→  // Derive question statuses from state
    73→  function getStatus(qid: string): QuestionStatus {
    74→    const isVisited = visited.has(qid);
    75→    const isAnswered = qid in answers;
    76→    const isMarked = markedForReview.has(qid);
    77→    if (!isVisited) return 'not-visited';
    78→    if (isMarked && isAnswered) return 'marked-answered';
    79→    if (isMarked) return 'marked';
    80→    if (isAnswered) return 'answered';
    81→    return 'not-answered';
    82→  }
    83→
    84→  function handleSelectAnswer(questionId: string, option: string) {
    85→    setAnswer(questionId, option);
    86→  }
    87→
    88→  function handleMarkForReview() {
    89→    if (!currentQuestion) return;
    90→    setMarkedForReview((prev) => {
    91→      const next = new Set(prev);
    92→      if (next.has(currentQuestion.id)) next.delete(currentQuestion.id);
    93→      else next.add(currentQuestion.id);
    94→      return next;
    95→    });
    96→  }
    97→
    98→  function handleClearResponse() {
    99→    if (!currentQuestion) return;
   100→    const newAnswers = { ...answers };
   101→    delete newAnswers[currentQuestion.id];
   102→    useAppStore.setState({ answers: newAnswers });
   103→  }
   104→
   105→  function handleSaveAndNext() {
   106→    if (currentQuestionIndex < totalQuestions - 1) {
   107→      setCurrentQuestionIndex(currentQuestionIndex + 1);
   108→    }
   109→  }
   110→
   111→  const handleSubmitTest = useCallback(async () => {
   112→    if (!currentTest || submitting) return;
   113→    setSubmitting(true);
   114→    setIsTestActive(false);
   115→    const timeTaken = currentTest.timeLimit - timeRemaining;
   116→    try {
   117→      const body: Record<string, unknown> = {
   118→        testId: currentTest.id,
   119→        answers,
   120→        timeTaken,
   121→      };
   122→      if (studentId) body.studentId = studentId;
   123→      else if (deviceId) body.deviceId = deviceId;
   124→
   125→      const res = await fetch('/api/attempts', {
   126→        method: 'POST',
   127→        headers: { 'Content-Type': 'application/json' },
   128→        body: JSON.stringify(body),
   129→      });
   130→      const data = await res.json();
   131→
   132→      if (!res.ok) {
   133→        // API returned an error (400/403/500)
   134→        if (data.error === 'FREE_LIMIT_REACHED') {
   135→          setShowSubscriptionModal(true);
   136→        }
   137→        setView('home');
   138→        return;
   139→      }
   140→
   141→      // Validate response has the fields ResultsPage needs
   142→      if (!data.answerDetails || !Array.isArray(data.answerDetails) || data.answerDetails.length === 0) {
   143→        console.error('Invalid submit response — missing answerDetails:', data);
   144→        setView('home');
   145→        return;
   146→      }
   147→
   148→      // Update student data from response
   149→      if (data.updatedStudent) {
   150→        setUser(data.updatedStudent);
   151→      } else {
   152→        setStudentData({ freeTestsUsed: 0, isSubscribed: useAppStore.getState().isSubscribed });
   153→      }
   154→      setLastResult(data);
   155→      setView('results');
   156→    } catch (err) {
   157→      console.error('Submit test exception:', err);
   158→      setView('home');
   159→    } finally {
   160→      setSubmitting(false);
   161→    }
   162→  }, [currentTest, studentId, deviceId, answers, timeRemaining, submitting, setLastResult, setIsTestActive, setView, setShowSubscriptionModal, setUser, setStudentData]);
   163→
   164→  // Timer
   165→  useEffect(() => {
   166→    if (!isTestActive || timeRemaining <= 0) return;
   167→    const timer = setInterval(() => { setTimeRemaining(prev => Math.max(0, prev - 1)); }, 1000);
   168→    return () => clearInterval(timer);
   169→  }, [isTestActive, setTimeRemaining]);
   170→
   171→  // Auto-submit
   172→  useEffect(() => {
   173→    if (timeRemaining <= 0 && isTestActive) handleSubmitTest();
   174→  }, [timeRemaining, isTestActive, handleSubmitTest]);
   175→
   176→  if (!currentTest) return null;
   177→
   178→  // Handle blank/empty test (no questions)
   179→  if ((currentTest.questions?.length ?? 0) === 0) {
   180→    return (
   181→      <div className="min-h-[60vh] flex items-center justify-center">
   182→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="max-w-md w-full">
   183→          <Card className="border-0 shadow-lg">
   184→            <CardContent className="p-6 text-center">
   185→              <div className="w-16 h-16 rounded-full bg-amber-100 flex items-center justify-center mx-auto mb-4">
   186→                <AlertTriangle className="w-8 h-8 text-amber-600" />
   187→              </div>
   188→              <h2 className="text-xl font-bold mb-2">No Questions Available</h2>
   189→              <p className="text-sm text-muted-foreground mb-6">
   190→                This test &ldquo;{currentTest.title}&rdquo; doesn&apos;t have any questions yet. Please try again later.
   191→              </p>
   192→              <Button
   193→                className="bg-blue-600 hover:bg-blue-700 font-semibold"
   194→                onClick={() => { setIsTestActive(false); clearAnswers(); setView('tests'); }}
   195→              >
   196→                <ChevronLeft className="w-4 h-4 mr-2" /> Go Back to Tests
   197→              </Button>
   198→            </CardContent>
   199→          </Card>
   200→        </motion.div>
   201→      </div>
   202→    );
   203→  }
   204→
   205→  const options = [
   206→    { key: 'A', value: currentQuestion.optionA },
   207→    { key: 'B', value: currentQuestion.optionB },
   208→    { key: 'C', value: currentQuestion.optionC },
   209→    { key: 'D', value: currentQuestion.optionD },
   210→  ];
   211→
   212→  const isTimeLow = timeRemaining <= 60;
   213→  const answeredCount = Object.keys(answers).length;
   214→  const markedCount = markedForReview.size;
   215→  const notVisitedCount = questions.filter(q => !visited.has(q.id)).length;
   216→  const notAnsweredCount = questions.filter(q => visited.has(q.id) && !(q.id in answers)).length;
   217→
   218→  const paletteColors: Record<QuestionStatus, string> = {
   219→    'not-visited': 'bg-gray-200 text-gray-600',
   220→    'not-answered': 'bg-red-500 text-white',
   221→    'answered': 'bg-green-600 text-white',
   222→    'marked': 'bg-purple-600 text-white',
   223→    'marked-answered': 'bg-purple-500 text-white ring-2 ring-green-400',
   224→  };
   225→
   226→  const paletteLabels: Record<QuestionStatus, string> = {
   227→    'not-visited': 'Not Visited',
   228→    'not-answered': 'Not Answered',
   229→    'answered': 'Answered',
   230→    'marked': 'Marked for Review',
   231→    'marked-answered': 'Marked & Answered',
   232→  };
   233→
   234→  // Instructions modal
   235→  if (showInstructions) {
   236→    return (
   237→      <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="max-w-2xl mx-auto px-2 sm:px-0">
   238→        <Card className="border-2 border-blue-200">
   239→          <CardContent className="p-4 sm:p-6 md:p-8">
   240→            <div className="text-center mb-5">
   241→              <h1 className="text-lg sm:text-xl font-bold text-blue-900">{currentTest.examName}</h1>
   242→              <p className="text-base sm:text-lg text-blue-700 mt-1">{currentTest.title}</p>
   243→            </div>
   244→            <div className="bg-blue-50 rounded-xl p-3 sm:p-4 mb-5 space-y-2.5 text-sm">
   245→              <div className="flex justify-between"><span>Total Questions:</span><span className="font-semibold">{totalQuestions}</span></div>
   246→              <div className="flex justify-between"><span>Time Duration:</span><span className="font-semibold">{formatTime(currentTest.timeLimit)}</span></div>
   247→              <div className="flex justify-between"><span>Difficulty:</span><Badge variant="secondary">{currentTest.difficulty}</Badge></div>
   248→              {user && <div className="flex justify-between"><span>Candidate:</span><span className="font-semibold">{user.name}</span></div>}
   249→            </div>
   250→            <h3 className="font-semibold text-sm sm:text-base mb-3">Instructions:</h3>
   251→            <ul className="space-y-1.5 text-xs sm:text-sm text-muted-foreground mb-5 list-disc pl-5">
   252→              <li>Each question carries equal marks. No negative marking.</li>
   253→              <li>Navigate between questions using the Question Palette.</li>
   254→              <li>Use <strong>Save &amp; Next</strong> to save answer and move to next question.</li>
   255→              <li>Use <strong>Mark for Review</strong> to flag a question for later.</li>
   256→              <li>Use <strong>Clear Response</strong> to remove your selected answer.</li>
   257→              <li>Test will auto-submit when the timer reaches zero.</li>
   258→            </ul>
   259→            {/* Legend - responsive grid for mobile */}
   260→            <div className="grid grid-cols-2 sm:flex sm:flex-wrap sm:items-center sm:justify-center gap-2 sm:gap-3 text-xs mb-5">
   261→              {(['not-visited', 'not-answered', 'answered', 'marked', 'marked-answered'] as QuestionStatus[]).map((status) => (
   262→                <div key={status} className="flex items-center gap-1.5">
   263→                  <div className={`w-4 h-4 rounded shrink-0 ${paletteColors[status]}`} />
   264→                  <span className="text-gray-700">{paletteLabels[status]}</span>
   265→                </div>
   266→              ))}
   267→            </div>
   268→            {/* Buttons - stack on very small screens */}
   269→            <div className="flex flex-col sm:flex-row items-stretch sm:items-center gap-2.5 sm:gap-3">
   270→              <Button variant="outline" className="h-11 sm:h-12 px-6 w-full sm:w-auto" onClick={() => { setIsTestActive(false); clearAnswers(); setView('tests'); }}>
   271→                <ChevronLeft className="w-4 h-4 mr-1" /> Back
   272→              </Button>
   273→              <Button className="flex-1 h-11 sm:h-12 bg-blue-600 hover:bg-blue-700 text-white font-semibold text-sm sm:text-base" onClick={() => setShowInstructions(false)}>
   274→                I have read the instructions — Start Test
   275→              </Button>
   276→            </div>
   277→          </CardContent>
   278→        </Card>
   279→      </motion.div>
   280→    );
   281→  }
   282→
   283→  return (
   284→    <div className="flex flex-col h-[calc(100vh-3.5rem)] sm:h-[calc(100vh-56px)]">
   285→      {/* TOP BAR - Government Exam Style */}
   286→      <div className="bg-blue-900 text-white px-2 sm:px-3 py-1.5 sm:py-2 flex items-center justify-between shrink-0 z-20">
   287→        <div className="flex items-center gap-1.5 sm:gap-2 min-w-0">
   288→          <Button variant="ghost" size="icon" className="text-white hover:bg-blue-800 h-8 w-8 shrink-0" onClick={() => { if (confirm('Quit test? Your progress will be lost.')) { setIsTestActive(false); clearAnswers(); setMarkedForReview(new Set()); setView('tests'); } }}>
   289→            <X className="w-4 h-4" />
   290→          </Button>
   291→          <div className="min-w-0 hidden sm:block">
   292→            <p className="text-xs text-blue-300 truncate">{currentTest.examName} — {currentTest.category?.name || 'General'}</p>
   293→            <p className="text-sm font-medium truncate">{currentTest.title}</p>
   294→          </div>
   295→          <div className="min-w-0 sm:hidden">
   296→            <p className="text-xs font-medium truncate">Q{currentQuestionIndex + 1}/{totalQuestions}</p>
   297→          </div>
   298→        </div>
   299→        <div className="flex items-center gap-1.5 sm:gap-3">
   300→          <button onClick={() => setLanguage(language === 'en' ? 'hi' : 'en')} className="hidden sm:flex items-center gap-1 text-xs bg-blue-800 hover:bg-blue-700 px-2.5 py-1.5 rounded-lg transition-colors">
   301→            <Languages className="w-3.5 h-3.5" />
   302→            {language === 'en' ? 'हिंदी' : 'English'}
   303→          </button>
   304→          <div className={`flex items-center gap-1 sm:gap-1.5 px-2 sm:px-3 py-1 sm:py-1.5 rounded-lg font-mono font-bold text-sm sm:text-lg ${isTimeLow ? 'bg-red-600 animate-pulse' : 'bg-blue-800'}`}>
   305→            <Clock className="w-3.5 h-3.5 sm:w-4 sm:h-4" />
   306→            <span>{formatTime(timeRemaining)}</span>
   307→          </div>
   308→          <button className="hidden sm:flex items-center gap-1.5 bg-blue-800 hover:bg-blue-700 px-3 py-1.5 rounded-lg text-xs transition-colors">
   309→            <User className="w-3.5 h-3.5" />
   310→            <span>{user?.name || 'Candidate'}</span>
   311→          </button>
   312→        </div>
   313→      </div>
   314→
   315→      {/* MAIN CONTENT */}
   316→      <div className="flex flex-1 overflow-hidden">
   317→        {/* LEFT: Question Area */}
   318→        <div className="flex-1 overflow-y-auto p-3 md:p-5">
   319→          <AnimatePresence mode="wait">
   320→            <motion.div key={currentQuestionIndex} initial={{ opacity: 0, x: 30 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -30 }} transition={{ duration: 0.15 }}>
   321→              <div className="max-w-3xl mx-auto">
   322→                {/* Question Header */}
   323→                <div className="flex items-center justify-between mb-4">
   324→                  <div className="flex items-center gap-2">
   325→                    <Badge className="bg-blue-100 text-blue-800 hover:bg-blue-100 text-xs">
   326→                      Question {currentQuestionIndex + 1} of {totalQuestions}
   327→                    </Badge>
   328→                    <Badge variant="outline" className="text-xs">
   329→                      Section: {currentQuestion.section || 'General'}
   330→                    </Badge>
   331→                  </div>
   332→                  {markedForReview.has(currentQuestion.id) && (
   333→                    <Badge className="bg-purple-100 text-purple-800 hover:bg-purple-100 text-xs">
   334→                      <Flag className="w-3 h-3 mr-1" /> Marked for Review
   335→                    </Badge>
   336→                  )}
   337→                </div>
   338→
   339→                {/* Question Text */}
   340→                <div className="bg-white border rounded-xl p-4 md:p-6 mb-4 shadow-sm">
   341→                  <p className="text-base md:text-lg font-medium leading-relaxed">{currentQuestion.question}</p>
   342→                </div>
   343→
   344→                {/* Options */}
   345→                <div className="space-y-2.5 mb-6">
   346→                  {options.map((opt) => {
   347→                    const isSelected = answers[currentQuestion.id] === opt.key;
   348→                    return (
   349→                      <motion.button
   350→                        key={opt.key}
   351→                        whileTap={{ scale: 0.995 }}
   352→                        onClick={() => handleSelectAnswer(currentQuestion.id, opt.key)}
   353→                        className={`w-full text-left p-3.5 md:p-4 rounded-xl border-2 transition-all flex items-start gap-3 ${
   354→                          isSelected
   355→                            ? 'border-blue-600 bg-blue-50 shadow-sm'
   356→                            : 'border-gray-200 bg-white hover:border-blue-300 hover:bg-blue-50/30'
   357→                        }`}
   358→                      >
   359→                        <span className={`w-8 h-8 rounded-full flex items-center justify-center shrink-0 text-sm font-bold border-2 transition-all ${
   360→                          isSelected ? 'border-blue-600 bg-blue-600 text-white' : 'border-gray-300 text-gray-500'
   361→                        }`}>
   362→                          {opt.key}
   363→                        </span>
   364→                        <span className={`pt-1 text-sm md:text-base ${isSelected ? 'font-medium text-blue-900' : 'text-gray-700'}`}>
   365→                          {opt.value}
   366→                        </span>
   367→                      </motion.button>
   368→                    );
   369→                  })}
   370→                </div>
   371→
   372→                {/* Action Buttons - desktop only (mobile has bottom bar) */}
   373→                <div className="hidden md:flex flex-wrap items-center gap-2">
   374→                  <Button variant="outline" size="sm" onClick={handleClearResponse} disabled={!answers[currentQuestion.id]} className="text-xs sm:text-sm">
   375→                    <RotateCcw className="w-3.5 h-3.5 mr-1" /> Clear
   376→                  </Button>
   377→                  <Button variant="outline" size="sm" className={`text-xs sm:text-sm ${markedForReview.has(currentQuestion.id) ? 'border-purple-500 text-purple-700 bg-purple-50' : ''}`} onClick={handleMarkForReview}>
   378→                    <Flag className="w-3.5 h-3.5 mr-1" /> {markedForReview.has(currentQuestion.id) ? 'Unmark' : 'Mark'}
   379→                  </Button>
   380→                  <div className="flex-1" />
   381→                  <Button variant="outline" size="sm" onClick={() => setCurrentQuestionIndex(Math.max(0, currentQuestionIndex - 1))} disabled={currentQuestionIndex === 0} className="text-xs sm:text-sm">
   382→                    <ChevronLeft className="w-4 h-4 mr-1" /> Prev
   383→                  </Button>
   384→                  <Button size="sm" className="bg-blue-600 hover:bg-blue-700 text-xs sm:text-sm" onClick={handleSaveAndNext} disabled={currentQuestionIndex >= totalQuestions - 1}>
   385→                    Save &amp; Next <ChevronRight className="w-4 h-4 ml-1" />
   386→                  </Button>
   387→                </div>
   388→              </div>
   389→            </motion.div>
   390→          </AnimatePresence>
   391→        </div>
   392→
   393→        {/* RIGHT: Question Palette */}
   394→        <div className="hidden md:flex flex-col w-72 border-l bg-gray-50 shrink-0">
   395→          <div className="p-3 border-b bg-white">
   396→            <h3 className="font-semibold text-sm text-blue-900">Question Palette</h3>
   397→          </div>
   398→
   399→          {/* Palette Grid */}
   400→          <div className="p-3 flex-1 overflow-y-auto">
   401→            <div className="grid grid-cols-5 gap-1.5 mb-4">
   402→              {questions.map((q, i) => {
   403→                const status = getStatus(q.id);
   404→                const isCurrent = i === currentQuestionIndex;
   405→                return (
   406→                  <button
   407→                    key={q.id}
   408→                    onClick={() => setCurrentQuestionIndex(i)}
   409→                    className={`w-10 h-10 rounded-lg text-xs font-bold flex items-center justify-center transition-all ${
   410→                      paletteColors[status]
   411→                    } ${isCurrent ? 'ring-2 ring-blue-400 ring-offset-1 scale-105' : 'hover:scale-105'}`}
   412→                  >
   413→                    {i + 1}
   414→                  </button>
   415→                );
   416→              })}
   417→            </div>
   418→
   419→            {/* Legend */}
   420→            <div className="space-y-1.5 text-xs">
   421→              {(['not-visited', 'not-answered', 'answered', 'marked', 'marked-answered'] as QuestionStatus[]).map((status) => (
   422→                <div key={status} className="flex items-center gap-2">
   423→                  <div className={`w-4 h-4 rounded ${paletteColors[status]}`} />
   424→                  <span>{paletteLabels[status]}</span>
   425→                </div>
   426→              ))}
   427→            </div>
   428→          </div>
   429→
   430→          {/* Stats + Submit */}
   431→          <div className="p-3 border-t bg-white space-y-3">
   432→            <div className="grid grid-cols-2 gap-2 text-xs">
   433→              <div className="bg-green-50 rounded-lg p-2 text-center">
   434→                <p className="font-bold text-green-700 text-base">{answeredCount}</p>
   435→                <p className="text-green-600">Answered</p>
   436→              </div>
   437→              <div className="bg-red-50 rounded-lg p-2 text-center">
   438→                <p className="font-bold text-red-700 text-base">{notAnsweredCount}</p>
   439→                <p className="text-red-600">Not Answered</p>
   440→              </div>
   441→              <div className="bg-purple-50 rounded-lg p-2 text-center">
   442→                <p className="font-bold text-purple-700 text-base">{markedCount}</p>
   443→                <p className="text-purple-600">Marked</p>
   444→              </div>
   445→              <div className="bg-gray-100 rounded-lg p-2 text-center">
   446→                <p className="font-bold text-gray-700 text-base">{notVisitedCount}</p>
   447→                <p className="text-gray-500">Not Visited</p>
   448→              </div>
   449→            </div>
   450→            <Button className="w-full bg-red-600 hover:bg-red-700 h-10 font-semibold" onClick={() => setShowConfirmSubmit(true)} disabled={submitting}>
   451→              <Send className="w-4 h-4 mr-1.5" /> Submit Test
   452→            </Button>
   453→          </div>
   454→        </div>
   455→      </div>
   456→
   457→      {/* MOBILE: Bottom bar */}
   458→      <div className="md:hidden border-t bg-white p-2 shrink-0 safe-bottom">
   459→        <div className="flex items-center gap-1.5">

   466→          <Button variant="outline" size="sm" className="flex-1 text-[11px] px-1.5 h-9" onClick={handleMarkForReview}>
   467→            <Flag className="w-3.5 h-3.5 mr-0.5" /> {markedForReview.has(currentQuestion?.id) ? 'Unmark' : 'Mark'}

   473→        <div className="flex items-center gap-1.5 mt-1.5">
   474→          <Button size="sm" className="flex-[2] bg-blue-600 hover:bg-blue-700 text-[11px] px-1.5 h-9" onClick={handleSaveAndNext} disabled={currentQuestionIndex >= totalQuestions - 1}>
   475→            Save &amp; Next <ChevronRight className="w-3.5 h-3.5 ml-0.5" />
   476→          </Button>
   477→          <Button size="sm" className="flex-1 bg-red-600 hover:bg-red-700 text-[11px] px-1.5 h-9" onClick={() => setShowConfirmSubmit(true)} disabled={submitting}>

   486→              <div className="p-3">
   487→                <div className="flex items-center justify-between mb-2">
   488→                  <span className="text-xs font-semibold">Question Palette</span>
   489→                  <button onClick={() => setShowProfilePanel(false)}><X className="w-4 h-4" /></button>
   490→                </div>
   491→                <div className="grid grid-cols-8 gap-1 mb-3">
   492→                  {questions.map((q, i) => {

   494→                    const isCurrent = i === currentQuestionIndex;
   495→                    return (
   496→                      <button key={q.id} onClick={() => { setCurrentQuestionIndex(i); setShowProfilePanel(false); }}

   501→                  })}
   502→                </div>
   503→                <div className="flex flex-wrap gap-3 text-[10px]">
   504→                  {(['not-visited', 'not-answered', 'answered', 'marked', 'marked-answered'] as QuestionStatus[]).map((s) => (

   506→                      <div className={`w-3 h-3 rounded ${paletteColors[s]}`} />

   510→                </div>
   511→              </div>
   512→            </motion.div>

   514→        </AnimatePresence>
   515→      </div>
   516→
   517→      {/* Submit Confirmation */}
   518→      <AnimatePresence>
   519→        {showConfirmSubmit && (
   520→          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/50" onClick={() => setShowConfirmSubmit(false)}>
   521→            <motion.div initial={{ scale: 0.9 }} animate={{ scale: 1 }} exit={{ scale: 0.9 }} className="bg-white rounded-2xl p-6 max-w-sm w-full shadow-xl" onClick={(e) => e.stopPropagation()}>
   522→              <AlertTriangle className="w-12 h-12 text-amber-500 mx-auto mb-4" />
   523→              <h3 className="text-lg font-bold text-center mb-2">Submit Test?</h3>
   524→              <div className="bg-gray-50 rounded-xl p-3 mb-4 space-y-1.5 text-sm">
   525→                <div className="flex justify-between"><span className="text-muted-foreground">Answered:</span><span className="font-semibold text-green-700">{answeredCount}</span></div>
   526→                <div className="flex justify-between"><span className="text-muted-foreground">Not Answered:</span><span className="font-semibold text-red-700">{totalQuestions - answeredCount}</span></div>
   527→                <div className="flex justify-between"><span className="text-muted-foreground">Marked for Review:</span><span className="font-semibold text-purple-700">{markedCount}</span></div>
   528→              </div>

   531→              )}
   532→              <div className="flex gap-3">
   533→                <Button variant="outline" className="flex-1" onClick={() => setShowConfirmSubmit(false)}>Go Back</Button>
   534→                <Button className="flex-1 bg-red-600 hover:bg-red-700" onClick={() => { setShowConfirmSubmit(false); handleSubmitTest(); }} disabled={submitting}>
   535→                  {submitting ? 'Submitting...' : 'Submit Now'}

   537→              </div>
   538→            </motion.div>
   539→          </motion.div>
   540→        )}
   541→      </AnimatePresence>
   542→    </div>
