     7→import { Card, CardContent } from '@/components/ui/card';
     8→import { motion, AnimatePresence } from 'framer-motion';
     9→import {

    21→} from 'lucide-react';

    23→
    24→function formatTime(seconds: number): string {
    25→  if (!seconds || !isFinite(seconds) || seconds < 0) return '0:00';
    26→  const hrs = Math.floor(seconds / 3600);
    27→  const mins = Math.floor((seconds % 3600) / 60);
    28→  const secs = seconds % 60;
    29→  if (hrs > 0) return `${hrs}:${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
    30→  return `${mins}:${secs.toString().padStart(2, '0')}`;
    31→}
    32→
    33→type QuestionStatus = 'not-visited' | 'not-answered' | 'answered' | 'marked' | 'marked-answered';
    34→
    35→export function TestTakingPage() {
    36→  const {
    37→    currentTest,
    38→    answers,
    39→    setAnswer,
    40→    timeRemaining,
    41→    setTimeRemaining,
    42→    isTestActive,
    43→    setIsTestActive,
    44→    currentQuestionIndex,
    45→    setCurrentQuestionIndex,
    46→    deviceId,
    47→    user,
    48→    setView,
    49→    setLastResult,
    50→    clearAnswers,
    51→    setShowSubscriptionModal,
    52→    setUser,
    53→    setStudentData,
    54→  } = useAppStore();
    55→
    56→  const [markedForReview, setMarkedForReview] = useState<Set<string>>(new Set());
    57→  const [showConfirmSubmit, setShowConfirmSubmit] = useState(false);
    58→  const [showProfilePanel, setShowProfilePanel] = useState(false);
    59→  const [visited, setVisited] = useState<Set<string>>(new Set());
    60→  const [language, setLanguage] = useState<'en' | 'hi'>('en');
    61→  const [showInstructions, setShowInstructions] = useState(true);
    62→  const [submitting, setSubmitting] = useState(false);
    63→
    64→  const questions = currentTest?.questions || [];
    65→  const currentQuestion = questions[currentQuestionIndex];
    66→  const totalQuestions = questions.length;
    67→  const studentId = user?.id || null;
    68→
    69→  // Visit current question when navigating
    70→  if (currentQuestion && isTestActive && !visited.has(currentQuestion.id)) {
    71→    setVisited((prev) => new Set(prev).add(currentQuestion.id));
    72→  }
    73→
    74→  // Derive question statuses from state
    75→  function getStatus(qid: string): QuestionStatus {
    76→    const isVisited = visited.has(qid);
    77→    const isAnswered = qid in answers;
    78→    const isMarked = markedForReview.has(qid);
    79→    if (!isVisited) return 'not-visited';
    80→    if (isMarked && isAnswered) return 'marked-answered';
    81→    if (isMarked) return 'marked';
    82→    if (isAnswered) return 'answered';
    83→    return 'not-answered';
    84→  }
    85→
    86→  function handleSelectAnswer(questionId: string, option: string) {
    87→    setAnswer(questionId, option);
    88→  }
    89→
    90→  function handleMarkForReview() {
    91→    if (!currentQuestion) return;
    92→    setMarkedForReview((prev) => {
    93→      const next = new Set(prev);
    94→      if (next.has(currentQuestion.id)) next.delete(currentQuestion.id);
    95→      else next.add(currentQuestion.id);
    96→      return next;
    97→    });
    98→  }
    99→
   100→  function handleClearResponse() {
   101→    if (!currentQuestion) return;
   102→    const newAnswers = { ...answers };
   103→    delete newAnswers[currentQuestion.id];
   104→    useAppStore.setState({ answers: newAnswers });
   105→  }
   106→
   107→  function handleSaveAndNext() {
   108→    if (currentQuestionIndex < totalQuestions - 1) {
   109→      setCurrentQuestionIndex(currentQuestionIndex + 1);
   110→    }
   111→  }
   112→
   113→  const handleSubmitTest = useCallback(async () => {
   114→    if (!currentTest || submitting) return;
   115→    setSubmitting(true);
   116→    setIsTestActive(false);
   117→    const timeTaken = Number(currentTest.timeLimit || 0) - Number(timeRemaining || 0);
   118→    try {
   119→      const body: Record<string, unknown> = {
   120→        testId: currentTest.id,
   121→        answers,
   122→        timeTaken,
   123→      };
   124→      if (studentId) body.studentId = studentId;
   125→      else if (deviceId) body.deviceId = deviceId;
   126→
   127→      const res = await fetch('/api/attempts', {
   128→        method: 'POST',
   129→        headers: { 'Content-Type': 'application/json' },
   130→        body: JSON.stringify(body),
   131→      });
   132→      const data = await res.json();
   133→
   134→      // Handle free limit reached — show subscription modal but don't go home
   135→      if (!res.ok && data.error === 'FREE_LIMIT_REACHED') {
   136→        setShowSubscriptionModal(true);
   137→        setIsTestActive(true); // re-enable test so user can go back
   138→        setSubmitting(false);
   139→        return;
   140→      }
   141→
   142→      // Handle test locked
   143→      if (!res.ok && data.error === 'TEST_LOCKED') {
   144→        toast({ title: 'Test Locked', description: data.message || 'Subscribe to unlock this test.', variant: 'destructive' });
   145→        setIsTestActive(true);
   146→        setSubmitting(false);
   147→        return;
   148→      }
   149→
   150→      // Handle other API errors — show toast instead of silently going home
   151→      if (!res.ok) {
   152→        toast({ title: 'Submit Failed', description: data.error || 'Something went wrong. Please try again.', variant: 'destructive' });
   153→        setIsTestActive(true); // re-enable test so user can retry
   154→        setSubmitting(false);
   155→        return;
   156→      }
   157→
   158→      // API returned 200 — map response to AttemptResult and show results
   159→      const resultData: Record<string, unknown> = {
   160→        score: data.score,
   161→        correctAnswers: data.correctAnswers,
   162→        totalQuestions: data.totalQuestions,
   163→        answerDetails: data.answerDetails || [],
   164→        timeTaken: data.timeTaken,
   165→        // ResultsPage uses currentTest from store, so test field here is just for type safety
   166→        test: currentTest ? { title: currentTest.title, category: { name: currentTest.category?.name || '' } } : { title: '', category: { name: '' } },
   167→      };
   168→
   169→      // Update student data from response
   170→      if (data.updatedStudent) {
   171→        setUser(data.updatedStudent);
   172→      } else {
   173→        setStudentData({ freeTestsUsed: 0, isSubscribed: useAppStore.getState().isSubscribed });
   174→      }
   175→      setLastResult(resultData as any);
   176→      setView('results');
   177→    } catch (err) {
   178→      console.error('Submit test exception:', err);
   179→      toast({ title: 'Network Error', description: 'Could not submit test. Check your connection and try again.', variant: 'destructive' });
   180→      setIsTestActive(true); // re-enable test
   181→    } finally {
   182→      setSubmitting(false);
   183→    }
   184→  }, [currentTest, studentId, deviceId, answers, timeRemaining, submitting, setLastResult, setIsTestActive, setView, setShowSubscriptionModal, setUser, setStudentData]);
   185→
   186→  // Timer
   187→  useEffect(() => {
   188→    if (!isTestActive || timeRemaining <= 0) return;
   189→    const timer = setInterval(() => { setTimeRemaining(prev => Math.max(0, prev - 1)); }, 1000);
   190→    return () => clearInterval(timer);
   191→  }, [isTestActive, setTimeRemaining]);
   192→
   193→  // Auto-submit
   194→  useEffect(() => {
   195→    if (timeRemaining <= 0 && isTestActive) handleSubmitTest();
   196→  }, [timeRemaining, isTestActive, handleSubmitTest]);
   197→
   198→  if (!currentTest) return null;
   199→
   200→  // Handle blank/empty test (no questions)
   201→  if ((currentTest.questions?.length ?? 0) === 0) {
   202→    return (
   203→      <div className="min-h-[60vh] flex items-center justify-center">
   204→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="max-w-md w-full">
   205→          <Card className="border-0 shadow-lg">
   206→            <CardContent className="p-6 text-center">
   207→              <div className="w-16 h-16 rounded-full bg-amber-100 flex items-center justify-center mx-auto mb-4">
   208→                <AlertTriangle className="w-8 h-8 text-amber-600" />
   209→              </div>
   210→              <h2 className="text-xl font-bold mb-2">No Questions Available</h2>
   211→              <p className="text-sm text-muted-foreground mb-6">
   212→                This test &ldquo;{currentTest.title}&rdquo; doesn&apos;t have any questions yet. Please try again later.
   213→              </p>
   214→              <Button
   215→                className="bg-blue-600 hover:bg-blue-700 font-semibold"
   216→                onClick={() => { setIsTestActive(false); clearAnswers(); setView('tests'); }}
   217→              >
   218→                <ChevronLeft className="w-4 h-4 mr-2" /> Go Back to Tests
   219→              </Button>
   220→            </CardContent>
   221→          </Card>
   222→        </motion.div>
   223→      </div>
   224→    );
   225→  }
   226→
   227→  const options = [
   228→    { key: 'A', value: currentQuestion.optionA },
   229→    { key: 'B', value: currentQuestion.optionB },
   230→    { key: 'C', value: currentQuestion.optionC },
   231→    { key: 'D', value: currentQuestion.optionD },
   232→  ];
   233→
   234→  const isTimeLow = timeRemaining <= 60;
   235→  const answeredCount = Object.keys(answers).length;
   236→  const markedCount = markedForReview.size;
   237→  const notVisitedCount = questions.filter(q => !visited.has(q.id)).length;
   238→  const notAnsweredCount = questions.filter(q => visited.has(q.id) && !(q.id in answers)).length;
   239→
   240→  const paletteColors: Record<QuestionStatus, string> = {
   241→    'not-visited': 'bg-gray-200 text-gray-600',
   242→    'not-answered': 'bg-red-500 text-white',
   243→    'answered': 'bg-green-600 text-white',
   244→    'marked': 'bg-purple-600 text-white',
   245→    'marked-answered': 'bg-purple-500 text-white ring-2 ring-green-400',
   246→  };
   247→
   248→  const paletteLabels: Record<QuestionStatus, string> = {
   249→    'not-visited': 'Not Visited',
   250→    'not-answered': 'Not Answered',
   251→    'answered': 'Answered',
   252→    'marked': 'Marked for Review',
   253→    'marked-answered': 'Marked & Answered',
   254→  };
   255→
   256→  // Instructions modal
   257→  if (showInstructions) {
   258→    return (
   259→      <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="max-w-2xl mx-auto px-2 sm:px-0">
   260→        <Card className="border-2 border-blue-200">
   261→          <CardContent className="p-4 sm:p-6 md:p-8">
   262→            <div className="text-center mb-5">
   263→              <h1 className="text-lg sm:text-xl font-bold text-blue-900">{currentTest.examName}</h1>
   264→              <p className="text-base sm:text-lg text-blue-700 mt-1">{currentTest.title}</p>
   265→            </div>
   266→            <div className="bg-blue-50 rounded-xl p-3 sm:p-4 mb-5 space-y-2.5 text-sm">
   267→              <div className="flex justify-between"><span>Total Questions:</span><span className="font-semibold">{totalQuestions}</span></div>
   268→              <div className="flex justify-between"><span>Time Duration:</span><span className="font-semibold">{formatTime(currentTest.timeLimit)}</span></div>
   269→              <div className="flex justify-between"><span>Difficulty:</span><Badge variant="secondary">{currentTest.difficulty}</Badge></div>
   270→              {user && <div className="flex justify-between"><span>Candidate:</span><span className="font-semibold">{user.name}</span></div>}
   271→            </div>
   272→            <h3 className="font-semibold text-sm sm:text-base mb-3">Instructions:</h3>
   273→            <ul className="space-y-1.5 text-xs sm:text-sm text-muted-foreground mb-5 list-disc pl-5">
   274→              <li>Each question carries equal marks. No negative marking.</li>
   275→              <li>Navigate between questions using the Question Palette.</li>
   276→              <li>Use <strong>Save &amp; Next</strong> to save answer and move to next question.</li>
   277→              <li>Use <strong>Mark for Review</strong> to flag a question for later.</li>
   278→              <li>Use <strong>Clear Response</strong> to remove your selected answer.</li>
   279→              <li>Test will auto-submit when the timer reaches zero.</li>
   280→            </ul>
   281→            {/* Legend - responsive grid for mobile */}
   282→            <div className="grid grid-cols-2 sm:flex sm:flex-wrap sm:items-center sm:justify-center gap-2 sm:gap-3 text-xs mb-5">
   283→              {(['not-visited', 'not-answered', 'answered', 'marked', 'marked-answered'] as QuestionStatus[]).map((status) => (
   284→                <div key={status} className="flex items-center gap-1.5">
   285→                  <div className={`w-4 h-4 rounded shrink-0 ${paletteColors[status]}`} />
   286→                  <span className="text-gray-700">{paletteLabels[status]}</span>
   287→                </div>
   288→              ))}
   289→            </div>
   290→            {/* Buttons - stack on very small screens */}
   291→            <div className="flex flex-col sm:flex-row items-stretch sm:items-center gap-2.5 sm:gap-3">
   292→              <Button variant="outline" className="h-11 sm:h-12 px-6 w-full sm:w-auto" onClick={() => { setIsTestActive(false); clearAnswers(); setView('tests'); }}>
   293→                <ChevronLeft className="w-4 h-4 mr-1" /> Back
   294→              </Button>
   295→              <Button className="flex-1 h-11 sm:h-12 bg-blue-600 hover:bg-blue-700 text-white font-semibold text-sm sm:text-base" onClick={() => setShowInstructions(false)}>
   296→                I have read the instructions — Start Test
   297→              </Button>
   298→            </div>
   299→          </CardContent>
   300→        </Card>
   301→      </motion.div>
   302→    );
   303→  }
   304→
   305→  return (
   306→    <div className="flex flex-col h-[calc(100vh-3.5rem)] sm:h-[calc(100vh-56px)]">
   307→      {/* TOP BAR - Government Exam Style */}
   308→      <div className="bg-blue-900 text-white px-2 sm:px-3 py-1.5 sm:py-2 flex items-center justify-between shrink-0 z-20">
   309→        <div className="flex items-center gap-1.5 sm:gap-2 min-w-0">
   310→          <Button variant="ghost" size="icon" className="text-white hover:bg-blue-800 h-8 w-8 shrink-0" onClick={() => { if (confirm('Quit test? Your progress will be lost.')) { setIsTestActive(false); clearAnswers(); setMarkedForReview(new Set()); setView('tests'); } }}>
   311→            <X className="w-4 h-4" />
   312→          </Button>
   313→          <div className="min-w-0 hidden sm:block">
   314→            <p className="text-xs text-blue-300 truncate">{currentTest.examName} — {currentTest.category?.name || 'General'}</p>
   315→            <p className="text-sm font-medium truncate">{currentTest.title}</p>
   316→          </div>
   317→          <div className="min-w-0 sm:hidden">
   318→            <p className="text-xs font-medium truncate">Q{currentQuestionIndex + 1}/{totalQuestions}</p>
   319→          </div>
   320→        </div>
   321→        <div className="flex items-center gap-1.5 sm:gap-3">
   322→          <button onClick={() => setLanguage(language === 'en' ? 'hi' : 'en')} className="hidden sm:flex items-center gap-1 text-xs bg-blue-800 hover:bg-blue-700 px-2.5 py-1.5 rounded-lg transition-colors">
   323→            <Languages className="w-3.5 h-3.5" />
   324→            {language === 'en' ? 'हिंदी' : 'English'}
   325→          </button>
   326→          <div className={`flex items-center gap-1 sm:gap-1.5 px-2 sm:px-3 py-1 sm:py-1.5 rounded-lg font-mono font-bold text-sm sm:text-lg ${isTimeLow ? 'bg-red-600 animate-pulse' : 'bg-blue-800'}`}>
   327→            <Clock className="w-3.5 h-3.5 sm:w-4 sm:h-4" />
   328→            <span>{formatTime(timeRemaining)}</span>
   329→          </div>
   330→          <button className="hidden sm:flex items-center gap-1.5 bg-blue-800 hover:bg-blue-700 px-3 py-1.5 rounded-lg text-xs transition-colors">
   331→            <User className="w-3.5 h-3.5" />
   332→            <span>{user?.name || 'Candidate'}</span>
   333→          </button>
   334→        </div>
   335→      </div>
   336→
   337→      {/* MAIN CONTENT */}
   338→      <div className="flex flex-1 overflow-hidden">
   339→        {/* LEFT: Question Area */}
   340→        <div className="flex-1 overflow-y-auto p-3 md:p-5">
   341→          <AnimatePresence mode="wait">
   342→            <motion.div key={currentQuestionIndex} initial={{ opacity: 0, x: 30 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -30 }} transition={{ duration: 0.15 }}>
   343→              <div className="max-w-3xl mx-auto">
   344→                {/* Question Header */}
   345→                <div className="flex items-center justify-between mb-4">
   346→                  <div className="flex items-center gap-2">
   347→                    <Badge className="bg-blue-100 text-blue-800 hover:bg-blue-100 text-xs">
   348→                      Question {currentQuestionIndex + 1} of {totalQuestions}
   349→                    </Badge>
   350→                    <Badge variant="outline" className="text-xs">
   351→                      Section: {currentQuestion.section || 'General'}
   352→                    </Badge>
   353→                  </div>
   354→                  {markedForReview.has(currentQuestion.id) && (
   355→                    <Badge className="bg-purple-100 text-purple-800 hover:bg-purple-100 text-xs">
   356→                      <Flag className="w-3 h-3 mr-1" /> Marked for Review
   357→                    </Badge>
   358→                  )}
   359→                </div>
   360→
   361→                {/* Question Text */}
   362→                <div className="bg-white border rounded-xl p-4 md:p-6 mb-4 shadow-sm">
   363→                  <p className="text-base md:text-lg font-medium leading-relaxed">{currentQuestion.question}</p>
   364→                </div>
   365→
   366→                {/* Options */}
   367→                <div className="space-y-2.5 mb-6">
   368→                  {options.map((opt) => {
   369→                    const isSelected = answers[currentQuestion.id] === opt.key;
   370→                    return (
   371→                      <motion.button
   372→                        key={opt.key}
   373→                        whileTap={{ scale: 0.995 }}
   374→                        onClick={() => handleSelectAnswer(currentQuestion.id, opt.key)}
   375→                        className={`w-full text-left p-3.5 md:p-4 rounded-xl border-2 transition-all flex items-start gap-3 ${
   376→                          isSelected
   377→                            ? 'border-blue-600 bg-blue-50 shadow-sm'
   378→                            : 'border-gray-200 bg-white hover:border-blue-300 hover:bg-blue-50/30'
   379→                        }`}
   380→                      >
   381→                        <span className={`w-8 h-8 rounded-full flex items-center justify-center shrink-0 text-sm font-bold border-2 transition-all ${
   382→                          isSelected ? 'border-blue-600 bg-blue-600 text-white' : 'border-gray-300 text-gray-500'
   383→                        }`}>
   384→                          {opt.key}
   385→                        </span>
   386→                        <span className={`pt-1 text-sm md:text-base ${isSelected ? 'font-medium text-blue-900' : 'text-gray-700'}`}>
   387→                          {opt.value}
   388→                        </span>
   389→                      </motion.button>
   390→                    );
   391→                  })}
   392→                </div>
   393→
   394→                {/* Action Buttons - desktop only (mobile has bottom bar) */}
   395→                <div className="hidden md:flex flex-wrap items-center gap-2">
   396→                  <Button variant="outline" size="sm" onClick={handleClearResponse} disabled={!answers[currentQuestion.id]} className="text-xs sm:text-sm">
   397→                    <RotateCcw className="w-3.5 h-3.5 mr-1" /> Clear
   398→                  </Button>
   399→                  <Button variant="outline" size="sm" className={`text-xs sm:text-sm ${markedForReview.has(currentQuestion.id) ? 'border-purple-500 text-purple-700 bg-purple-50' : ''}`} onClick={handleMarkForReview}>
   400→                    <Flag className="w-3.5 h-3.5 mr-1" /> {markedForReview.has(currentQuestion.id) ? 'Unmark' : 'Mark'}
   401→                  </Button>
   402→                  <div className="flex-1" />
   403→                  <Button variant="outline" size="sm" onClick={() => setCurrentQuestionIndex(Math.max(0, currentQuestionIndex - 1))} disabled={currentQuestionIndex === 0} className="text-xs sm:text-sm">
   404→                    <ChevronLeft className="w-4 h-4 mr-1" /> Prev
   405→                  </Button>
   406→                  <Button size="sm" className="bg-blue-600 hover:bg-blue-700 text-xs sm:text-sm" onClick={handleSaveAndNext} disabled={currentQuestionIndex >= totalQuestions - 1}>
   407→                    Save &amp; Next <ChevronRight className="w-4 h-4 ml-1" />
   408→                  </Button>
   409→                </div>
   410→              </div>
   411→            </motion.div>
   412→          </AnimatePresence>
   413→        </div>
   414→
   415→        {/* RIGHT: Question Palette */}
   416→        <div className="hidden md:flex flex-col w-72 border-l bg-gray-50 shrink-0">
   417→          <div className="p-3 border-b bg-white">
   418→            <h3 className="font-semibold text-sm text-blue-900">Question Palette</h3>
   419→          </div>
   420→
   421→          {/* Palette Grid */}
   422→          <div className="p-3 flex-1 overflow-y-auto">
   423→            <div className="grid grid-cols-5 gap-1.5 mb-4">
   424→              {questions.map((q, i) => {
   425→                const status = getStatus(q.id);
   426→                const isCurrent = i === currentQuestionIndex;

   428→                  <button

   435→                    {i + 1}
   436→                  </button>
   437→                );
   438→              })}
   439→            </div>
   440→
   441→            {/* Legend */}
   442→            <div className="space-y-1.5 text-xs">
   443→              {(['not-visited', 'not-answered', 'answered', 'marked', 'marked-answered'] as QuestionStatus[]).map((status) => (
   444→                <div key={status} className="flex items-center gap-2">
   445→                  <div className={`w-4 h-4 rounded ${paletteColors[status]}`} />
   446→                  <span>{paletteLabels[status]}</span>
   447→                </div>
   448→              ))}
   449→            </div>
   450→          </div>
   451→
   452→          {/* Stats + Submit */}
   453→          <div className="p-3 border-t bg-white space-y-3">
   454→            <div className="grid grid-cols-2 gap-2 text-xs">
   455→              <div className="bg-green-50 rounded-lg p-2 text-center">
   456→                <p className="font-bold text-green-700 text-base">{answeredCount}</p>
   457→                <p className="text-green-600">Answered</p>

   543→            <motion.div initial={{ scale: 0.9 }} animate={{ scale: 1 }} exit={{ scale: 0.9 }} className="bg-white rounded-2xl p-6 max-w-sm w-full shadow-xl" onClick={(e) => e.stopPropagation()}>
