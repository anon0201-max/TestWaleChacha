    24→function formatTime(seconds: number): string {
    25→  if (!seconds || !isFinite(seconds) || seconds < 0) return '0:00';
    26→  const hrs = Math.floor(seconds / 3600);
    27→  const mins = Math.floor((seconds % 3600) / 60);
    28→  const secs = seconds % 60;
    29→  if (hrs > 0) return `${hrs}:${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
    30→  return `${mins}:${secs.toString().padStart(2, '0')}`;
    31→}

    87→            clearInterval(timerRef.current);
    88→            timerRef.current = null;

   103→  }, []); // eslint-disable-line react-hooks/exhaustive-deps
   104→

   108→      if (timerRef.current) {
   109→        clearInterval(timerRef.current);
   110→        timerRef.current = null;
   111→      }
   112→    };
   113→  }, []);
   114→
   115→  // Also keep store timeRemaining in sync for submit calculation
   116→  useEffect(() => {
   117→    useAppStore.setState({ timeRemaining: timerSeconds });
   118→  }, [timerSeconds]);
   119→
   120→  // Visit current question when navigating
   121→  if (currentQuestion && !visited.has(currentQuestion.id)) {
   122→    setVisited((prev) => new Set(prev).add(currentQuestion.id));

   125→  // Derive question statuses from state
   126→  function getStatus(qid: string): QuestionStatus {

   128→    const isAnswered = qid in answers;

   130→    if (!isVisited) return 'not-visited';
   131→    if (isMarked && isAnswered) return 'marked-answered';
   132→    if (isMarked) return 'marked';
   133→    if (isAnswered) return 'answered';
   134→    return 'not-answered';
   135→  }

   139→  }

   141→  function handleMarkForReview() {

   145→      if (next.has(currentQuestion.id)) next.delete(currentQuestion.id);
   146→      else next.add(currentQuestion.id);
   147→      return next;
   148→    });

   165→  function handleDismissInstructions() {

   169→  }

   171→  // Submit handler — uses refs and getState() to avoid stale closures
   172→  async function handleSubmitTest() {

   174→    if (submittingRef.current) return;
   175→
   176→    const state = useAppStore.getState();
   177→    const test = state.currentTest;
   178→    if (!test) return;
   179→
   180→    submittingRef.current = true;
   181→    setSubmitting(true);
   182→
   183→    // Stop timer immediately
   184→    if (timerRef.current) {
   185→      clearInterval(timerRef.current);
   186→      timerRef.current = null;
   187→    }
   188→    setTimerRunning(false);
   189→
   190→    const testLimit = Number(test.timeLimit) || 600;
   191→    const timeTaken = testLimit - timerSeconds;
   192→
   193→    try {
   194→      const body: Record<string, unknown> = {
   195→        testId: test.id,
   196→        answers: state.answers,
   197→        timeTaken: Math.max(0, timeTaken),
   198→      };
   199→      const sid = state.user?.id;
   200→      if (sid) body.studentId = sid;
   201→      else if (state.deviceId) body.deviceId = state.deviceId;
   202→
   203→      console.log('[SubmitTest] Sending:', JSON.stringify({ testId: body.testId, studentId: body.studentId, deviceId: body.deviceId, answerCount: Object.keys(body.answers as Record<string, string>).length, timeTaken: body.timeTaken }));
   204→      const res = await fetch('/api/attempts', {
   205→        method: 'POST',
   206→        headers: { 'Content-Type': 'application/json' },
   207→        body: JSON.stringify(body),
   208→      });
   209→      console.log('[SubmitTest] Response status:', res.status);
   210→      const data = await res.json();
   211→      console.log('[SubmitTest] Response data:', JSON.stringify(data).substring(0, 500));
   212→
   213→      // Handle free limit reached
   214→      if (!res.ok && data.error === 'FREE_LIMIT_REACHED') {
   215→        useAppStore.getState().setShowSubscriptionModal(true);
   216→        submittingRef.current = false;
   217→        setSubmitting(false);
   218→        return;
   219→      }
   220→
   221→      // Handle test locked
   222→      if (!res.ok && data.error === 'TEST_LOCKED') {
   223→        toast.error(data.message || 'Subscribe to unlock this test.');
   224→        submittingRef.current = false;
   225→        setSubmitting(false);
   226→        return;
   227→      }
   228→
   229→      // Handle other API errors
   230→      if (!res.ok) {
   231→        toast.error(data.error || data.message || 'Submit failed. Try again.');
   232→        submittingRef.current = false;
   233→        setSubmitting(false);
   234→        return;
   235→      }
   236→
   237→      // API returned 200 — set result and navigate to results
   238→      const resultData = {
   239→        score: data.score,
   240→        correctAnswers: data.correctAnswers,
   241→        totalQuestions: data.totalQuestions,
   242→        answerDetails: data.answerDetails || [],
   243→        timeTaken: data.timeTaken,
   244→        test: { title: test.title, category: { name: test.category?.name || '' } },
   245→      };
   246→
   247→      // Update student data
   248→      if (data.updatedStudent) {
   249→        useAppStore.getState().setUser(data.updatedStudent);
   250→      } else {
   251→        useAppStore.getState().setStudentData({ freeTestsUsed: 0, isSubscribed: state.isSubscribed });
   252→      }

   254→      // CRITICAL: Set result FIRST, then change view
   255→      // Use setTimeout to ensure state is committed before view switch
   256→      useAppStore.setState({
   257→        lastResult: resultData,
   258→        isTestActive: false,
   259→      });
   260→
   261→      // Small delay to ensure lastResult is committed to store
   262→      // before the view switch unmounts this component

   264→        useAppStore.setState({ currentView: 'results' });

   268→      try { toast.error('Network error. Check connection and try again.'); } catch {}
   269→      submittingRef.current = false;
   270→      setSubmitting(false);
   271→    }
   272→  }

   329→  const notVisitedCount = questions.filter(q => !visited.has(q.id)).length;
   330→  const notAnsweredCount = questions.filter(q => visited.has(q.id) && !(q.id in answers)).length;

   352→        <Card className="border-2 border-blue-200">
