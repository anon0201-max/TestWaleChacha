    23→    23→// ==================== ADMIN LOGIN ====================
    24→    24→function AdminLogin() {
    25→    25→  const { setAdminData } = useAppStore();
    26→    26→  const [username, setUsername] = useState('');
    27→    27→  const [password, setPassword] = useState('');
    28→    28→  const [error, setError] = useState('');
    29→    29→  const [loading, setLoading] = useState(false);
    30→    30→  const [showPassword, setShowPassword] = useState(false);
    31→    31→  const [changePasswordOpen, setChangePasswordOpen] = useState(false);
    32→    32→  const [cpCurrentPassword, setCpCurrentPassword] = useState('');
    33→    33→  const [cpNewPassword, setCpNewPassword] = useState('');
    34→    34→  const [cpConfirmPassword, setCpConfirmPassword] = useState('');
    35→    35→  const [cpLoading, setCpLoading] = useState(false);
    36→    36→  const [cpMsg, setCpMsg] = useState<{ type: 'success' | 'error'; text: string } | null>(null);
    37→    37→
    38→    38→  async function handleLogin() {
    39→    39→    setLoading(true);
    40→    40→    setError('');
    41→    41→    try {
    42→    42→      const res = await fetch('/api/admin/login', {
    43→    43→        method: 'POST',
    44→    44→        headers: { 'Content-Type': 'application/json' },
    45→    45→        body: JSON.stringify({ username, password }),
    46→    46→      });
    47→    47→      const data = await res.json();
    48→    48→      if (data.success) {
    49→    49→        setAdminData({ isLoggedIn: true });
    50→    50→      } else {
    51→    51→        setError(data.error || 'Login failed');
    52→    52→      }
    53→    53→    } catch { setError('Server error'); }
    54→    54→    setLoading(false);
    55→    55→  }
    56→    56→
    57→    57→  async function handleChangePassword() {
    58→    58→    setCpMsg(null);
    59→    59→    if (!cpCurrentPassword || !cpNewPassword || !cpConfirmPassword) {
    60→    60→      setCpMsg({ type: 'error', text: 'All fields are required' });
    61→    61→      return;
    62→    62→    }
    63→    63→    if (cpNewPassword !== cpConfirmPassword) {
    64→    64→      setCpMsg({ type: 'error', text: 'New password and confirm password do not match' });
    65→    65→      return;
    66→    66→    }
    67→    67→    if (cpNewPassword.length < 4) {
    68→    68→      setCpMsg({ type: 'error', text: 'New password must be at least 4 characters' });
    69→    69→      return;
    70→    70→    }
    71→    71→    setCpLoading(true);
    72→    72→    try {
    73→    73→      const res = await fetch('/api/admin/reset-password', {
    74→    74→        method: 'POST',
    75→    75→        headers: { 'Content-Type': 'application/json' },
    76→    76→        body: JSON.stringify({ currentPassword: cpCurrentPassword, newPassword: cpNewPassword }),
    77→    77→      });
    78→    78→      const data = await res.json();
    79→    79→      if (data.success) {
    80→    80→        setCpMsg({ type: 'success', text: 'Password updated successfully! Use new password to login next time.' });
    81→    81→        setCpCurrentPassword('');
    82→    82→        setCpNewPassword('');
    83→    83→        setCpConfirmPassword('');
    84→    84→      } else {
    85→    85→        setCpMsg({ type: 'error', text: data.error || 'Failed to update password' });
    86→    86→      }
    87→    87→    } catch {
    88→    88→      setCpMsg({ type: 'error', text: 'Server error. Please try again.' });
    89→    89→    }
    90→    90→    setCpLoading(false);
    91→    91→  }
    92→    92→
    93→    93→  return (
    94→    94→    <div className="min-h-[70vh] flex items-center justify-center px-4 pt-12">
    95→    95→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="w-full">
    96→    96→        <Card className="w-full max-w-md mx-auto shadow-xl border-0">
    97→    97→          <CardContent className="p-5 sm:p-6">
    98→    98→            <div className="text-center mb-6">
    99→    99→              <div className="w-16 h-16 rounded-2xl bg-blue-700 flex items-center justify-center mx-auto mb-4">
   100→   100→                <Shield className="w-8 h-8 text-white" />
   101→   101→              </div>
   102→   102→              <h2 className="text-xl font-bold text-gray-900">Admin Login</h2>
   103→   103→              <p className="text-xs text-muted-foreground mt-1">Enter credentials to manage tests & questions</p>
   104→   104→            </div>
   105→   105→            <div className="space-y-4">
   106→   106→              <div>
   107→   107→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Username</Label>
   108→   108→                <Input value={username} onChange={(e) => setUsername(e.target.value)} placeholder="admin" className="h-11" />
   109→   109→              </div>
   110→   110→              <div>
   111→   111→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Password</Label>
   112→   112→                <div className="relative">
   113→   113→                  <Input
   114→   114→                    type={showPassword ? 'text' : 'password'}
   115→   115→                    value={password}
   116→   116→                    onChange={(e) => setPassword(e.target.value)}
   117→   117→                    placeholder="••••••••"
   118→   118→                    className="h-11 pr-10"
   119→   119→                    onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
   120→   120→                  />
   121→   121→                  <button
   122→   122→                    type="button"
   123→   123→                    onClick={() => setShowPassword(!showPassword)}
   124→   124→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   125→   125→                  >
   126→   126→                    {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   127→   127→                  </button>
   128→   128→                </div>
   129→   129→              </div>
   130→   130→              <p className="text-right">
   131→   131→                <button onClick={() => setChangePasswordOpen(true)} className="text-xs text-blue-600 hover:text-blue-700 font-medium">Change Password?</button>
   132→   132→              </p>
   133→   133→              {error && (
   134→   134→                <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="bg-red-50 border border-red-200 text-red-700 text-sm rounded-xl p-3 flex items-center gap-2">
   135→   135→                  <AlertTriangle className="w-4 h-4 shrink-0" /> {error}
   136→   136→                </motion.div>
   137→   137→              )}
   138→   138→              <Button className="w-full bg-blue-700 hover:bg-blue-800 h-11 font-semibold" onClick={handleLogin} disabled={loading || !username || !password}>
   139→   139→                {loading ? 'Logging in...' : 'Login to Admin Panel'}
   140→   140→              </Button>
   141→   141→            </div>
   142→   142→          </CardContent>
   143→   143→        </Card>
   144→   144→
   145→   145→        {/* Admin Change Password Dialog */}
   146→   146→        <Dialog open={changePasswordOpen} onOpenChange={(open) => { if (!open) { setChangePasswordOpen(false); setCpMsg(null); setCpCurrentPassword(''); setCpNewPassword(''); setCpConfirmPassword(''); } }}>
   147→   147→          <DialogContent className="max-w-[calc(100vw-2rem)] sm:max-w-sm">
   148→   148→            <DialogHeader>
   149→   149→              <DialogTitle className="flex items-center gap-2 text-base"><Shield className="w-5 h-5 text-blue-600" />Change Admin Password</DialogTitle>
   150→   150→              <DialogDescription className="text-xs">Enter current password, then set new password</DialogDescription>
   151→   151→            </DialogHeader>
   152→   152→            <div className="space-y-3">
   153→   153→              {cpMsg && (
   154→   154→                <motion.div initial={{ opacity: 0, y: -5 }} animate={{ opacity: 1, y: 0 }}
   155→   155→                  className={`p-3 rounded-xl text-sm flex items-center gap-2 border ${
   156→   156→                    cpMsg.type === 'success' ? 'bg-green-50 border-green-200 text-green-700' : 'bg-red-50 border-red-200 text-red-700'
   157→   157→                  }`}>
   158→   158→                  {cpMsg.type === 'success' ? <CheckCircle2 className="w-4 h-4 shrink-0" /> : <AlertTriangle className="w-4 h-4 shrink-0" />}
   159→   159→                  {cpMsg.text}
   160→   160→                </motion.div>
   161→   161→              )}
   162→   162→              <div>
   163→   163→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Current Password</Label>
   164→   164→                <Input type="password" value={cpCurrentPassword} onChange={(e) => setCpCurrentPassword(e.target.value)} placeholder="Enter current password" className="h-11" />
   165→   165→              </div>
   166→   166→              <div>
   167→   167→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">New Password</Label>
   168→   168→                <Input type="password" value={cpNewPassword} onChange={(e) => setCpNewPassword(e.target.value)} placeholder="Enter new password (min 4 chars)" className="h-11" />
   169→   169→              </div>
   170→   170→              <div>
   171→   171→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Confirm New Password</Label>
   172→   172→                <Input type="password" value={cpConfirmPassword} onChange={(e) => setCpConfirmPassword(e.target.value)} placeholder="Confirm new password" className="h-11" onKeyDown={(e) => e.key === 'Enter' && handleChangePassword()} />
   173→   173→              </div>
   174→   174→              <div className="flex gap-2">
   175→   175→                <DialogClose asChild>
   176→   176→                  <Button variant="outline" className="flex-1 h-11">Cancel</Button>
   177→   177→                </DialogClose>
   178→   178→                <Button className="flex-1 h-11 bg-blue-700 hover:bg-blue-800 font-semibold" onClick={handleChangePassword} disabled={cpLoading || !cpCurrentPassword || !cpNewPassword || !cpConfirmPassword}>
   179→   179→                  {cpLoading ? <><Loader2 className="w-4 h-4 mr-2 animate-spin" />Updating...</> : <><Save className="w-4 h-4 mr-2" />Update Password</>}
   180→   180→                </Button>
   181→   181→              </div>
   182→   182→            </div>
   183→   183→          </DialogContent>
   184→   184→        </Dialog>
   185→   185→      </motion.div>
   186→   186→    </div>
   187→   187→  );
   188→   188→}

   842→   842→  function fillQuestionsFromExtract(extracted: QuestionForm[]): number {
   843→   843→    if (!extracted || extracted.length === 0) {
   844→   844→      setFillStatus({ msg: '⚠️ No questions received from VLM to fill.', type: 'error' });
   845→   845→      return 0;
   846→   846→    }
   847→   847→    const count = extracted.length;
   848→   848→    setFillStatus({ msg: `🔧 Filling ${count} questions into boxes...`, type: 'info' });
   849→   849→
   850→   850→    // Use ref to track where to start filling (supports multi-image upload)
   851→   851→    let startIdx = fillStartIndexRef.current;
   852→   852→
   853→   853→    setQuestions(prev => {
   854→   854→      const next = [...prev];
   855→   855→      // Clamp startIdx if it's beyond array length
   856→   856→      if (startIdx > next.length) startIdx = next.length;
   857→   857→      // Fill each extracted question into the boxes
   858→   858→      for (let k = 0; k < extracted.length; k++) {
   859→   859→        const targetIdx = startIdx + k;
   860→   860→        if (targetIdx < next.length) {
   861→   861→          // Replace entire box with extracted data (not merge — merge keeps empty defaults)
   862→   862→          next[targetIdx] = { ...extracted[k] };
   863→   863→        } else {
   864→   864→          next.push(extracted[k]);
   865→   865→        }
   866→   866→      }
   867→   867→      return next;
   868→   868→    });
   869→   869→
   870→   870→    // Advance the fill index for next image batch
   871→   871→    fillStartIndexRef.current = startIdx + count;
   872→   872→
   873→   873→    setTimeout(() => {
   874→   874→      setFillStatus({ msg: `✅ ${count} questions filled into boxes!`, type: 'success' });
   875→   875→      setTimeout(() => setFillStatus(null), 8000);
   876→   876→    }, 100);
   877→   877→    return count;
   878→   878→  }

   880→   880→  async function handleImageExtract(e: React.ChangeEvent<HTMLInputElement>) {
   881→   881→    const files = e.target.files;
   882→   882→    if (!files || files.length === 0) return;
   883→   883→    const fileList = Array.from(files);
   884→   884→    setExtractingImage(true);
   885→   885→    setExtractElapsed(0);
   886→   886→    const startTime = Date.now();

   888→   888→      setExtractElapsed(Math.floor((Date.now() - startTime) / 1000));
   889→   889→    }, 1000);
   890→   890→    try {
   891→   891→      let totalExtracted = 0;
   892→   892→      let hasError = false;
   893→   893→      for (let i = 0; i < fileList.length; i++) {
   894→   894→        const file = fileList[i];
   895→   895→        const formData = new FormData();
   896→   896→        formData.append('image', file);
   897→   897→        const controller = new AbortController();
   898→   898→        const timeoutId = setTimeout(() => controller.abort(), 300000);
   899→   899→        try {
   900→   900→          setFillStatus({ msg: `📤 Image ${i + 1}/${fileList.length}: Scanning with AI... (1-3 min)`, type: 'info' });
   901→   901→          // Use Next.js API route with z-ai-web-dev-sdk VLM + Grok fallback
   902→   902→          const res = await fetch('/api/admin/extract-question', { method: 'POST', body: formData, signal: controller.signal });
   903→   903→          clearTimeout(timeoutId);
   904→   904→          if (!res.ok) {
   905→   905→            const errText = await res.text().catch(() => 'Unknown error');
   906→   906→            setFillStatus({ msg: `❌ Image ${i + 1}: Server error (HTTP ${res.status}). ${errText.substring(0, 80)}`, type: 'error' });
   907→   907→            toast.error(`Image ${i + 1}: Server error ${res.status}`);
   908→   908→            hasError = true;
   909→   909→            continue;
   910→   910→          }
   911→   911→          const data = await res.json();
   912→   912→          if (data.success && data.questions && data.questions.length > 0) {
   913→   913→            const newQuestions: QuestionForm[] = data.questions.map((q: any) => ({
   914→   914→              question: q.question || '', optionA: q.optionA || '', optionB: q.optionB || '', optionC: q.optionC || '', optionD: q.optionD || '',
   915→   915→              correctOption: q.correctOption || 'A', explanation: q.explanation || '', section: q.section || 'General', negativeMark: q.negativeMark || '0',
   916→   916→            }));
   917→   917→            const filled = fillQuestionsFromExtract(newQuestions);
   918→   918→            totalExtracted += filled;
   919→   919→            toast.success(`Image ${i + 1}: ${filled} questions extracted & filled!`);
   920→   920→          } else if (data.success && data.question) {
   921→   921→            const newQ: QuestionForm = {
   922→   922→              question: data.question.question || '', optionA: data.question.optionA || '', optionB: data.question.optionB || '', optionC: data.question.optionC || '', optionD: data.question.optionD || '',
   923→   923→              correctOption: data.question.correctOption || 'A', explanation: data.question.explanation || '', section: data.question.section || 'General', negativeMark: data.question.negativeMark || '0',
   924→   924→            };
   925→   925→            const filled = fillQuestionsFromExtract([newQ]);
   926→   926→            totalExtracted += filled;
   927→   927→            toast.success(`Image ${i + 1}: 1 question extracted & filled!`);
   928→   928→          } else {
   929→   929→            hasError = true;
   930→   930→            const errMsg = data.error || 'Failed to extract questions';
   931→   931→            setFillStatus({ msg: `❌ ${errMsg}`, type: 'error' });
   932→   932→            toast.error(`Image ${i + 1}: ${errMsg}`);
   933→   933→          }
   934→   934→        } catch (err: any) {
   935→   935→          clearTimeout(timeoutId);
   936→   936→          hasError = true;
   937→   937→          if (err?.name === 'AbortError') {
   938→   938→            setFillStatus({ msg: '❌ Timed out (5 min). Try a smaller/clearer image.', type: 'error' });
   939→   939→          } else {
   940→   940→            setFillStatus({ msg: `❌ ${err?.message || 'Network error. Check your connection.'}`, type: 'error' });
   941→   941→          }
   942→   942→        }
   943→   943→      }
   944→   944→      if (totalExtracted > 0) {
   945→   945→        setFillStatus({ msg: `✅ Done! Total ${totalExtracted} questions extracted & filled.`, type: 'success' });
   946→   946→        toast.success(`🎉 Total: ${totalExtracted} questions extracted!`);
   947→   947→      } else if (hasError) {
   948→   948→        setFillStatus({ msg: `❌ No questions extracted. Check error messages above.`, type: 'error' });
   949→   949→      } else {
   950→   950→        setFillStatus({ msg: '⚠️ No questions could be extracted.', type: 'error' });
   951→   951→        toast.error('No questions could be extracted. Empty boxes remain for manual entry.');
   952→   952→      }
   953→   953→    } finally {
   954→   954→      clearInterval(timerInterval);
   955→   955→      setExtractingImage(false);
   956→   956→      setExtractElapsed(0);
   957→   957→      e.target.value = '';
   958→   958→    }
   959→   959→  }
