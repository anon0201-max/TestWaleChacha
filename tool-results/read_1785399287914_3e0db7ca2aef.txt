    23→// ==================== ADMIN LOGIN ====================
    24→function AdminLogin() {
    25→  const { setAdminData } = useAppStore();
    26→  const [username, setUsername] = useState('');
    27→  const [password, setPassword] = useState('');
    28→  const [error, setError] = useState('');
    29→  const [loading, setLoading] = useState(false);
    30→  const [showPassword, setShowPassword] = useState(false);
    31→  const [changePasswordOpen, setChangePasswordOpen] = useState(false);
    32→  const [cpCurrentPassword, setCpCurrentPassword] = useState('');
    33→  const [cpNewPassword, setCpNewPassword] = useState('');
    34→  const [cpConfirmPassword, setCpConfirmPassword] = useState('');
    35→  const [cpLoading, setCpLoading] = useState(false);
    36→  const [cpMsg, setCpMsg] = useState<{ type: 'success' | 'error'; text: string } | null>(null);
    37→
    38→  async function handleLogin() {
    39→    setLoading(true);
    40→    setError('');
    41→    try {
    42→      const res = await fetch('/api/admin/login', {
    43→        method: 'POST',
    44→        headers: { 'Content-Type': 'application/json' },
    45→        body: JSON.stringify({ username, password }),
    46→      });
    47→      const data = await res.json();
    48→      if (data.success) {
    49→        setAdminData({ isLoggedIn: true });
    50→      } else {
    51→        setError(data.error || 'Login failed');
    52→      }
    53→    } catch { setError('Server error'); }
    54→    setLoading(false);
    55→  }
    56→
    57→  async function handleChangePassword() {
    58→    setCpMsg(null);
    59→    if (!cpCurrentPassword || !cpNewPassword || !cpConfirmPassword) {
    60→      setCpMsg({ type: 'error', text: 'All fields are required' });
    61→      return;
    62→    }
    63→    if (cpNewPassword !== cpConfirmPassword) {
    64→      setCpMsg({ type: 'error', text: 'New password and confirm password do not match' });
    65→      return;
    66→    }
    67→    if (cpNewPassword.length < 4) {
    68→      setCpMsg({ type: 'error', text: 'New password must be at least 4 characters' });
    69→      return;
    70→    }
    71→    setCpLoading(true);
    72→    try {
    73→      const res = await fetch('/api/admin/reset-password', {
    74→        method: 'POST',
    75→        headers: { 'Content-Type': 'application/json' },
    76→        body: JSON.stringify({ currentPassword: cpCurrentPassword, newPassword: cpNewPassword }),
    77→      });
    78→      const data = await res.json();
    79→      if (data.success) {
    80→        setCpMsg({ type: 'success', text: 'Password updated successfully! Use new password to login next time.' });
    81→        setCpCurrentPassword('');
    82→        setCpNewPassword('');
    83→        setCpConfirmPassword('');
    84→      } else {
    85→        setCpMsg({ type: 'error', text: data.error || 'Failed to update password' });
    86→      }
    87→    } catch {
    88→      setCpMsg({ type: 'error', text: 'Server error. Please try again.' });
    89→    }
    90→    setCpLoading(false);
    91→  }
    92→
    93→  return (
    94→    <div className="min-h-[70vh] flex items-center justify-center px-4 pt-12">
    95→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="w-full">
    96→        <Card className="w-full max-w-md mx-auto shadow-xl border-0">
    97→          <CardContent className="p-5 sm:p-6">
    98→            <div className="text-center mb-6">
    99→              <div className="w-16 h-16 rounded-2xl bg-blue-700 flex items-center justify-center mx-auto mb-4">
   100→                <Shield className="w-8 h-8 text-white" />
   101→              </div>
   102→              <h2 className="text-xl font-bold text-gray-900">Admin Login</h2>
   103→              <p className="text-xs text-muted-foreground mt-1">Enter credentials to manage tests & questions</p>
   104→            </div>
   105→            <div className="space-y-4">
   106→              <div>
   107→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Username</Label>
   108→                <Input value={username} onChange={(e) => setUsername(e.target.value)} placeholder="admin" className="h-11" />
   109→              </div>
   110→              <div>
   111→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Password</Label>
   112→                <div className="relative">
   113→                  <Input
   114→                    type={showPassword ? 'text' : 'password'}
   115→                    value={password}
   116→                    onChange={(e) => setPassword(e.target.value)}
   117→                    placeholder="••••••••"
   118→                    className="h-11 pr-10"
   119→                    onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
   120→                  />
   121→                  <button
   122→                    type="button"
   123→                    onClick={() => setShowPassword(!showPassword)}
   124→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   125→                  >
   126→                    {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   127→                  </button>
   128→                </div>
   129→              </div>
   130→              <p className="text-right">
   131→                <button onClick={() => setChangePasswordOpen(true)} className="text-xs text-blue-600 hover:text-blue-700 font-medium">Change Password?</button>
   132→              </p>
   133→              {error && (
   134→                <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="bg-red-50 border border-red-200 text-red-700 text-sm rounded-xl p-3 flex items-center gap-2">
   135→                  <AlertTriangle className="w-4 h-4 shrink-0" /> {error}
   136→                </motion.div>
   137→              )}
   138→              <Button className="w-full bg-blue-700 hover:bg-blue-800 h-11 font-semibold" onClick={handleLogin} disabled={loading || !username || !password}>
   139→                {loading ? 'Logging in...' : 'Login to Admin Panel'}
   140→              </Button>
   141→            </div>
   142→          </CardContent>
   143→        </Card>
   144→
   145→        {/* Admin Change Password Dialog */}
   146→        <Dialog open={changePasswordOpen} onOpenChange={(open) => { if (!open) { setChangePasswordOpen(false); setCpMsg(null); setCpCurrentPassword(''); setCpNewPassword(''); setCpConfirmPassword(''); } }}>
   147→          <DialogContent className="max-w-[calc(100vw-2rem)] sm:max-w-sm">
   148→            <DialogHeader>
   149→              <DialogTitle className="flex items-center gap-2 text-base"><Shield className="w-5 h-5 text-blue-600" />Change Admin Password</DialogTitle>
   150→              <DialogDescription className="text-xs">Enter current password, then set new password</DialogDescription>
   151→            </DialogHeader>
   152→            <div className="space-y-3">
   153→              {cpMsg && (
   154→                <motion.div initial={{ opacity: 0, y: -5 }} animate={{ opacity: 1, y: 0 }}
   155→                  className={`p-3 rounded-xl text-sm flex items-center gap-2 border ${
   156→                    cpMsg.type === 'success' ? 'bg-green-50 border-green-200 text-green-700' : 'bg-red-50 border-red-200 text-red-700'
   157→                  }`}>
   158→                  {cpMsg.type === 'success' ? <CheckCircle2 className="w-4 h-4 shrink-0" /> : <AlertTriangle className="w-4 h-4 shrink-0" />}
   159→                  {cpMsg.text}
   160→                </motion.div>
   161→              )}
   162→              <div>
   163→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Current Password</Label>
   164→                <Input type="password" value={cpCurrentPassword} onChange={(e) => setCpCurrentPassword(e.target.value)} placeholder="Enter current password" className="h-11" />
   165→              </div>
   166→              <div>
   167→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">New Password</Label>
   168→                <Input type="password" value={cpNewPassword} onChange={(e) => setCpNewPassword(e.target.value)} placeholder="Enter new password (min 4 chars)" className="h-11" />
   169→              </div>
   170→              <div>
   171→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Confirm New Password</Label>
   172→                <Input type="password" value={cpConfirmPassword} onChange={(e) => setCpConfirmPassword(e.target.value)} placeholder="Confirm new password" className="h-11" onKeyDown={(e) => e.key === 'Enter' && handleChangePassword()} />
   173→              </div>
   174→              <div className="flex gap-2">
   175→                <DialogClose asChild>
   176→                  <Button variant="outline" className="flex-1 h-11">Cancel</Button>
   177→                </DialogClose>
   178→                <Button className="flex-1 h-11 bg-blue-700 hover:bg-blue-800 font-semibold" onClick={handleChangePassword} disabled={cpLoading || !cpCurrentPassword || !cpNewPassword || !cpConfirmPassword}>
   179→                  {cpLoading ? <><Loader2 className="w-4 h-4 mr-2 animate-spin" />Updating...</> : <><Save className="w-4 h-4 mr-2" />Update Password</>}
   180→                </Button>
   181→              </div>
   182→            </div>
   183→          </DialogContent>
   184→        </Dialog>
   185→      </motion.div>
   186→    </div>
   187→  );
   188→}

   420→// ==================== CATEGORIES TAB ====================
   421→function AdminCategoriesTab({ onRefresh }: { onRefresh: () => void }) {
   422→  const { categories, setCategories } = useAppStore();
   423→  const [newName, setNewName] = useState('');
   424→  const [newSlug, setNewSlug] = useState('');
   425→  const [newExamType, setNewExamType] = useState('General');
   426→  const [adding, setAdding] = useState(false);
   427→
   428→  async function handleAdd() {
   429→    if (!newName || !newSlug) return;
   430→    setAdding(true);
   431→    await fetch('/api/admin/categories', {
   432→      method: 'POST',
   433→      headers: { 'Content-Type': 'application/json' },

   939→  function fillQuestionsFromExtract(extracted: QuestionForm[]): number {
   940→    if (!extracted || extracted.length === 0) {
   941→      setFillStatus({ msg: '⚠️ No questions received from VLM to fill.', type: 'error' });
   942→      return 0;
   943→    }
   944→    const count = extracted.length;
   945→    setFillStatus({ msg: `🔧 Filling ${count} questions into boxes...`, type: 'info' });
   946→
   947→    // Use ref to track where to start filling (supports multi-image upload)
   948→    let startIdx = fillStartIndexRef.current;
   949→
   950→    setQuestions(prev => {
   951→      const next = [...prev];
   952→      // Clamp startIdx if it's beyond array length
   953→      if (startIdx > next.length) startIdx = next.length;
   954→      // Fill each extracted question into the boxes
   955→      for (let k = 0; k < extracted.length; k++) {
   956→        const targetIdx = startIdx + k;
   957→        if (targetIdx < next.length) {
   958→          // Replace entire box with extracted data (not merge — merge keeps empty defaults)
   959→          next[targetIdx] = { ...extracted[k] };
   960→        } else {
   961→          next.push(extracted[k]);
   962→        }
   963→      }
   964→      return next;
   965→    });
   966→
   967→    // Advance the fill index for next image batch
   968→    fillStartIndexRef.current = startIdx + count;
   969→
   970→    setTimeout(() => {
   971→      setFillStatus({ msg: `✅ ${count} questions filled into boxes!`, type: 'success' });
   972→      setTimeout(() => setFillStatus(null), 8000);
   973→    }, 100);
   974→    return count;
   975→  }

   977→  async function handleImageExtract(e: React.ChangeEvent<HTMLInputElement>) {
   978→    const files = e.target.files;
   979→    if (!files || files.length === 0) return;
   980→    const fileList = Array.from(files);
   981→    setExtractingImage(true);
   982→    setExtractElapsed(0);
   983→    const startTime = Date.now();
   984→    const timerInterval = setInterval(() => {
   985→      setExtractElapsed(Math.floor((Date.now() - startTime) / 1000));
   986→    }, 1000);
   987→    try {
   988→      let totalExtracted = 0;
   989→      let hasError = false;
   990→      for (let i = 0; i < fileList.length; i++) {
   991→        const file = fileList[i];
   992→        const formData = new FormData();
   993→        formData.append('image', file);
   994→        const controller = new AbortController();
   995→        const timeoutId = setTimeout(() => controller.abort(), 300000);
   996→        try {
   997→          setFillStatus({ msg: `📤 Image ${i + 1}/${fileList.length}: Scanning with AI... (1-3 min)`, type: 'info' });
   998→          // Use Next.js API route with z-ai-web-dev-sdk VLM + Grok fallback
   999→          const res = await fetch('/api/admin/extract-question', { method: 'POST', body: formData, signal: controller.signal });
  1000→          clearTimeout(timeoutId);
  1001→          if (!res.ok) {
  1002→            const errText = await res.text().catch(() => 'Unknown error');
  1003→            setFillStatus({ msg: `❌ Image ${i + 1}: Server error (HTTP ${res.status}). ${errText.substring(0, 80)}`, type: 'error' });
  1004→            toast.error(`Image ${i + 1}: Server error ${res.status}`);
  1005→            hasError = true;
  1006→            continue;
  1007→          }
  1008→          const data = await res.json();
  1009→          if (data.success && data.questions && data.questions.length > 0) {
  1010→            const newQuestions: QuestionForm[] = data.questions.map((q: any) => ({
  1011→              question: q.question || '', optionA: q.optionA || '', optionB: q.optionB || '', optionC: q.optionC || '', optionD: q.optionD || '',
  1012→              correctOption: q.correctOption || 'A', explanation: q.explanation || '', section: q.section || 'General', negativeMark: q.negativeMark || '0',
  1013→            }));
  1014→            const filled = fillQuestionsFromExtract(newQuestions);
  1015→            totalExtracted += filled;
  1016→            toast.success(`Image ${i + 1}: ${filled} questions extracted & filled!`);
  1017→          } else if (data.success && data.question) {
  1018→            const newQ: QuestionForm = {
  1019→              question: data.question.question || '', optionA: data.question.optionA || '', optionB: data.question.optionB || '', optionC: data.question.optionC || '', optionD: data.question.optionD || '',
  1020→              correctOption: data.question.correctOption || 'A', explanation: data.question.explanation || '', section: data.question.section || 'General', negativeMark: data.question.negativeMark || '0',
  1021→            };
  1022→            const filled = fillQuestionsFromExtract([newQ]);
  1023→            totalExtracted += filled;
  1024→            toast.success(`Image ${i + 1}: 1 question extracted & filled!`);
  1025→          } else {
  1026→            hasError = true;
  1027→            const errMsg = data.error || 'Failed to extract questions';
  1028→            setFillStatus({ msg: `❌ ${errMsg}`, type: 'error' });
  1029→            toast.error(`Image ${i + 1}: ${errMsg}`);
  1030→          }
  1031→        } catch (err: any) {
  1032→          clearTimeout(timeoutId);
  1033→          hasError = true;
  1034→          if (err?.name === 'AbortError') {
  1035→            setFillStatus({ msg: '❌ Timed out (5 min). Try a smaller/clearer image.', type: 'error' });
  1036→          } else {
  1037→            setFillStatus({ msg: `❌ ${err?.message || 'Network error. Check your connection.'}`, type: 'error' });
  1038→          }
  1039→        }
  1040→      }
  1041→      if (totalExtracted > 0) {
  1042→        setFillStatus({ msg: `✅ Done! Total ${totalExtracted} questions extracted & filled.`, type: 'success' });
  1043→        toast.success(`🎉 Total: ${totalExtracted} questions extracted!`);
  1044→      } else if (hasError) {
  1045→        setFillStatus({ msg: `❌ No questions extracted. Check error messages above.`, type: 'error' });
  1046→      } else {
  1047→        setFillStatus({ msg: '⚠️ No questions could be extracted.', type: 'error' });
  1048→        toast.error('No questions could be extracted. Empty boxes remain for manual entry.');
  1049→      }
  1050→    } finally {
  1051→      clearInterval(timerInterval);
  1052→      setExtractingImage(false);
  1053→      setExtractElapsed(0);
  1054→      e.target.value = '';
  1055→    }
  1056→  }

  1642→function AdminSettingsTab() {

  1648→
  1649→  async function handleResetPassword() {
  1650→    setMessage(null);
  1651→    if (!currentPassword || !newPassword || !confirmPassword) {
  1652→      setMessage({ type: 'error', text: 'All fields are required' });
  1653→      return;
  1654→    }
  1655→    if (newPassword !== confirmPassword) {
  1656→      setMessage({ type: 'error', text: 'Passwords do not match' });
  1657→      return;
  1658→    }
  1659→    if (newPassword.length < 4) {
  1660→      setMessage({ type: 'error', text: 'New password must be at least 4 characters' });
  1661→      return;
  1662→    }
  1663→    setLoading(true);
  1664→    try {
  1665→      const res = await fetch('/api/admin/reset-password', {
  1666→        method: 'POST',
  1667→        headers: { 'Content-Type': 'application/json' },
  1668→        body: JSON.stringify({ currentPassword, newPassword }),
  1669→      });
  1670→      const data = await res.json();
  1671→      if (data.success) {
  1672→        setMessage({ type: 'success', text: 'Password updated successfully!' });
  1673→        setCurrentPassword('');
  1674→        setNewPassword('');
  1675→        setConfirmPassword('');
  1676→      } else {
  1677→        setMessage({ type: 'error', text: data.error || 'Failed to reset password' });
  1678→      }
  1679→    } catch {
  1680→      setMessage({ type: 'error', text: 'Server error.' });
  1681→    }
  1682→    setLoading(false);
  1683→  }
  1684→
  1685→  return (
  1686→    <div className="space-y-3 sm:space-y-4">
  1687→      <Card className="border-0 shadow-sm">
  1688→        <CardContent className="p-4 sm:p-5">
  1689→          <h3 className="text-sm font-semibold mb-1 flex items-center gap-2">
  1690→            <Shield className="w-4 h-4 text-blue-600" /> Reset Admin Password

  1692→          <p className="text-xs text-muted-foreground mb-4">Change your admin login password.</p>
  1693→
  1694→          {message && (
  1695→            <motion.div initial={{ opacity: 0, y: -5 }} animate={{ opacity: 1, y: 0 }}
  1696→              className={`mb-4 p-3 rounded-xl text-sm flex items-center gap-2 border ${

  1698→              }`}>
  1699→              {message.type === 'success' ? <CheckCircle2 className="w-4 h-4 shrink-0" /> : <AlertTriangle className="w-4 h-4 shrink-0" />}
  1700→              <span className="break-all">{message.text}</span>
  1701→            </motion.div>
  1702→          )}
  1703→
  1704→          <div className="space-y-3 max-w-md">
  1705→            <div>
  1706→              <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Current Password</Label>
