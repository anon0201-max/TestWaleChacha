    22→// ==================== ADMIN LOGIN ====================
    23→function AdminLogin() {
    24→  const { setAdminData } = useAppStore();
    25→  const [username, setUsername] = useState('');
    26→  const [password, setPassword] = useState('');
    27→  const [error, setError] = useState('');
    28→  const [loading, setLoading] = useState(false);
    29→  const [showPassword, setShowPassword] = useState(false);
    30→
    31→  async function handleLogin() {
    32→    setLoading(true);
    33→    setError('');
    34→    try {
    35→      const res = await fetch('/api/admin/login', {
    36→        method: 'POST',
    37→        headers: { 'Content-Type': 'application/json' },
    38→        body: JSON.stringify({ username, password }),
    39→      });
    40→      const data = await res.json();
    41→      if (data.success) {
    42→        setAdminData({ isLoggedIn: true });
    43→      } else {
    44→        setError(data.error || 'Login failed');
    45→      }
    46→    } catch { setError('Server error'); }
    47→    setLoading(false);
    48→  }
    49→
    50→  return (
    51→    <div className="min-h-[70vh] flex items-center justify-center">
    52→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
    53→        <Card className="w-full max-w-sm shadow-xl border-0">
    54→          <CardContent className="p-6">
    55→            <div className="text-center mb-6">
    56→              <div className="w-16 h-16 rounded-2xl bg-blue-700 flex items-center justify-center mx-auto mb-4">
    57→                <Shield className="w-8 h-8 text-white" />
    58→              </div>
    59→              <h2 className="text-xl font-bold text-gray-900">Admin Login</h2>
    60→              <p className="text-xs text-muted-foreground mt-1">Enter credentials to manage tests & questions</p>
    61→            </div>
    62→            <div className="space-y-4">
    63→              <div>
    64→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Username</Label>
    65→                <Input value={username} onChange={(e) => setUsername(e.target.value)} placeholder="admin" className="h-11" />
    66→              </div>
    67→              <div>
    68→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Password</Label>
    69→                <div className="relative">
    70→                  <Input
    71→                    type={showPassword ? 'text' : 'password'}
    72→                    value={password}
    73→                    onChange={(e) => setPassword(e.target.value)}
    74→                    placeholder="••••••••"
    75→                    className="h-11 pr-10"
    76→                    onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
    77→                  />
    78→                  <button
    79→                    type="button"
    80→                    onClick={() => setShowPassword(!showPassword)}
    81→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
    82→                  >
    83→                    {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
    84→                  </button>
    85→                </div>
    86→              </div>
    87→              {error && (
    88→                <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="bg-red-50 border border-red-200 text-red-700 text-sm rounded-xl p-3 flex items-center gap-2">
    89→                  <AlertTriangle className="w-4 h-4 shrink-0" /> {error}
    90→                </motion.div>
    91→              )}
    92→              <Button className="w-full bg-blue-700 hover:bg-blue-800 h-11 font-semibold" onClick={handleLogin} disabled={loading || !username || !password}>
    93→                {loading ? 'Logging in...' : 'Login to Admin Panel'}
    94→              </Button>
    95→            </div>
    96→          </CardContent>
    97→        </Card>
    98→      </motion.div>
    99→    </div>
   100→  );
   101→}

   335→  async function handleAdd() {
   336→    if (!newName || !newSlug) return;
   337→    setAdding(true);
   338→    await fetch('/api/admin/categories', {
   339→      method: 'POST',
   340→      headers: { 'Content-Type': 'application/json' },
   341→      body: JSON.stringify({ name: newName, slug: newSlug, examType: newExamType }),
   342→    });
   343→    setNewName(''); setNewSlug('');
   344→    setAdding(false);
   345→    onRefresh();
   346→  }

   351→      method: 'DELETE',
   352→      headers: { 'Content-Type': 'application/json' },

   354→    });
   355→    onRefresh();
   356→  }
   357→
   358→  return (
   359→    <div className="space-y-4">

   427→// ==================== TESTS TAB ====================
   428→function AdminTestsTab({ onRefresh }: { onRefresh: () => void }) {
   429→  const { tests } = useAppStore();
   430→  const [expandedTestId, setExpandedTestId] = useState<string | null>(null);
   431→  const [testQuestions, setTestQuestions] = useState<Record<string, unknown[]>>({});
   432→  const [loadingQ, setLoadingQ] = useState<string | null>(null);
   433→
   434→  async function fetchQuestions(testId: string) {
   435→    if (testQuestions[testId]) {
   436→      setExpandedTestId(expandedTestId === testId ? null : testId);
   437→      return;
   438→    }
   439→    setLoadingQ(testId);
   440→    try {
   441→      const res = await fetch(`/api/tests/${testId}`);
   442→      if (res.ok) {
   443→        const data = await res.json();
   444→        setTestQuestions(prev => ({ ...prev, [testId]: data.questions || [] }));
   445→        setExpandedTestId(testId);
   446→      }
   447→    } catch {}
   448→    setLoadingQ(null);
   449→  }
   450→
   451→  async function handleDeleteTest(id: string) {
   452→    if (!confirm('Delete this test and ALL its questions? This cannot be undone.')) return;
   453→    await fetch('/api/admin/tests', {
   454→      method: 'DELETE',
   455→      headers: { 'Content-Type': 'application/json' },
   456→      body: JSON.stringify({ id }),
   457→    });
   458→    onRefresh();
   459→  }
   460→
   461→  async function handleDeleteQuestion(questionId: string, testId: string) {
   462→    if (!confirm('Delete this question?')) return;
   463→    await fetch('/api/admin/tests/questions', {
   464→      method: 'DELETE',
   465→      headers: { 'Content-Type': 'application/json' },

   768→  // Fill existing empty boxes with extracted data — finds first empty box dynamically.

   770→  function fillQuestionsFromExtract(extracted: QuestionForm[]): number {
   771→    if (!extracted || extracted.length === 0) {
   772→      console.warn('⚠️ fillQuestionsFromExtract: no questions to fill');
   773→      setFillStatus({ msg: '⚠️ No questions received from VLM to fill.', type: 'error' });
   774→      return 0;
   775→    }
   776→    console.log(`🔧 fillQuestionsFromExtract: START — filling ${extracted.length} questions`);
   777→    console.log(`🔧 First extracted question:`, extracted[0]?.question?.substring(0, 60));
   778→    setFillStatus({ msg: `🔧 Filling ${extracted.length} questions into boxes...`, type: 'info' });
   779→
   780→    let filledCount = 0;
   781→    setQuestions(prev => {
   782→      console.log(`🔧 setQuestions updater — prev count: ${prev.length}`);
   783→      const next = [...prev];
   784→
   785→      // Find first empty box and fill sequentially from there
   786→      let startIdx = -1;
   787→      for (let i = 0; i < next.length; i++) {
   788→        if (!next[i].question.trim()) {
   789→          startIdx = i;
   790→          break;
   791→        }
   792→      }
   793→      console.log(`🔧 First empty box at index: ${startIdx}`);
   794→
   795→      if (startIdx === -1) {
   796→        // No empty boxes — append all
   797→        console.log(`🔧 No empty boxes — appending ${extracted.length}`);
   798→        for (const ex of extracted) {
   799→          next.push(ex);
   800→          filledCount++;
   801→        }
   802→      } else {
   803→        for (let k = 0; k < extracted.length; k++) {
   804→          const targetIdx = startIdx + k;
   805→          if (targetIdx < next.length) {
   806→            next[targetIdx] = { ...next[targetIdx], ...extracted[k] };
   807→          } else {
   808→            next.push(extracted[k]);
   809→          }
   810→          filledCount++;
   811→        }
   812→        console.log(`🔧 Filled boxes ${startIdx}–${startIdx + extracted.length - 1}`);
   813→      }
   814→
   815→      console.log(`🔧 AFTER fill — total: ${next.length}, Box[0]:`, next[0]?.question?.substring(0, 60));
   816→      return next;
   817→    });
   818→
   819→    // Show success status (slight delay so it appears after state update)
   820→    setTimeout(() => {
   821→      setFillStatus({ msg: `✅ ${extracted.length} questions filled into boxes!`, type: 'success' });
   822→      // Auto-clear after 8 seconds
   823→      setTimeout(() => setFillStatus(null), 8000);
   824→    }, 100);
   825→
   826→    return filledCount;

   829→  async function handleImageExtract(e: React.ChangeEvent<HTMLInputElement>) {
   830→    const files = e.target.files;
   831→    if (!files || files.length === 0) return;
   832→
   833→    const fileList = Array.from(files);

   843→    try {

   852→
   853→        // Use AbortController with 5-minute timeout per file (VLM can take 2-3 min for large images)
   854→        const controller = new AbortController();
   855→        const timeoutId = setTimeout(() => controller.abort(), 300000);
   856→
   857→        try {
   858→          console.log(`📤 Uploading image ${i + 1}/${fileList.length}: ${file.name}`);
   859→          setFillStatus({ msg: `📤 Uploading image to VLM... (this can take 1-3 minutes)`, type: 'info' });

   861→            method: 'POST',
   862→            body: formData,
   863→            signal: controller.signal,
   864→          });
   865→          clearTimeout(timeoutId);
   866→
   867→          console.log(`📡 Response status: ${res.status}`);
   868→          if (!res.ok) {
   869→            const errText = await res.text().catch(() => 'Unknown error');
   870→            console.error(`❌ HTTP ${res.status}:`, errText.substring(0, 200));
   871→            setFillStatus({ msg: `❌ VLM error (HTTP ${res.status}): ${errText.substring(0, 100)}`, type: 'error' });
   872→            toast.error(`Image ${i + 1}: VLM returned HTTP ${res.status}`);
   873→            hasError = true;
   874→            continue;
   875→          }
   876→          const data = await res.json();
   877→          console.log(`📋 Extraction result:`, { success: data.success, count: data.count, hasQuestions: !!data.questions, error: data.error });
   878→
   879→          if (data.success && data.questions && data.questions.length > 0) {
   880→            console.log(`🎉 VLM returned ${data.questions.length} questions — mapping to QuestionForm...`);
   881→            const newQuestions: QuestionForm[] = data.questions.map((q: any) => ({
   882→              question: q.question || '',
   883→              optionA: q.optionA || '',
   884→              optionB: q.optionB || '',
   885→              optionC: q.optionC || '',
   886→              optionD: q.optionD || '',
   887→              correctOption: q.correctOption || 'A',
   888→              explanation: q.explanation || '',
   889→              section: q.section || 'General',
   890→              negativeMark: q.negativeMark || '0',
   891→            }));
   892→            console.log(`🎉 Mapped ${newQuestions.length} questions. First:`, newQuestions[0]);
   893→
   894→            // FILL existing empty boxes with extracted data
   895→            const filled = fillQuestionsFromExtract(newQuestions);
   896→            totalExtracted += newQuestions.length;
   897→            console.log(`✅ Fill complete — ${filled} boxes filled (total extracted so far: ${totalExtracted})`);
   898→            toast.success(`Image ${i + 1}: ${newQuestions.length} questions extracted & filled!`);
   899→          } else if (data.success && data.question) {
   900→            // Fallback: single question
   901→            const newQ: QuestionForm = {
   902→              question: data.question.question || '',
   903→              optionA: data.question.optionA || '',
   904→              optionB: data.question.optionB || '',
   905→              optionC: data.question.optionC || '',
   906→              optionD: data.question.optionD || '',
   907→              correctOption: data.question.correctOption || 'A',
   908→              explanation: data.question.explanation || '',
   909→              section: data.question.section || 'General',
   910→              negativeMark: data.question.negativeMark || '0',
   911→            };
   912→            const filled = fillQuestionsFromExtract([newQ]);
   913→            totalExtracted += 1;
   914→            console.log(`✅ Fill complete — ${filled} boxes filled`);
   915→            toast.success(`Image ${i + 1}: 1 question extracted & filled!`);
   916→          } else {
   917→            hasError = true;
   918→            console.error(`❌ Extraction failed for image ${i + 1}:`, data.error);
   919→            setFillStatus({ msg: `❌ ${data.error || 'Failed to extract questions'}`, type: 'error' });
   920→            toast.error(`Image ${i + 1}: ${data.error || 'Failed to extract'}`);
   921→          }
   922→        } catch (err: any) {
   923→          clearTimeout(timeoutId);
   924→          hasError = true;
   925→          console.error(`❌ Fetch error for image ${i + 1}:`, err);
   926→          if (err?.name === 'AbortError') {
   927→            setFillStatus({ msg: '❌ Timed out (5 min). Try a smaller image.', type: 'error' });
   928→            toast.error(`Image ${i + 1}: Timed out (5 min). Try a smaller image.`);
   929→          } else {
   930→            setFillStatus({ msg: `❌ ${err?.message || 'Failed to extract'}`, type: 'error' });
   931→            toast.error(`Image ${i + 1}: ${err?.message || 'Failed to extract'}`);
   932→          }
   933→        }
   934→      }
   935→
   936→      if (totalExtracted > 0) {
   937→        toast.success(`🎉 Total: ${totalExtracted} questions extracted from ${fileList.length} image${fileList.length > 1 ? 's' : ''} & filled into boxes!`);
   938→      } else if (!hasError) {
   939→        toast.error('No questions could be extracted from the image(s). Empty boxes remain for manual entry.');
   940→      }
   941→    } finally {
   942→      clearInterval(timerInterval);
   943→      setExtractingImage(false);
   944→      setExtractElapsed(0);
   945→      e.target.value = '';
   946→    }
   947→  }
