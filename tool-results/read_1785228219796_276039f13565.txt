    22→// ==================== ADMIN LOGIN ====================
    23→function AdminLogin() {
    24→  const { setAdminData } = useAppStore();
    25→  const [username, setUsername] = useState('');
    26→  const [password, setPassword] = useState('');
    27→  const [error, setError] = useState('');
    28→  const [loading, setLoading] = useState(false);
    29→  const [showPassword, setShowPassword] = useState(false);
    30→  const [adminForgotOpen, setAdminForgotOpen] = useState(false);
    31→  const [forgotNewPassword, setForgotNewPassword] = useState('');
    32→  const [forgotConfirmPassword, setForgotConfirmPassword] = useState('');
    33→  const [forgotLoading, setForgotLoading] = useState(false);
    34→  const [forgotMsg, setForgotMsg] = useState<{ type: 'success' | 'error'; text: string } | null>(null);
    35→
    36→  async function handleLogin() {
    37→    setLoading(true);
    38→    setError('');
    39→    try {
    40→      const res = await fetch('/api/admin/login', {
    41→        method: 'POST',
    42→        headers: { 'Content-Type': 'application/json' },
    43→        body: JSON.stringify({ username, password }),
    44→      });
    45→      const data = await res.json();
    46→      if (data.success) {
    47→        setAdminData({ isLoggedIn: true });
    48→      } else {
    49→        setError(data.error || 'Login failed');
    50→      }
    51→    } catch { setError('Server error'); }
    52→    setLoading(false);
    53→  }
    54→
    55→  async function handleAdminForgotPassword() {
    56→    setForgotMsg(null);
    57→    if (!forgotNewPassword || !forgotConfirmPassword) {
    58→      setForgotMsg({ type: 'error', text: 'All fields are required' });
    59→      return;
    60→    }
    61→    if (forgotNewPassword !== forgotConfirmPassword) {
    62→      setForgotMsg({ type: 'error', text: 'Passwords do not match' });
    63→      return;
    64→    }
    65→    if (forgotNewPassword.length < 4) {
    66→      setForgotMsg({ type: 'error', text: 'Password must be at least 4 characters' });
    67→      return;
    68→    }
    69→    setForgotLoading(true);
    70→    try {
    71→      const res = await fetch('/api/admin/reset-password', {
    72→        method: 'POST',
    73→        headers: { 'Content-Type': 'application/json' },
    74→        body: JSON.stringify({ currentPassword: '', newPassword: forgotNewPassword }),
    75→      });
    76→      const data = await res.json();
    77→      if (data.success) {
    78→        setForgotMsg({ type: 'success', text: 'Admin password reset successfully! You can now login with the new password.' });
    79→        setForgotNewPassword('');
    80→        setForgotConfirmPassword('');
    81→      } else {
    82→        setForgotMsg({ type: 'error', text: data.error || 'Failed to reset password' });
    83→      }
    84→    } catch {
    85→      setForgotMsg({ type: 'error', text: 'Server error. Please try again.' });
    86→    }
    87→    setForgotLoading(false);
    88→  }
    89→
    90→  return (
    91→    <div className="min-h-[70vh] flex items-center justify-center px-4 pt-12">
    92→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="w-full">
    93→        <Card className="w-full max-w-md mx-auto shadow-xl border-0">
    94→          <CardContent className="p-6">
    95→            <div className="text-center mb-6">
    96→              <div className="w-16 h-16 rounded-2xl bg-blue-700 flex items-center justify-center mx-auto mb-4">
    97→                <Shield className="w-8 h-8 text-white" />
    98→              </div>
    99→              <h2 className="text-xl font-bold text-gray-900">Admin Login</h2>
   100→              <p className="text-xs text-muted-foreground mt-1">Enter credentials to manage tests & questions</p>
   101→            </div>
   102→            <div className="space-y-4">
   103→              <div>
   104→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Username</Label>
   105→                <Input value={username} onChange={(e) => setUsername(e.target.value)} placeholder="admin" className="h-11" />
   106→              </div>
   107→              <div>
   108→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Password</Label>
   109→                <div className="relative">
   110→                  <Input
   111→                    type={showPassword ? 'text' : 'password'}
   112→                    value={password}
   113→                    onChange={(e) => setPassword(e.target.value)}
   114→                    placeholder="••••••••"
   115→                    className="h-11 pr-10"
   116→                    onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
   117→                  />
   118→                  <button
   119→                    type="button"
   120→                    onClick={() => setShowPassword(!showPassword)}
   121→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   122→                  >
   123→                    {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   124→                  </button>
   125→                </div>
   126→              </div>
   127→              <p className="text-right">
   128→                <button onClick={() => setAdminForgotOpen(true)} className="text-xs text-blue-600 hover:text-blue-700 font-medium">Forgot Password?</button>
   129→              </p>
   130→              {error && (
   131→                <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="bg-red-50 border border-red-200 text-red-700 text-sm rounded-xl p-3 flex items-center gap-2">
   132→                  <AlertTriangle className="w-4 h-4 shrink-0" /> {error}
   133→                </motion.div>
   134→              )}
   135→              <Button className="w-full bg-blue-700 hover:bg-blue-800 h-11 font-semibold" onClick={handleLogin} disabled={loading || !username || !password}>
   136→                {loading ? 'Logging in...' : 'Login to Admin Panel'}
   137→              </Button>
   138→            </div>
   139→          </CardContent>
   140→        </Card>
   141→
   142→        {/* Admin Forgot Password Dialog */}
   143→        <Dialog open={adminForgotOpen} onOpenChange={(open) => { if (!open) { setAdminForgotOpen(false); setForgotMsg(null); } }}>
   144→          <DialogContent className="max-w-sm">
   145→            <DialogHeader>
   146→              <DialogTitle className="flex items-center gap-2"><Shield className="w-5 h-5 text-blue-600" />Reset Admin Password</DialogTitle>
   147→              <DialogDescription>Set a new password for admin login</DialogDescription>
   148→            </DialogHeader>
   149→            <div className="space-y-3">
   150→              {forgotMsg && (
   151→                <motion.div initial={{ opacity: 0, y: -5 }} animate={{ opacity: 1, y: 0 }}
   152→                  className={`p-3 rounded-xl text-sm flex items-center gap-2 border ${
   153→                    forgotMsg.type === 'success' ? 'bg-green-50 border-green-200 text-green-700' : 'bg-red-50 border-red-200 text-red-700'
   154→                  }`}>
   155→                  {forgotMsg.type === 'success' ? <CheckCircle2 className="w-4 h-4 shrink-0" /> : <AlertTriangle className="w-4 h-4 shrink-0" />}
   156→                  {forgotMsg.text}
   157→                </motion.div>
   158→              )}
   159→              <div>
   160→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">New Password</Label>
   161→                <Input type="password" value={forgotNewPassword} onChange={(e) => setForgotNewPassword(e.target.value)} placeholder="Enter new password (min 4 chars)" className="h-11" />
   162→              </div>
   163→              <div>
   164→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Confirm New Password</Label>
   165→                <Input type="password" value={forgotConfirmPassword} onChange={(e) => setForgotConfirmPassword(e.target.value)} placeholder="Confirm new password" className="h-11" onKeyDown={(e) => e.key === 'Enter' && handleAdminForgotPassword()} />
   166→              </div>
   167→              <div className="flex gap-2">
   168→                <DialogClose asChild>
   169→                  <Button variant="outline" className="flex-1 h-11">Cancel</Button>
   170→                </DialogClose>
   171→                <Button className="flex-1 h-11 bg-blue-700 hover:bg-blue-800 font-semibold" onClick={handleAdminForgotPassword} disabled={forgotLoading || !forgotNewPassword || !forgotConfirmPassword}>
   172→                  {forgotLoading ? <><Loader2 className="w-4 h-4 mr-2 animate-spin" />Resetting...</> : <><Save className="w-4 h-4 mr-2" />Reset Password</>}
   173→                </Button>
   174→              </div>
   175→            </div>
   176→          </DialogContent>
   177→        </Dialog>
   178→      </motion.div>
   179→    </div>
   180→  );
   181→}

   413→// ==================== CATEGORIES TAB ====================
   414→function AdminCategoriesTab({ onRefresh }: { onRefresh: () => void }) {
   415→  const { categories, setCategories } = useAppStore();
   416→  const [newName, setNewName] = useState('');
   417→  const [newSlug, setNewSlug] = useState('');
   418→  const [newExamType, setNewExamType] = useState('General');
   419→  const [adding, setAdding] = useState(false);
   420→
   421→  async function handleAdd() {
   422→    if (!newName || !newSlug) return;
   423→    setAdding(true);
   424→    await fetch('/api/admin/categories', {
   425→      method: 'POST',
   426→      headers: { 'Content-Type': 'application/json' },
   427→      body: JSON.stringify({ name: newName, slug: newSlug, examType: newExamType }),
   428→    });
   429→    setNewName(''); setNewSlug('');
   430→    setAdding(false);
   431→    onRefresh();
   432→  }
   433→
   434→  async function handleDelete(id: string) {
   435→    if (!confirm('Delete this category and ALL its tests & questions? This cannot be undone.')) return;
   436→    await fetch('/api/admin/categories', {
   437→      method: 'DELETE',
   438→      headers: { 'Content-Type': 'application/json' },
   439→      body: JSON.stringify({ id }),
   440→    });
   441→    onRefresh();
   442→  }
   443→
   444→  return (
   445→    <div className="space-y-4">
   446→      {/* Add Category */}
   447→      <Card className="border-0 shadow-sm">
   448→        <CardContent className="p-5">
   449→          <h3 className="text-sm font-semibold mb-3 flex items-center gap-2"><Plus className="w-4 h-4 text-blue-600" /> Add New Category</h3>
   450→          <div className="grid grid-cols-1 sm:grid-cols-4 gap-3">
