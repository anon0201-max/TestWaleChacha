    23→// ==================== ADMIN LOGIN ====================
    24→function AdminLogin() {
    25→  const { setAdminData } = useAppStore();
    26→  const [username, setUsername] = useState('');
    27→  const [password, setPassword] = useState('');
    28→  const [error, setError] = useState('');
    29→  const [loading, setLoading] = useState(false);
    30→  const [showPassword, setShowPassword] = useState(false);
    31→  const [changePasswordOpen, setChangePasswordOpen] = useState(false);
    32→  const [cpCurrentPassword, setCpCurrentPassword] = useState('');
    33→  const [cpNewPassword, setCpNewPassword] = useState('');
    34→  const [cpConfirmPassword, setCpConfirmPassword] = useState('');
    35→  const [cpLoading, setCpLoading] = useState(false);
    36→  const [cpMsg, setCpMsg] = useState<{ type: 'success' | 'error'; text: string } | null>(null);
    37→
    38→  async function handleLogin() {
    39→    setLoading(true);
    40→    setError('');
    41→    try {
    42→      const res = await fetch('/api/admin/login', {
    43→        method: 'POST',
    44→        headers: { 'Content-Type': 'application/json' },
    45→        body: JSON.stringify({ username, password }),
    46→      });
    47→      const data = await res.json();
    48→      if (data.success) {
    49→        setAdminData({ isLoggedIn: true });
    50→      } else {
    51→        setError(data.error || 'Login failed');
    52→      }
    53→    } catch { setError('Server error'); }
    54→    setLoading(false);
    55→  }
    56→
    57→  async function handleChangePassword() {
    58→    setCpMsg(null);
    59→    if (!cpCurrentPassword || !cpNewPassword || !cpConfirmPassword) {
    60→      setCpMsg({ type: 'error', text: 'All fields are required' });
    61→      return;
    62→    }
    63→    if (cpNewPassword !== cpConfirmPassword) {
    64→      setCpMsg({ type: 'error', text: 'New password and confirm password do not match' });
    65→      return;
    66→    }
    67→    if (cpNewPassword.length < 4) {
    68→      setCpMsg({ type: 'error', text: 'New password must be at least 4 characters' });
    69→      return;
    70→    }
    71→    setCpLoading(true);
    72→    try {
    73→      const res = await fetch('/api/admin/reset-password', {
    74→        method: 'POST',
    75→        headers: { 'Content-Type': 'application/json' },
    76→        body: JSON.stringify({ currentPassword: cpCurrentPassword, newPassword: cpNewPassword }),
    77→      });
    78→      const data = await res.json();
    79→      if (data.success) {
    80→        setCpMsg({ type: 'success', text: 'Password updated successfully! Use new password to login next time.' });
    81→        setCpCurrentPassword('');
    82→        setCpNewPassword('');
    83→        setCpConfirmPassword('');
    84→      } else {
    85→        setCpMsg({ type: 'error', text: data.error || 'Failed to update password' });
    86→      }
    87→    } catch {
    88→      setCpMsg({ type: 'error', text: 'Server error. Please try again.' });
    89→    }
    90→    setCpLoading(false);
    91→  }
    92→
    93→  return (
    94→    <div className="min-h-[70vh] flex items-center justify-center px-4 pt-12">
    95→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="w-full">
    96→        <Card className="w-full max-w-md mx-auto shadow-xl border-0">
    97→          <CardContent className="p-5 sm:p-6">
    98→            <div className="text-center mb-6">
    99→              <div className="w-16 h-16 rounded-2xl bg-blue-700 flex items-center justify-center mx-auto mb-4">
   100→                <Shield className="w-8 h-8 text-white" />
   101→              </div>
   102→              <h2 className="text-xl font-bold text-gray-900">Admin Login</h2>
   103→              <p className="text-xs text-muted-foreground mt-1">Enter credentials to manage tests & questions</p>
   104→            </div>
   105→            <div className="space-y-4">
   106→              <div>
   107→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Username</Label>
   108→                <Input value={username} onChange={(e) => setUsername(e.target.value)} placeholder="admin" className="h-11" />
   109→              </div>
   110→              <div>
   111→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Password</Label>
   112→                <div className="relative">
   113→                  <Input
   114→                    type={showPassword ? 'text' : 'password'}
   115→                    value={password}
   116→                    onChange={(e) => setPassword(e.target.value)}
   117→                    placeholder="••••••••"
   118→                    className="h-11 pr-10"
   119→                    onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
   120→                  />
   121→                  <button
   122→                    type="button"
   123→                    onClick={() => setShowPassword(!showPassword)}
   124→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   125→                  >
   126→                    {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   127→                  </button>
   128→                </div>
   129→              </div>
   130→              <p className="text-right">
   131→                <button onClick={() => setChangePasswordOpen(true)} className="text-xs text-blue-600 hover:text-blue-700 font-medium">Change Password?</button>
   132→              </p>
   133→              {error && (
   134→                <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="bg-red-50 border border-red-200 text-red-700 text-sm rounded-xl p-3 flex items-center gap-2">
   135→                  <AlertTriangle className="w-4 h-4 shrink-0" /> {error}
   136→                </motion.div>
   137→              )}
   138→              <Button className="w-full bg-blue-700 hover:bg-blue-800 h-11 font-semibold" onClick={handleLogin} disabled={loading || !username || !password}>
   139→                {loading ? 'Logging in...' : 'Login to Admin Panel'}
   140→              </Button>
   141→            </div>
   142→          </CardContent>
   143→        </Card>
   144→
   145→        {/* Admin Change Password Dialog */}
   146→        <Dialog open={changePasswordOpen} onOpenChange={(open) => { if (!open) { setChangePasswordOpen(false); setCpMsg(null); setCpCurrentPassword(''); setCpNewPassword(''); setCpConfirmPassword(''); } }}>
   147→          <DialogContent className="max-w-[calc(100vw-2rem)] sm:max-w-sm">
   148→            <DialogHeader>
   149→              <DialogTitle className="flex items-center gap-2 text-base"><Shield className="w-5 h-5 text-blue-600" />Change Admin Password</DialogTitle>
   150→              <DialogDescription className="text-xs">Enter current password, then set new password</DialogDescription>
   151→            </DialogHeader>
   152→            <div className="space-y-3">
   153→              {cpMsg && (
   154→                <motion.div initial={{ opacity: 0, y: -5 }} animate={{ opacity: 1, y: 0 }}
   155→                  className={`p-3 rounded-xl text-sm flex items-center gap-2 border ${
   156→                    cpMsg.type === 'success' ? 'bg-green-50 border-green-200 text-green-700' : 'bg-red-50 border-red-200 text-red-700'
   157→                  }`}>
   158→                  {cpMsg.type === 'success' ? <CheckCircle2 className="w-4 h-4 shrink-0" /> : <AlertTriangle className="w-4 h-4 shrink-0" />}
   159→                  {cpMsg.text}
   160→                </motion.div>
   161→              )}
   162→              <div>
   163→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Current Password</Label>
   164→                <Input type="password" value={cpCurrentPassword} onChange={(e) => setCpCurrentPassword(e.target.value)} placeholder="Enter current password" className="h-11" />
   165→              </div>
   166→              <div>
   167→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">New Password</Label>
   168→                <Input type="password" value={cpNewPassword} onChange={(e) => setCpNewPassword(e.target.value)} placeholder="Enter new password (min 4 chars)" className="h-11" />
   169→              </div>
   170→              <div>
   171→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Confirm New Password</Label>
   172→                <Input type="password" value={cpConfirmPassword} onChange={(e) => setCpConfirmPassword(e.target.value)} placeholder="Confirm new password" className="h-11" onKeyDown={(e) => e.key === 'Enter' && handleChangePassword()} />
   173→              </div>
   174→              <div className="flex gap-2">
   175→                <DialogClose asChild>
   176→                  <Button variant="outline" className="flex-1 h-11">Cancel</Button>
   177→                </DialogClose>
   178→                <Button className="flex-1 h-11 bg-blue-700 hover:bg-blue-800 font-semibold" onClick={handleChangePassword} disabled={cpLoading || !cpCurrentPassword || !cpNewPassword || !cpConfirmPassword}>
   179→                  {cpLoading ? <><Loader2 className="w-4 h-4 mr-2 animate-spin" />Updating...</> : <><Save className="w-4 h-4 mr-2" />Update Password</>}
   180→                </Button>
   181→              </div>
   182→            </div>
   183→          </DialogContent>
   184→        </Dialog>
   185→      </motion.div>
   186→    </div>
   187→  );
   188→}

   420→// ==================== CATEGORIES TAB ====================
   421→function AdminCategoriesTab({ onRefresh }: { onRefresh: () => void }) {
   422→  const { categories, setCategories } = useAppStore();
   423→  const [newName, setNewName] = useState('');
   424→  const [newSlug, setNewSlug] = useState('');
   425→  const [newExamType, setNewExamType] = useState('General');
   426→  const [adding, setAdding] = useState(false);
   427→
   428→  async function handleAdd() {
   429→    if (!newName || !newSlug) return;
   430→    setAdding(true);
   431→    await fetch('/api/admin/categories', {
   432→      method: 'POST',
   433→      headers: { 'Content-Type': 'application/json' },

   950→  function fillQuestionsFromExtract(extracted: QuestionForm[]): number {
   951→    if (!extracted || extracted.length === 0) {
   952→      setFillStatus({ msg: '⚠️ No questions received from VLM to fill.', type: 'error' });
   953→      return 0;
   954→    }
   955→    const count = extracted.length;
   956→    setFillStatus({ msg: `🔧 Filling ${count} questions into boxes...`, type: 'info' });
   957→
   958→    // Use ref to track where to start filling (supports multi-image upload)
   959→    let startIdx = fillStartIndexRef.current;
   960→
   961→    setQuestions(prev => {
   962→      const next = [...prev];
   963→      // Clamp startIdx if it's beyond array length
   964→      if (startIdx > next.length) startIdx = next.length;
   965→      // Fill each extracted question into the boxes
   966→      for (let k = 0; k < extracted.length; k++) {
   967→        const targetIdx = startIdx + k;
   968→        if (targetIdx < next.length) {
   969→          // Replace entire box with extracted data (not merge — merge keeps empty defaults)
   970→          next[targetIdx] = { ...extracted[k] };
   971→        } else {
   972→          next.push(extracted[k]);
   973→        }
   974→      }
   975→      return next;
   976→    });
   977→
   978→    // Advance the fill index for next image batch
   979→    fillStartIndexRef.current = startIdx + count;
   980→
   981→    setTimeout(() => {
   982→      setFillStatus({ msg: `✅ ${count} questions filled into boxes!`, type: 'success' });
   983→      setTimeout(() => setFillStatus(null), 8000);
   984→    }, 100);
   985→    return count;
   986→  }

   988→  async function handleImageExtract(e: React.ChangeEvent<HTMLInputElement>) {
   989→    const files = e.target.files;
   990→    if (!files || files.length === 0) return;
   991→    const fileList = Array.from(files);
   992→    setExtractingImage(true);
   993→    setExtractElapsed(0);
   994→    const startTime = Date.now();
   995→    const timerInterval = setInterval(() => {
   996→      setExtractElapsed(Math.floor((Date.now() - startTime) / 1000));
   997→    }, 1000);
   998→    try {
   999→      let totalExtracted = 0;
  1000→      let hasError = false;
  1001→      for (let i = 0; i < fileList.length; i++) {
  1002→        const file = fileList[i];
  1003→        const formData = new FormData();
  1004→        formData.append('image', file);
  1005→        const controller = new AbortController();
  1006→        const timeoutId = setTimeout(() => controller.abort(), 300000);
  1007→        try {
  1008→          setFillStatus({ msg: `📤 Image ${i + 1}/${fileList.length}: Scanning with AI... (1-3 min)`, type: 'info' });
  1009→          // Use Next.js API route with z-ai-web-dev-sdk VLM + Grok fallback
  1010→          const res = await fetch('/api/admin/extract-question', { method: 'POST', body: formData, signal: controller.signal });
  1011→          clearTimeout(timeoutId);
  1012→          if (!res.ok) {
  1013→            const errText = await res.text().catch(() => 'Unknown error');
  1014→            setFillStatus({ msg: `❌ Image ${i + 1}: Server error (HTTP ${res.status}). ${errText.substring(0, 80)}`, type: 'error' });
  1015→            toast.error(`Image ${i + 1}: Server error ${res.status}`);
  1016→            hasError = true;
  1017→            continue;
  1018→          }
  1019→          const data = await res.json();
  1020→          if (data.success && data.questions && data.questions.length > 0) {
  1021→            const newQuestions: QuestionForm[] = data.questions.map((q: any) => ({
  1022→              question: q.question || '', optionA: q.optionA || '', optionB: q.optionB || '', optionC: q.optionC || '', optionD: q.optionD || '',
  1023→              correctOption: q.correctOption || 'A', explanation: q.explanation || '', section: q.section || 'General', negativeMark: q.negativeMark || '0',
  1024→            }));
  1025→            const filled = fillQuestionsFromExtract(newQuestions);
  1026→            totalExtracted += filled;
  1027→            toast.success(`Image ${i + 1}: ${filled} questions extracted & filled!`);
  1028→          } else if (data.success && data.question) {
  1029→            const newQ: QuestionForm = {
  1030→              question: data.question.question || '', optionA: data.question.optionA || '', optionB: data.question.optionB || '', optionC: data.question.optionC || '', optionD: data.question.optionD || '',
  1031→              correctOption: data.question.correctOption || 'A', explanation: data.question.explanation || '', section: data.question.section || 'General', negativeMark: data.question.negativeMark || '0',
  1032→            };
  1033→            const filled = fillQuestionsFromExtract([newQ]);
  1034→            totalExtracted += filled;
  1035→            toast.success(`Image ${i + 1}: 1 question extracted & filled!`);
  1036→          } else {
  1037→            hasError = true;
  1038→            const errMsg = data.error || 'Failed to extract questions';
  1039→            setFillStatus({ msg: `❌ ${errMsg}`, type: 'error' });
  1040→            toast.error(`Image ${i + 1}: ${errMsg}`);
  1041→          }
  1042→        } catch (err: any) {
  1043→          clearTimeout(timeoutId);
  1044→          hasError = true;
  1045→          if (err?.name === 'AbortError') {
  1046→            setFillStatus({ msg: '❌ Timed out (5 min). Try a smaller/clearer image.', type: 'error' });
  1047→          } else {
  1048→            setFillStatus({ msg: `❌ ${err?.message || 'Network error. Check your connection.'}`, type: 'error' });
  1049→          }
  1050→        }
  1051→      }
  1052→      if (totalExtracted > 0) {
  1053→        setFillStatus({ msg: `✅ Done! Total ${totalExtracted} questions extracted & filled.`, type: 'success' });
  1054→        toast.success(`🎉 Total: ${totalExtracted} questions extracted!`);
  1055→      } else if (hasError) {
  1056→        setFillStatus({ msg: `❌ No questions extracted. Check error messages above.`, type: 'error' });
  1057→      } else {
  1058→        setFillStatus({ msg: '⚠️ No questions could be extracted.', type: 'error' });
  1059→        toast.error('No questions could be extracted. Empty boxes remain for manual entry.');
  1060→      }
  1061→    } finally {
  1062→      clearInterval(timerInterval);
  1063→      setExtractingImage(false);
  1064→      setExtractElapsed(0);
  1065→      e.target.value = '';
  1066→    }
  1067→  }

  1673→function AdminSettingsTab() {
  1674→  const [currentPassword, setCurrentPassword] = useState('');
  1675→  const [newPassword, setNewPassword] = useState('');
  1676→  const [confirmPassword, setConfirmPassword] = useState('');
  1677→  const [loading, setLoading] = useState(false);
  1678→  const [message, setMessage] = useState<{ type: 'success' | 'error'; text: string } | null>(null);
  1679→
  1680→  async function handleResetPassword() {
  1681→    setMessage(null);
  1682→    if (!currentPassword || !newPassword || !confirmPassword) {
  1683→      setMessage({ type: 'error', text: 'All fields are required' });
  1684→      return;
  1685→    }
  1686→    if (newPassword !== confirmPassword) {
  1687→      setMessage({ type: 'error', text: 'Passwords do not match' });
  1688→      return;
  1689→    }
  1690→    if (newPassword.length < 4) {
  1691→      setMessage({ type: 'error', text: 'New password must be at least 4 characters' });
  1692→      return;
  1693→    }
  1694→    setLoading(true);
  1695→    try {
  1696→      const res = await fetch('/api/admin/reset-password', {
  1697→        method: 'POST',
  1698→        headers: { 'Content-Type': 'application/json' },
  1699→        body: JSON.stringify({ currentPassword, newPassword }),
  1700→      });
  1701→      const data = await res.json();
  1702→      if (data.success) {
  1703→        setMessage({ type: 'success', text: 'Password updated successfully!' });
  1704→        setCurrentPassword('');
  1705→        setNewPassword('');
  1706→        setConfirmPassword('');
  1707→      } else {
  1708→        setMessage({ type: 'error', text: data.error || 'Failed to reset password' });
  1709→      }
  1710→    } catch {
  1711→      setMessage({ type: 'error', text: 'Server error.' });
  1712→    }
  1713→    setLoading(false);
  1714→  }
  1715→
  1716→  return (
  1717→    <div className="space-y-3 sm:space-y-4">
  1718→      <Card className="border-0 shadow-sm">
  1719→        <CardContent className="p-4 sm:p-5">
