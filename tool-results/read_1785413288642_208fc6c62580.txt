    24→    24→function formatTime(seconds: number): string {
    25→    25→  if (!seconds || !isFinite(seconds) || seconds < 0) return '0:00';
    26→    26→  const hrs = Math.floor(seconds / 3600);
    27→    27→  const mins = Math.floor((seconds % 3600) / 60);
    28→    28→  const secs = seconds % 60;
    29→    29→  if (hrs > 0) return `${hrs}:${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
    30→    30→  return `${mins}:${secs.toString().padStart(2, '0')}`;
    31→    31→}
