    23→// ==================== ADMIN LOGIN ====================
    24→function AdminLogin() {
    25→  const { setAdminData } = useAppStore();
    26→  const [username, setUsername] = useState('');
    27→  const [password, setPassword] = useState('');
    28→  const [error, setError] = useState('');
    29→  const [loading, setLoading] = useState(false);
    30→  const [showPassword, setShowPassword] = useState(false);
    31→  const [changePasswordOpen, setChangePasswordOpen] = useState(false);
    32→  const [cpCurrentPassword, setCpCurrentPassword] = useState('');
    33→  const [cpNewPassword, setCpNewPassword] = useState('');
    34→  const [cpConfirmPassword, setCpConfirmPassword] = useState('');
    35→  const [cpLoading, setCpLoading] = useState(false);
    36→  const [cpMsg, setCpMsg] = useState<{ type: 'success' | 'error'; text: string } | null>(null);
    37→
    38→  async function handleLogin() {
    39→    setLoading(true);
    40→    setError('');
    41→    try {
    42→      const res = await fetch('/api/admin/login', {
    43→        method: 'POST',
    44→        headers: { 'Content-Type': 'application/json' },
    45→        body: JSON.stringify({ username, password }),
    46→      });
    47→      const data = await res.json();
    48→      if (data.success) {
    49→        setAdminData({ isLoggedIn: true });
    50→      } else {
    51→        setError(data.error || 'Login failed');
    52→      }
    53→    } catch { setError('Server error'); }
    54→    setLoading(false);
    55→  }
    56→
    57→  async function handleChangePassword() {
    58→    setCpMsg(null);
    59→    if (!cpCurrentPassword || !cpNewPassword || !cpConfirmPassword) {
    60→      setCpMsg({ type: 'error', text: 'All fields are required' });
    61→      return;
    62→    }
    63→    if (cpNewPassword !== cpConfirmPassword) {
    64→      setCpMsg({ type: 'error', text: 'New password and confirm password do not match' });
    65→      return;
    66→    }
    67→    if (cpNewPassword.length < 4) {
    68→      setCpMsg({ type: 'error', text: 'New password must be at least 4 characters' });
    69→      return;
    70→    }
    71→    setCpLoading(true);
    72→    try {
    73→      const res = await fetch('/api/admin/reset-password', {
    74→        method: 'POST',
    75→        headers: { 'Content-Type': 'application/json' },
    76→        body: JSON.stringify({ currentPassword: cpCurrentPassword, newPassword: cpNewPassword }),
    77→      });
    78→      const data = await res.json();
    79→      if (data.success) {
    80→        setCpMsg({ type: 'success', text: 'Password updated successfully! Use new password to login next time.' });
    81→        setCpCurrentPassword('');
    82→        setCpNewPassword('');
    83→        setCpConfirmPassword('');
    84→      } else {
    85→        setCpMsg({ type: 'error', text: data.error || 'Failed to update password' });
    86→      }
    87→    } catch {
    88→      setCpMsg({ type: 'error', text: 'Server error. Please try again.' });
    89→    }
    90→    setCpLoading(false);
    91→  }
    92→
    93→  return (
    94→    <div className="min-h-[70vh] flex items-center justify-center px-4 pt-12">
    95→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="w-full">
    96→        <Card className="w-full max-w-md mx-auto shadow-xl border-0">
    97→          <CardContent className="p-5 sm:p-6">
    98→            <div className="text-center mb-6">
    99→              <div className="w-16 h-16 rounded-2xl bg-blue-700 flex items-center justify-center mx-auto mb-4">
   100→                <Shield className="w-8 h-8 text-white" />
   101→              </div>
   102→              <h2 className="text-xl font-bold text-gray-900">Admin Login</h2>
   103→              <p className="text-xs text-muted-foreground mt-1">Enter credentials to manage tests & questions</p>
   104→            </div>
   105→            <div className="space-y-4">
   106→              <div>
   107→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Username</Label>
   108→                <Input value={username} onChange={(e) => setUsername(e.target.value)} placeholder="admin" className="h-11" />
   109→              </div>
   110→              <div>
   111→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Password</Label>
   112→                <div className="relative">
   113→                  <Input
   114→                    type={showPassword ? 'text' : 'password'}
   115→                    value={password}
   116→                    onChange={(e) => setPassword(e.target.value)}
   117→                    placeholder="••••••••"
   118→                    className="h-11 pr-10"
   119→                    onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
   120→                  />
   121→                  <button
   122→                    type="button"
   123→                    onClick={() => setShowPassword(!showPassword)}
   124→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   125→                  >
   126→                    {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   127→                  </button>
   128→                </div>
   129→              </div>
   130→              <p className="text-right">
   131→                <button onClick={() => setChangePasswordOpen(true)} className="text-xs text-blue-600 hover:text-blue-700 font-medium">Change Password?</button>
   132→              </p>
   133→              {error && (
   134→                <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="bg-red-50 border border-red-200 text-red-700 text-sm rounded-xl p-3 flex items-center gap-2">
   135→                  <AlertTriangle className="w-4 h-4 shrink-0" /> {error}
   136→                </motion.div>
   137→              )}
   138→              <Button className="w-full bg-blue-700 hover:bg-blue-800 h-11 font-semibold" onClick={handleLogin} disabled={loading || !username || !password}>
   139→                {loading ? 'Logging in...' : 'Login to Admin Panel'}
   140→              </Button>
   141→            </div>
   142→          </CardContent>
   143→        </Card>
   144→
   145→        {/* Admin Change Password Dialog */}
   146→        <Dialog open={changePasswordOpen} onOpenChange={(open) => { if (!open) { setChangePasswordOpen(false); setCpMsg(null); setCpCurrentPassword(''); setCpNewPassword(''); setCpConfirmPassword(''); } }}>
   147→          <DialogContent className="max-w-[calc(100vw-2rem)] sm:max-w-sm">
   148→            <DialogHeader>
   149→              <DialogTitle className="flex items-center gap-2 text-base"><Shield className="w-5 h-5 text-blue-600" />Change Admin Password</DialogTitle>
   150→              <DialogDescription className="text-xs">Enter current password, then set new password</DialogDescription>
   151→            </DialogHeader>
   152→            <div className="space-y-3">
   153→              {cpMsg && (
   154→                <motion.div initial={{ opacity: 0, y: -5 }} animate={{ opacity: 1, y: 0 }}
   155→                  className={`p-3 rounded-xl text-sm flex items-center gap-2 border ${
   156→                    cpMsg.type === 'success' ? 'bg-green-50 border-green-200 text-green-700' : 'bg-red-50 border-red-200 text-red-700'
   157→                  }`}>
   158→                  {cpMsg.type === 'success' ? <CheckCircle2 className="w-4 h-4 shrink-0" /> : <AlertTriangle className="w-4 h-4 shrink-0" />}
   159→                  {cpMsg.text}
   160→                </motion.div>
   161→              )}
   162→              <div>
   163→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Current Password</Label>
   164→                <Input type="password" value={cpCurrentPassword} onChange={(e) => setCpCurrentPassword(e.target.value)} placeholder="Enter current password" className="h-11" />
   165→              </div>
   166→              <div>
   167→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">New Password</Label>
   168→                <Input type="password" value={cpNewPassword} onChange={(e) => setCpNewPassword(e.target.value)} placeholder="Enter new password (min 4 chars)" className="h-11" />
   169→              </div>
   170→              <div>
   171→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Confirm New Password</Label>
   172→                <Input type="password" value={cpConfirmPassword} onChange={(e) => setCpConfirmPassword(e.target.value)} placeholder="Confirm new password" className="h-11" onKeyDown={(e) => e.key === 'Enter' && handleChangePassword()} />
   173→              </div>
   174→              <div className="flex gap-2">
   175→                <DialogClose asChild>
   176→                  <Button variant="outline" className="flex-1 h-11">Cancel</Button>
   177→                </DialogClose>
   178→                <Button className="flex-1 h-11 bg-blue-700 hover:bg-blue-800 font-semibold" onClick={handleChangePassword} disabled={cpLoading || !cpCurrentPassword || !cpNewPassword || !cpConfirmPassword}>
   179→                  {cpLoading ? <><Loader2 className="w-4 h-4 mr-2 animate-spin" />Updating...</> : <><Save className="w-4 h-4 mr-2" />Update Password</>}
   180→                </Button>
   181→              </div>
   182→            </div>
   183→          </DialogContent>
   184→        </Dialog>
   185→      </motion.div>
   186→    </div>
   187→  );
   188→}

   420→// ==================== CATEGORIES TAB ====================
   421→function AdminCategoriesTab({ onRefresh }: { onRefresh: () => void }) {
   422→  const { categories, setCategories } = useAppStore();
   423→  const [newName, setNewName] = useState('');
   424→  const [newSlug, setNewSlug] = useState('');
   425→  const [newExamType, setNewExamType] = useState('General');
   426→  const [adding, setAdding] = useState(false);
   427→
   428→  async function handleAdd() {
   429→    if (!newName || !newSlug) return;
   430→    setAdding(true);
   431→    await fetch('/api/admin/categories', {
   432→      method: 'POST',
   433→      headers: { 'Content-Type': 'application/json' },

   964→  function fillQuestionsFromExtract(extracted: QuestionForm[]): number {
   965→    if (!extracted || extracted.length === 0) {
   966→      setFillStatus({ msg: '⚠️ No questions received from VLM to fill.', type: 'error' });
   967→      return 0;
   968→    }
   969→    const count = extracted.length;
   970→    setFillStatus({ msg: `🔧 Filling ${count} questions into boxes...`, type: 'info' });
   971→
   972→    // Use ref to track where to start filling (supports multi-image upload)
   973→    let startIdx = fillStartIndexRef.current;
   974→
   975→    setQuestions(prev => {
   976→      const next = [...prev];
   977→      // Clamp startIdx if it's beyond array length
   978→      if (startIdx > next.length) startIdx = next.length;
   979→      // Fill each extracted question into the boxes
   980→      for (let k = 0; k < extracted.length; k++) {
   981→        const targetIdx = startIdx + k;
   982→        if (targetIdx < next.length) {
   983→          // Replace entire box with extracted data (not merge — merge keeps empty defaults)
   984→          next[targetIdx] = { ...extracted[k] };
   985→        } else {
   986→          next.push(extracted[k]);
   987→        }
   988→      }
   989→      return next;
   990→    });
   991→
   992→    // Advance the fill index for next image batch
   993→    fillStartIndexRef.current = startIdx + count;
   994→
   995→    setTimeout(() => {
   996→      setFillStatus({ msg: `✅ ${count} questions filled into boxes!`, type: 'success' });
   997→      setTimeout(() => setFillStatus(null), 8000);
   998→    }, 100);
   999→    return count;
  1000→  }

  1002→  async function handleImageExtract(e: React.ChangeEvent<HTMLInputElement>) {
  1003→    const files = e.target.files;
  1004→    if (!files || files.length === 0) return;
  1005→    const fileList = Array.from(files);
  1006→    setExtractingImage(true);
  1007→    setExtractElapsed(0);
  1008→    const startTime = Date.now();
  1009→    const timerInterval = setInterval(() => {
  1010→      setExtractElapsed(Math.floor((Date.now() - startTime) / 1000));
  1011→    }, 1000);
  1012→    try {
  1013→      let totalExtracted = 0;
  1014→      let hasError = false;
  1015→      for (let i = 0; i < fileList.length; i++) {
  1016→        const file = fileList[i];
  1017→        const formData = new FormData();
  1018→        formData.append('image', file);
  1019→        const controller = new AbortController();
  1020→        const timeoutId = setTimeout(() => controller.abort(), 300000);
  1021→        try {
  1022→          setFillStatus({ msg: `📤 Image ${i + 1}/${fileList.length}: Scanning with AI... (1-3 min)`, type: 'info' });
  1023→          // Use Next.js API route with z-ai-web-dev-sdk VLM + Grok fallback
  1024→          const res = await fetch('/api/admin/extract-question', { method: 'POST', body: formData, signal: controller.signal });
  1025→          clearTimeout(timeoutId);
  1026→          if (!res.ok) {
  1027→            const errText = await res.text().catch(() => 'Unknown error');
  1028→            setFillStatus({ msg: `❌ Image ${i + 1}: Server error (HTTP ${res.status}). ${errText.substring(0, 80)}`, type: 'error' });
  1029→            toast.error(`Image ${i + 1}: Server error ${res.status}`);
  1030→            hasError = true;
  1031→            continue;
  1032→          }
  1033→          const data = await res.json();
  1034→          if (data.success && data.questions && data.questions.length > 0) {
  1035→            const newQuestions: QuestionForm[] = data.questions.map((q: any) => ({
  1036→              question: q.question || '', optionA: q.optionA || '', optionB: q.optionB || '', optionC: q.optionC || '', optionD: q.optionD || '',
  1037→              correctOption: q.correctOption || 'A', explanation: q.explanation || '', section: q.section || 'General', negativeMark: q.negativeMark || '0',
  1038→            }));
  1039→            const filled = fillQuestionsFromExtract(newQuestions);
  1040→            totalExtracted += filled;
  1041→            toast.success(`Image ${i + 1}: ${filled} questions extracted & filled!`);
  1042→          } else if (data.success && data.question) {
  1043→            const newQ: QuestionForm = {
  1044→              question: data.question.question || '', optionA: data.question.optionA || '', optionB: data.question.optionB || '', optionC: data.question.optionC || '', optionD: data.question.optionD || '',
  1045→              correctOption: data.question.correctOption || 'A', explanation: data.question.explanation || '', section: data.question.section || 'General', negativeMark: data.question.negativeMark || '0',
  1046→            };
  1047→            const filled = fillQuestionsFromExtract([newQ]);
  1048→            totalExtracted += filled;
  1049→            toast.success(`Image ${i + 1}: 1 question extracted & filled!`);
  1050→          } else {
  1051→            hasError = true;
  1052→            const errMsg = data.error || 'Failed to extract questions';
  1053→            setFillStatus({ msg: `❌ ${errMsg}`, type: 'error' });
  1054→            toast.error(`Image ${i + 1}: ${errMsg}`);
  1055→          }
  1056→        } catch (err: any) {
  1057→          clearTimeout(timeoutId);
  1058→          hasError = true;
  1059→          if (err?.name === 'AbortError') {
  1060→            setFillStatus({ msg: '❌ Timed out (5 min). Try a smaller/clearer image.', type: 'error' });
  1061→          } else {
  1062→            setFillStatus({ msg: `❌ ${err?.message || 'Network error. Check your connection.'}`, type: 'error' });
  1063→          }
  1064→        }
  1065→      }
  1066→      if (totalExtracted > 0) {
  1067→        setFillStatus({ msg: `✅ Done! Total ${totalExtracted} questions extracted & filled.`, type: 'success' });
  1068→        toast.success(`🎉 Total: ${totalExtracted} questions extracted!`);
  1069→      } else if (hasError) {
  1070→        setFillStatus({ msg: `❌ No questions extracted. Check error messages above.`, type: 'error' });
  1071→      } else {
  1072→        setFillStatus({ msg: '⚠️ No questions could be extracted.', type: 'error' });
  1073→        toast.error('No questions could be extracted. Empty boxes remain for manual entry.');
  1074→      }
  1075→    } finally {
  1076→      clearInterval(timerInterval);
  1077→      setExtractingImage(false);
  1078→      setExtractElapsed(0);
  1079→      e.target.value = '';
  1080→    }
  1081→  }

  1687→function AdminSettingsTab() {
  1688→  const [currentPassword, setCurrentPassword] = useState('');
  1689→  const [newPassword, setNewPassword] = useState('');
  1690→  const [confirmPassword, setConfirmPassword] = useState('');
  1691→  const [loading, setLoading] = useState(false);
  1692→  const [message, setMessage] = useState<{ type: 'success' | 'error'; text: string } | null>(null);
  1693→
  1694→  async function handleResetPassword() {
  1695→    setMessage(null);
  1696→    if (!currentPassword || !newPassword || !confirmPassword) {
  1697→      setMessage({ type: 'error', text: 'All fields are required' });
  1698→      return;
  1699→    }
  1700→    if (newPassword !== confirmPassword) {
  1701→      setMessage({ type: 'error', text: 'Passwords do not match' });
  1702→      return;
  1703→    }
  1704→    if (newPassword.length < 4) {
  1705→      setMessage({ type: 'error', text: 'New password must be at least 4 characters' });
  1706→      return;
  1707→    }
  1708→    setLoading(true);
  1709→    try {
  1710→      const res = await fetch('/api/admin/reset-password', {
  1711→        method: 'POST',
  1712→        headers: { 'Content-Type': 'application/json' },
  1713→        body: JSON.stringify({ currentPassword, newPassword }),
  1714→      });
  1715→      const data = await res.json();
