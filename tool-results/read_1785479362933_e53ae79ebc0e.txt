    89→// Scroll-triggered section wrapper — items animate in as they enter viewport
    90→function ScrollSection({ children, className = '' }: { children: React.ReactNode; className?: string }) {
    91→  const ref = useRef(null);
    92→  const show = useInView(ref, { once: true, margin: '-60px' });
    93→  return (
    94→    <motion.section ref={ref} initial="hidden" animate={show ? 'visible' : 'hidden'} variants={stagger} className={className}>
    95→      {children}
    96→    </motion.section>
    97→  );
