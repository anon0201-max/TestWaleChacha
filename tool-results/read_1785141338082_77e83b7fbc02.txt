    21→// ==================== ADMIN LOGIN ====================
    22→function AdminLogin() {
    23→  const { setAdminData } = useAppStore();
    24→  const [username, setUsername] = useState('');
    25→  const [password, setPassword] = useState('');
    26→  const [error, setError] = useState('');
    27→  const [loading, setLoading] = useState(false);
    28→  const [showPassword, setShowPassword] = useState(false);
    29→
    30→  async function handleLogin() {
    31→    setLoading(true);
    32→    setError('');
    33→    try {
    34→      const res = await fetch('/api/admin/login', {
    35→        method: 'POST',
    36→        headers: { 'Content-Type': 'application/json' },
    37→        body: JSON.stringify({ username, password }),
    38→      });
    39→      const data = await res.json();
    40→      if (data.success) {
    41→        setAdminData({ isLoggedIn: true });
    42→      } else {
    43→        setError(data.error || 'Login failed');
    44→      }
    45→    } catch { setError('Server error'); }
    46→    setLoading(false);
    47→  }
    48→
    49→  return (
    50→    <div className="min-h-[70vh] flex items-center justify-center">
    51→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
    52→        <Card className="w-full max-w-sm shadow-xl border-0">
    53→          <CardContent className="p-6">
    54→            <div className="text-center mb-6">
    55→              <div className="w-16 h-16 rounded-2xl bg-blue-700 flex items-center justify-center mx-auto mb-4">
    56→                <Shield className="w-8 h-8 text-white" />
    57→              </div>
    58→              <h2 className="text-xl font-bold text-gray-900">Admin Login</h2>
    59→              <p className="text-xs text-muted-foreground mt-1">Enter credentials to manage tests & questions</p>
    60→            </div>
    61→            <div className="space-y-4">
    62→              <div>
    63→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Username</Label>
    64→                <Input value={username} onChange={(e) => setUsername(e.target.value)} placeholder="admin" className="h-11" />
    65→              </div>
    66→              <div>
    67→                <Label className="text-xs font-medium text-muted-foreground mb-1.5 block">Password</Label>
    68→                <div className="relative">
    69→                  <Input
    70→                    type={showPassword ? 'text' : 'password'}
    71→                    value={password}
    72→                    onChange={(e) => setPassword(e.target.value)}
    73→                    placeholder="••••••••"
    74→                    className="h-11 pr-10"
    75→                    onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
    76→                  />
    77→                  <button
    78→                    type="button"
    79→                    onClick={() => setShowPassword(!showPassword)}
    80→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
    81→                  >
    82→                    {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
    83→                  </button>
    84→                </div>
    85→              </div>
    86→              {error && (
    87→                <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="bg-red-50 border border-red-200 text-red-700 text-sm rounded-xl p-3 flex items-center gap-2">
    88→                  <AlertTriangle className="w-4 h-4 shrink-0" /> {error}
    89→                </motion.div>
    90→              )}
    91→              <Button className="w-full bg-blue-700 hover:bg-blue-800 h-11 font-semibold" onClick={handleLogin} disabled={loading || !username || !password}>
    92→                {loading ? 'Logging in...' : 'Login to Admin Panel'}
    93→              </Button>
    94→            </div>
    95→          </CardContent>
    96→        </Card>
    97→      </motion.div>
    98→    </div>
    99→  );
   100→}

   312→// ==================== CATEGORIES TAB ====================

   315→  const [newName, setNewName] = useState('');
   316→  const [newSlug, setNewSlug] = useState('');
   317→  const [newExamType, setNewExamType] = useState('General');
   318→  const [adding, setAdding] = useState(false);
   319→
   320→  async function handleAdd() {
   321→    if (!newName || !newSlug) return;
   322→    setAdding(true);
   323→    await fetch('/api/admin/categories', {
   324→      method: 'POST',
   325→      headers: { 'Content-Type': 'application/json' },
   326→      body: JSON.stringify({ name: newName, slug: newSlug, examType: newExamType }),
   327→    });
   328→    setNewName(''); setNewSlug('');
   329→    setAdding(false);
   330→    onRefresh();
   331→  }
   332→

   336→      method: 'DELETE',
   337→      headers: { 'Content-Type': 'application/json' },
   338→      body: JSON.stringify({ id }),
   339→    });
   340→    onRefresh();
   341→  }
   342→
   343→  return (
   344→    <div className="space-y-4">
   345→      {/* Add Category */}
   346→      <Card className="border-0 shadow-sm">

   412→// ==================== TESTS TAB ====================
   413→function AdminTestsTab({ onRefresh }: { onRefresh: () => void }) {
   414→  const { tests } = useAppStore();
   415→  const [expandedTestId, setExpandedTestId] = useState<string | null>(null);
   416→  const [testQuestions, setTestQuestions] = useState<Record<string, unknown[]>>({});
   417→  const [loadingQ, setLoadingQ] = useState<string | null>(null);
   418→
   419→  async function fetchQuestions(testId: string) {
   420→    if (testQuestions[testId]) {
   421→      setExpandedTestId(expandedTestId === testId ? null : testId);
   422→      return;
   423→    }
   424→    setLoadingQ(testId);
   425→    try {
   426→      const res = await fetch(`/api/tests/${testId}`);
   427→      if (res.ok) {
   428→        const data = await res.json();
   429→        setTestQuestions(prev => ({ ...prev, [testId]: data.questions || [] }));
   430→        setExpandedTestId(testId);
   431→      }
   432→    } catch {}
   433→    setLoadingQ(null);
   434→  }
   435→
   436→  async function handleDeleteTest(id: string) {
   437→    if (!confirm('Delete this test and ALL its questions? This cannot be undone.')) return;
   438→    await fetch('/api/admin/tests', {
   439→      method: 'DELETE',
   440→      headers: { 'Content-Type': 'application/json' },
   441→      body: JSON.stringify({ id }),
   442→    });
   443→    onRefresh();
   444→  }
   445→
   446→  async function handleDeleteQuestion(questionId: string, testId: string) {
   447→    if (!confirm('Delete this question?')) return;
   448→    await fetch('/api/admin/tests/questions', {
   449→      method: 'DELETE',
   450→      headers: { 'Content-Type': 'application/json' },
   451→      body: JSON.stringify({ questionId }),
   452→    });
   453→    // Refetch
   454→    const res = await fetch(`/api/tests/${testId}`);
   455→    if (res.ok) {
   456→      const data = await res.json();
   457→      setTestQuestions(prev => ({ ...prev, [testId]: data.questions || [] }));
   458→    }
   459→    onRefresh();
   460→  }
   461→
   462→  return (
   463→    <div className="space-y-3">
   464→      <div className="flex items-center justify-between">
   465→        <h3 className="text-sm font-semibold">All Tests ({tests.length})</h3>
   466→        <Badge variant="secondary" className="text-[10px]">{tests.reduce((s, t) => s + t.totalQuestions, 0)} total questions</Badge>
   467→      </div>
   468→
   469→      {tests.length === 0 ? (
   470→        <Card className="border-0 shadow-sm">
   471→          <CardContent className="p-10 text-center">
   472→            <FileText className="w-10 h-10 text-muted-foreground mx-auto mb-3" />
   473→            <p className="text-sm text-muted-foreground">No tests yet. Go to &quot;Create Test&quot; tab to add your first test.</p>
   474→          </CardContent>
   475→        </Card>
   476→      ) : (
   477→        tests.map((test) => (

   699→  async function handleImageExtract(e: React.ChangeEvent<HTMLInputElement>) {

   703→    const fileList = Array.from(files);

   707→    // Start a timer to show elapsed seconds

   709→    const timerInterval = setInterval(() => {
   710→      setExtractElapsed(Math.floor((Date.now() - startTime) / 1000));
   711→    }, 1000);
   712→
   713→    try {
   714→      let totalExtracted = 0;
   715→      let hasError = false;
   716→
   717→      // Process each image file sequentially
   718→      for (let i = 0; i < fileList.length; i++) {
   719→        const file = fileList[i];
   720→        const formData = new FormData();
   721→        formData.append('image', file);
   722→
   723→        // Use AbortController with 5-minute timeout per file (VLM can take 2-3 min for large images)
   724→        const controller = new AbortController();
   725→        const timeoutId = setTimeout(() => controller.abort(), 300000);
   726→
   727→        try {
   728→          console.log(`📤 Uploading image ${i + 1}/${fileList.length}: ${file.name}`);
   729→          const res = await fetch('/api/admin/extract-question', {
   730→            method: 'POST',
   731→            body: formData,
   732→            signal: controller.signal,
   733→          });
   734→          clearTimeout(timeoutId);
   735→
   736→          console.log(`📡 Response status: ${res.status}`);
   737→          const data = await res.json();
   738→          console.log(`📋 Extraction result:`, { success: data.success, count: data.count, hasQuestions: !!data.questions, error: data.error });
   739→
   740→          if (data.success && data.questions && data.questions.length > 0) {
   741→            const newQuestions: QuestionForm[] = data.questions.map((q: any) => ({
   742→              question: q.question || '',
   743→              optionA: q.optionA || '',
   744→              optionB: q.optionB || '',
   745→              optionC: q.optionC || '',
   746→              optionD: q.optionD || '',
   747→              correctOption: q.correctOption || 'A',
   748→              explanation: q.explanation || '',
   749→              section: q.section || 'General',
   750→              negativeMark: q.negativeMark || '0',
   751→            }));
   752→
   753→            // CRITICAL FIX: Use functional state update to avoid stale closure
   754→            // After 2-3 minutes of waiting, the `questions` variable in closure is stale
   755→            setQuestions(prev => [...prev, ...newQuestions]);
   756→            totalExtracted += newQuestions.length;
   757→            console.log(`✅ Added ${newQuestions.length} questions to form (total so far: ${totalExtracted})`);
   758→            toast.success(`Image ${i + 1}: ${newQuestions.length} questions extracted!`);
   759→          } else if (data.success && data.question) {
   760→            // Fallback: single question
   761→            const newQ: QuestionForm = {
   762→              question: data.question.question || '',
   763→              optionA: data.question.optionA || '',
   764→              optionB: data.question.optionB || '',
   765→              optionC: data.question.optionC || '',
   766→              optionD: data.question.optionD || '',
   767→              correctOption: data.question.correctOption || 'A',
   768→              explanation: data.question.explanation || '',
   769→              section: data.question.section || 'General',
   770→              negativeMark: data.question.negativeMark || '0',
   771→            };
   772→            // CRITICAL FIX: functional state update
   773→            setQuestions(prev => [...prev, newQ]);
   774→            totalExtracted += 1;
   775→            console.log(`✅ Added 1 question to form`);
   776→            toast.success(`Image ${i + 1}: 1 question extracted!`);
   777→          } else {
   778→            hasError = true;
   779→            console.error(`❌ Extraction failed for image ${i + 1}:`, data.error);
   780→            toast.error(`Image ${i + 1}: ${data.error || 'Failed to extract'}`);
   781→          }
   782→        } catch (err: any) {
   783→          clearTimeout(timeoutId);
   784→          hasError = true;
   785→          console.error(`❌ Fetch error for image ${i + 1}:`, err);
   786→          if (err?.name === 'AbortError') {
   787→            toast.error(`Image ${i + 1}: Timed out (5 min). Try a smaller image.`);
   788→          } else {
   789→            toast.error(`Image ${i + 1}: ${err?.message || 'Failed to extract'}`);
   790→          }
   791→        }
   792→      }
   793→
   794→      if (totalExtracted > 0) {
   795→        toast.success(`🎉 Total: ${totalExtracted} questions extracted from ${fileList.length} image${fileList.length > 1 ? 's' : ''}!`);
   796→      } else if (!hasError) {
   797→        toast.error('No questions could be extracted from the image(s).');
   798→      }
   799→    } finally {
   800→      clearInterval(timerInterval);
   801→      setExtractingImage(false);
   802→      setExtractElapsed(0);
   803→      e.target.value = '';
   804→    }
   805→  }
